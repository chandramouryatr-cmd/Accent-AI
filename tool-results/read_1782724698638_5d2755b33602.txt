  1804→  function drawWave(ts) {
  1805→    if (!waveCtx) return;

  1807→    waveCtx.clearRect(0, 0, waveCanvas.width, waveCanvas.height);
  1808→    const cy = waveCanvas.height * 0.55;
  1809→    waveConfig.forEach(w => {
  1810→      waveCtx.beginPath();
  1811→      for (let x = 0; x <= waveCanvas.width; x += 3) {
  1812→        const y = cy
  1813→          + Math.sin(x * w.freq + wt * w.speed + w.phase) * w.amp
  1814→          + Math.sin(x * w.freq * 1.7 + wt * w.speed * 0.6) * (w.amp * 0.28);
  1815→        x === 0 ? waveCtx.moveTo(x, y) : waveCtx.lineTo(x, y);
  1816→      }
  1817→      waveCtx.strokeStyle = w.color;
  1818→      waveCtx.lineWidth = 1.8;
  1819→      waveCtx.stroke();
  1820→    });
  1821→    waveRaf = requestAnimationFrame(drawWave);
  1822→  }
