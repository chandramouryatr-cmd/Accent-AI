    56→function PracticeContent() {
    57→  const accent = useAppStore((s) => s.accent);
    58→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    59→  const [diff, setDiff] = useState<Difficulty>("medium");
    60→  const [phraseIdx, setPhraseIdx] = useState(0);
    61→  const [speed, setSpeed] = useState<number>(1);
    62→  const [step, setStep] = useState<"listen" | "speak" | "results">("listen");
    63→  const [recording, setRecording] = useState(false);
    64→  const [score, setScore] = useState<number | null>(null);
    65→  // Track previous difficulty so we can reset the practice flow when it
    66→  // changes — using the official "adjust state during render" pattern
    67→  // instead of setState-in-effect (react-hooks/set-state-in-effect).
    68→  const [prevDiff, setPrevDiff] = useState<Difficulty>(diff);
    69→
    70→  const phrases = PHRASES[diff];
    71→  const phrase = phrases[phraseIdx];
    72→  const tip = TIPS[diff][phraseIdx % TIPS[diff].length];
    73→
    74→  const nextPhrase = () => {
    75→    setPhraseIdx((i) => (i + 1) % phrases.length);
    76→    setStep("listen");
    77→    setScore(null);
    78→    setRecording(false);
    79→  };
    80→
    81→  const handleSpeak = useCallback(() => {
    82→    speak(phrase.text, { accent, rate: speed });
    83→    setStep("speak");
    84→  }, [phrase, accent, speed]);
    85→
    86→  const SPEED_PRESETS: { label: string; value: number }[] = [
    87→    { label: "0.6×", value: 0.6 },

    97→      setScore(s);
    98→      setStep("results");
    99→      addSpeakingTime(5);
   100→    } else {
   101→      setRecording(true);
   102→      setTimeout(() => {
   103→        setRecording(false);
   104→        const s = 65 + Math.floor(Math.random() * 30);
   105→        setScore(s);
   106→        setStep("results");
   107→        addSpeakingTime(5);
   108→      }, 4000);
   109→    }
   110→  };
   111→
   112→  // Reset the practice flow when difficulty changes.
   113→  // Official React pattern: adjust state during render based on a tracked
   114→  // previous value (avoids setState-in-effect cascades).
   115→  if (diff !== prevDiff) {
   116→    setPrevDiff(diff);
   117→    setStep("listen");
   118→    setScore(null);
   119→    setRecording(false);
   120→  }

   122→  const stepLabels = ["Listen", "Speak", "Results"];

   124→  return (
   125→    <>
   126→      {/* Step flow */}
   127→      <div className="flex items-center justify-between px-2">
