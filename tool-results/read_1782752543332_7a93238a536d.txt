    15→// ─── Animated Counter Hook ─────────────────────────────────────────────────
    16→function useAnimatedCounter(target: number, duration = 1200): number {
    17→  const [display, setDisplay] = useState(0);
    18→  const prevRef = useRef(0);
    19→  useEffect(() => {
    20→    if (typeof target !== "number" || isNaN(target)) return;
    21→    const start = prevRef.current;
    22→    const diff = target - start;
    23→    if (diff === 0) return;
    24→    const startTime = performance.now();
    25→    let raf: number;
    26→    const tick = (now: number) => {
    27→      const elapsed = now - startTime;
    28→      const progress = Math.min(1, elapsed / duration);
    29→      // ease-out cubic
    30→      const eased = 1 - Math.pow(1 - progress, 3);
    31→      const current = Math.round(start + diff * eased);
    32→      setDisplay(current);
    33→      if (progress < 1) {
    34→        raf = requestAnimationFrame(tick);
    35→      } else {
    36→        prevRef.current = target;
    37→      }
    38→    };
    39→    raf = requestAnimationFrame(tick);
    40→    return () => cancelAnimationFrame(raf);
    41→  }, [target, duration]);
    42→  return display;
    43→}

    45→function AnimatedStatValue({ value }: { value: number | string }) {
    46→  // If value is a string like "85%" or "3m", animate the numeric part
    47→  const str = String(value);
    48→  const match = str.match(/^(\d+(?:\.\d+)?)(.*)$/);
    49→  const num = match ? parseFloat(match[1]) : 0;
    50→  const suffix = match ? match[2] : "";
    51→  const animated = useAnimatedCounter(num);
    52→  if (!match) return <>{str}</>;
    53→  return <>{animated}{suffix}</>;
    54→}

    60→// ─── Tip of the Day ───────────────────────────────────────────────────────
    61→function dayOfYearTipIndex(): number {
    62→  const now = new Date();
    63→  const start = new Date(now.getFullYear(), 0, 0);
    64→  const dayOfYear = Math.floor(
    65→    (now.getTime() - start.getTime()) / 86400000
    66→  );
    67→  return dayOfYear % TIPS.length;
    68→}
