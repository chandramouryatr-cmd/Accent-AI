    29→/**
    30→ * Returns the "primary audio" text for a lesson step — the text that should
    31→ * be spoken when the user presses Space (or wants a quick replay).
    32→ * Returns `null` for step types that have no meaningful auto-play audio.
    33→ */
    34→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    35→  if (!step) return null;
    36→  switch (step.type) {
    37→    case "intro":
    38→      return step.title;
    39→    case "example":
    40→      return step.phrase;
    41→    case "mouth-diagram":
    42→      return step.exampleWord || null;
    43→    case "compare":
    44→      return step.nativePhrase;
    45→    case "stress-bars":
    46→      return step.word;
    47→    case "rhythm":
    48→      return step.phrase;
    49→    case "linking":
    50→      return step.words.join(" ");
    51→    case "shadow":
    52→      return step.phrase;
    53→    case "intonation":
    54→      return step.phrase;
    55→    case "practice":
    56→      return step.phrase;
    57→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    58→    default:
    59→      return null;
    60→  }
    61→}

   131→/** Format seconds as mm:ss */
   132→function formatTimer(seconds: number): string {
   133→  const m = Math.floor(seconds / 60);
   134→  const s = Math.floor(seconds % 60);
   135→  return `${m}:${s.toString().padStart(2, "0")}`;
   136→}

   138→export function LessonModal({ lesson, onClose, onNext }: Props) {
   139→  const [stepIdx, setStepIdx] = useState(0);
   140→  const [direction, setDirection] = useState(1);
   141→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
   142→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
   143→  const [recording, setRecording] = useState(false);
   144→  const [showConfetti, setShowConfetti] = useState(false);
   145→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
   146→  const [showSpaceHint, setShowSpaceHint] = useState(false);
   147→  // Track the previous stepIdx so we can reset per-step interactive state
   148→  // when the step changes — using the official "adjust state during render"
   149→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
   150→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
   151→  const [showNotesPanel, setShowNotesPanel] = useState(false);
   152→  const ttsSpeed = 0.95;
   153→  // Speaking state — drives the header indicator (Bug 3 fix)
   154→  const [speaking, setSpeaking] = useState(false);
   155→
   156→  // ── Lesson Timer ──
   157→  const [timerDisplay, setTimerDisplay] = useState("0:00");
   158→  const timerStartRef = useRef<number>(Date.now());
   159→  const timerAccumulatedRef = useRef<number>(lessonTimerAccumulated.get(lesson.id) ?? 0);
   160→
   161→  // Scroll container ref — reset to top on step change (Bug 1 fix)
   162→  const scrollContainerRef = useRef<HTMLDivElement>(null);
   163→  // Root ref — for attaching one-time TTS unlock listener (Bug 3 fix)
   164→  const rootRef = useRef<HTMLDivElement>(null);
   165→
   166→  const accent = useAppStore((s) => s.accent);
   167→  const completeLesson = useAppStore((s) => s.completeLesson);
   168→  const markReviewed = useAppStore((s) => s.markReviewed);
   169→  const lessons = useAppStore((s) => s.lessons);
   170→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
   171→  const hasLessonNote = useAppStore((s) => (s.lessonNotes[lesson.id] ?? "").trim().length > 0);
   172→
   173→  const step = lesson.steps[stepIdx];
   174→  const isLast = stepIdx === lesson.steps.length - 1;
   175→  const isFirst = stepIdx === 0;
   176→  const totalSteps = lesson.steps.length;
   177→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
   178→
   179→  // Preload voices on mount
   180→  useEffect(() => {
   181→    loadVoices();
   182→  }, []);
   183→
   184→  // Reset scroll position to top whenever the step changes (Bug 1 fix).
   185→  // Without this, scrolling down on step N opens step N+1 already scrolled
   186→  // to the bottom.
   187→  useEffect(() => {
   188→    if (scrollContainerRef.current) {
   189→      scrollContainerRef.current.scrollTop = 0;
   190→    }
   191→  }, [stepIdx]);
   192→
   193→  // Unlock TTS on first user interaction with the modal (Bug 3 fix).
   194→  // Mobile Safari / Chrome Android block speechSynthesis.speak() until a
   195→  // user gesture has occurred. We attach a one-time pointerdown listener
   196→  // to the modal root that calls unlockTTS() and then removes itself.
   197→  useEffect(() => {
   198→    const root = rootRef.current;
   199→    if (!root) return;
   200→    const onFirstInteract = () => {
   201→      unlockTTS();
   202→      root.removeEventListener("pointerdown", onFirstInteract);
   203→    };
   204→    root.addEventListener("pointerdown", onFirstInteract);
   205→    return () => root.removeEventListener("pointerdown", onFirstInteract);
   206→  }, []);
   207→
   208→  // Subscribe to TTS speaking state so the header indicator stays in sync
   209→  // with actual audio playback (Bug 3 fix).
   210→  useEffect(() => {
   211→    const unsub = onSpeakingChange(() => setSpeaking(isSpeaking()));
   212→    setSpeaking(isSpeaking());
   213→    return unsub;
   214→  }, []);
   215→
   216→  // ── Timer tick effect ──
   217→  useEffect(() => {
   218→    timerStartRef.current = Date.now();
   219→    const tick = () => {
   220→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   221→      setTimerDisplay(formatTimer(elapsed));
   222→    };
   223→    tick(); // initial tick
   224→    const id = setInterval(tick, 1000);
   225→    return () => {
   226→      clearInterval(id);
   227→      // Save accumulated time for this lesson when modal unmounts (pause)
   228→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   229→      timerAccumulatedRef.current = elapsed;
   230→      lessonTimerAccumulated.set(lesson.id, elapsed);
   231→    };
   232→  }, [lesson.id]);
   233→
   234→  // Reset per-step interactive state when stepIdx changes.
   235→  // Official React pattern: adjust state during render based on a tracked
   236→  // previous value. React re-renders immediately without committing the
   237→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   238→  if (stepIdx !== prevStepIdx) {
   239→    setPrevStepIdx(stepIdx);
   240→    setQuizAnswer(null);
   241→    setPracticeScore(null);
   242→    setRecording(false);
   243→    // Reset the Space-to-play hint based on whether the new step has audio.
   244→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   245→  }
   246→
   247→  // Stop TTS on unmount or step change
   248→  useEffect(() => {
   249→    return () => stopSpeaking();
   250→  }, [stepIdx]);
   251→
   252→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   253→  // The hint itself is shown/hidden via the render-time adjustment above
   254→  // (avoids setState-in-effect cascades).
   255→  useEffect(() => {
   256→    if (!showSpaceHint) return;
   257→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   258→    return () => clearTimeout(t);
   259→  }, [showSpaceHint]);
   260→
   261→  const handleSpeak = useCallback(
   262→    (text: string) => {
   263→      speak(text, { accent, rate: ttsSpeed });
   264→    },
   265→    [accent, ttsSpeed]
   266→  );
   267→
   268→  const goNext = useCallback(() => {
   269→    if (isLast) {
   270→      // already at completion step
   271→      return;
   272→    }
   273→    setDirection(1);
   274→    setStepIdx((i) => i + 1);
   275→  }, [isLast]);
   276→
   277→  const goPrev = useCallback(() => {
   278→    if (isFirst) return;
   279→    setDirection(-1);
   280→    setStepIdx((i) => i - 1);
   281→  }, [isFirst]);
   282→
   283→  // Mark completion when reaching completion step
   284→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   285→  const alreadyCompleted = lessons[lesson.id]?.completed;
   286→
   287→  const handleComplete = useCallback(() => {
   288→    if (alreadyCompleted) return;
   289→    if (completionStep?.type === "completion") {
   290→      // Calculate total time spent
   291→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   292→      completeLesson(
   293→        lesson.id,
   294→        practiceScore ?? 85,
   295→        completionStep.xp,
   296→        completionStep.badge,
   297→        Math.floor(elapsed)
   298→      );
   299→      // Mark as reviewed since they just completed it
   300→      markReviewed(lesson.id);
   301→      addSpeakingTime(8);
   302→      setShowConfetti(true);
   303→      setTimeout(() => setShowConfetti(false), 4000);
   304→      // Clear timer state for this lesson
   305→      lessonTimerAccumulated.delete(lesson.id);
   306→    }
   307→  }, [alreadyCompleted, completionStep, completeLesson, markReviewed, lesson.id, practiceScore, addSpeakingTime]);
   308→
   309→  // Auto-trigger completion when user reaches the completion step.
   310→  // handleComplete calls setState (setShowConfetti + store actions), so we
   311→  // defer it out of the synchronous effect body to satisfy
   312→  // react-hooks/set-state-in-effect (no cascading renders).
   313→  useEffect(() => {
   314→    if (step?.type !== "completion" || alreadyCompleted) return;
   315→    const id = setTimeout(() => handleComplete(), 0);
   316→    return () => clearTimeout(id);
   317→  }, [step, alreadyCompleted, handleComplete]);
   318→
   319→  // ESC to close, Arrows to navigate, Space to play step audio
   320→  useEffect(() => {
   321→    const onKey = (e: KeyboardEvent) => {
   322→      if (e.key === "Escape") {
   323→        // Close the notes panel first if it's open; otherwise close the lesson.
   324→        if (showNotesPanel) {
   325→          setShowNotesPanel(false);
   326→          return;
   327→        }
   328→        onClose();
   329→        return;
   330→      }
   331→      if (e.key === "ArrowRight") {
   332→        goNext();
   333→        return;
   334→      }
   335→      if (e.key === "ArrowLeft") {
   336→        goPrev();
   337→        return;
   338→      }
   339→      // Space → play the current step's primary audio
   340→      if (e.key === " " || e.code === "Space") {
   341→        // When the notes panel is open, the user is taking notes — don't
   342→        // hijack Space for audio playback.
   343→        if (showNotesPanel) return;
   344→        const target = e.target as HTMLElement | null;
   345→        const isTyping =
   346→          target &&
   347→          (target.tagName === "INPUT" ||
   348→            target.tagName === "TEXTAREA" ||
   349→            target.isContentEditable);
   350→        if (isTyping) return;
   351→
   352→        // Don't capture Space when another overlay is open that needs it
   353→        // (AI Coach chat input, or this very shortcuts overlay).
   354→        const coachOpen = document.querySelector(
   355→          '[aria-label="AccentAI Coach chat"]'
   356→        );
   357→        if (coachOpen) return;
   358→        const overlay = document.getElementById("shortcuts-overlay");
   359→        if (overlay && !overlay.classList.contains("hidden")) return;
   360→
   361→        // If an interactive element (button / link) is focused, let the
   362→        // browser's native Space-to-click behaviour win so users who tab
   363→        // through the UI can still activate controls with Space.
   364→        const active = document.activeElement as HTMLElement | null;
   365→        if (
   366→          active &&
   367→          (active.tagName === "BUTTON" ||
   368→            active.tagName === "A" ||
   369→            active.getAttribute("role") === "button")
   370→        ) {
   371→          return;
   372→        }
   373→
   374→        const text = getPrimaryAudioText(step);
   375→        if (text) {
   376→          e.preventDefault();
   377→          handleSpeak(text);
   378→        }
   379→      }
   380→    };
   381→    window.addEventListener("keydown", onKey);
   382→    return () => window.removeEventListener("keydown", onKey);
   383→  }, [goNext, goPrev, onClose, step, handleSpeak, showNotesPanel]);
   384→
   385→  return (
   386→    <motion.div
   387→      ref={rootRef}
   388→      initial={{ opacity: 0 }}
   389→      animate={{ opacity: 1 }}
   390→      exit={{ opacity: 0 }}
   391→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col overflow-hidden"
   392→    >
   393→      {/* Content */}
   394→      <div className="relative z-10 flex flex-col flex-1 min-h-0">
   395→      {showConfetti && <Confetti count={100} />}
   396→
   397→      {/* Header */}
   398→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg)] safe-top">
   399→        <button
   400→          onClick={onClose}
   401→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   402→          aria-label="Close lesson"
   403→        >
   404→          <X className="w-5 h-5" />
   405→        </button>
   406→        <div className="flex-1 px-3 text-center">
   407→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   408→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   409→          </div>
   410→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   411→        </div>
   412→        <div className="flex items-center gap-1.5">
   413→          <motion.button
   414→            onClick={() => setShowNotesPanel(true)}
   415→            whileTap={{ scale: 0.9 }}
   416→            whileHover={{ scale: 1.05 }}
   417→            className="relative w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition text-[var(--t2)] hover:text-[var(--p3)]"
   418→            aria-label="Open lesson notes"
   419→            aria-expanded={showNotesPanel}
   420→          >
   421→            <NotebookPen style={{ width: 18, height: 18 }} />
   422→            <AnimatePresence>
   423→              {hasLessonNote && (
   424→                <motion.span
   425→                  key="note-dot"
   426→                  initial={{ scale: 0 }}
   427→                  animate={{ scale: 1 }}
   428→                  exit={{ scale: 0 }}
   429→                  transition={{ type: "spring", stiffness: 400, damping: 18 }}
   430→                  className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-[var(--p3)] ring-2 ring-[var(--bg2)]"
   431→                  style={{ boxShadow: "0 0 6px rgba(167,139,250,0.8)" }}
   432→                />
   433→              )}
   434→            </AnimatePresence>
   435→          </motion.button>
   436→          {/* Speaking indicator (Bug 3 fix) — pulsing dot + icon while TTS is active */}
   437→          {speaking && (
   438→            <span
   439→              className="flex items-center gap-1 px-2 py-1 rounded-lg border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]"
   440→              aria-label="Audio playing"
   441→            >
   442→              <motion.span
   443→                animate={{ scale: [1, 1.5, 1], opacity: [1, 0.4, 1] }}
   444→                transition={{ duration: 0.9, repeat: Infinity, ease: "easeInOut" }}
   445→                className="w-1.5 h-1.5 rounded-full bg-[var(--p)]"
   446→              />
   447→              <Volume2 className="w-3 h-3" />
   448→            </span>
   449→          )}
   450→          {/* Timer display */}
   451→          <div className="flex items-center gap-1 px-2 py-1 rounded-lg border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]">
   452→            <span className="text-[10px]">⏱</span>
   453→            <span>{timerDisplay}</span>
   454→          </div>
   455→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   456→          <ProgressRing pct={pct} size={36} stroke={3} />
   457→        </div>
   458→      </div>
   459→
   460→      {/* ── Minimal step progress bar ── */}
   461→      <div className="px-3 py-2 bg-[var(--bg)] border-b border-[var(--border)]">
   462→        <div className="relative max-w-2xl mx-auto" style={{ height: 32 }}>
   463→          {/* Thin gray track */}
   464→          <div className="absolute left-0 right-0 top-1/2 -translate-y-1/2 h-[2px] rounded-full bg-[var(--border)]" />
   465→
   466→          {/* Solid fill up to the current step */}
   467→          <div
   468→            className="absolute top-1/2 -translate-y-1/2 h-[2px] rounded-full bg-[var(--p)] transition-all duration-300 ease-out"
   469→            style={{
   470→              left: 0,
   471→              width: `${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}%`,
   472→            }}
   473→          />
   474→
   475→          {/* Step dots — current = solid p, past = solid gray, upcoming = outline */}
   476→          <div className="absolute inset-0 flex items-center justify-between">
   477→            {lesson.steps.map((s, i) => {
   478→              const isCurrent = i === stepIdx;
   479→              const isPast = i < stepIdx;
   480→              const isUpcoming = i > stepIdx;
   481→              const icon = STEP_ICON[s.type] || "•";
   482→              const stepTitle = s.title || s.type.replace(/-/g, " ");
   483→
   484→              return (
   485→                <button
   486→                  key={i}
   487→                  onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   488→                  className="relative flex items-center justify-center rounded-full cursor-pointer group transition-transform hover:scale-110 active:scale-95"
   489→                  style={{
   490→                    width: isCurrent ? 26 : 20,
   491→                    height: isCurrent ? 26 : 20,
   492→                    fontSize: isCurrent ? 11 : 9,
   493→                    background: isCurrent
   494→                      ? "var(--p)"
   495→                      : isPast
   496→                      ? "var(--t2)"
   497→                      : "var(--bg)",
   498→                    border: isUpcoming ? "1px solid var(--border2)" : "none",
   499→                    color: isCurrent ? "#fff" : isPast ? "var(--bg)" : "var(--t3)",
   500→                  }}
   501→                  aria-label={`Step ${i + 1}: ${stepTitle}`}
   502→                >
   503→                  <span className="leading-none select-none">{icon}</span>
   504→
   505→                  {/* Hover tooltip */}
   506→                  <span className="absolute -top-7 left-1/2 -translate-x-1/2 px-1.5 py-0.5 rounded bg-[var(--bg3)] border border-[var(--border2)] text-[9px] text-[var(--t2)] whitespace-nowrap opacity-0 group-hover:opacity-100 transition-opacity pointer-events-none font-mono z-20">
   507→                    {stepTitle}
   508→                  </span>
   509→                </button>
   510→              );
   511→            })}
   512→          </div>
   513→        </div>
   514→      </div>
   515→
   516→      {/* Step content */}
   517→      <div ref={scrollContainerRef} className="flex-1 overflow-y-auto relative">
   518→        {/* Subtle fade overlay between steps */}
   519→        <AnimatePresence>
   520→          <StepTransitionOverlay key={stepIdx} />

   691→interface StepRendererProps {
   692→  step: LessonStep;
   693→  lesson: Lesson;
   694→  speak: (text: string) => void;
   695→  quizAnswer: number | null;
   696→  setQuizAnswer: (n: number | null) => void;
   697→  practiceScore: number | null;
   698→  setPracticeScore: (n: number | null) => void;
   699→  recording: boolean;
   700→  setRecording: (b: boolean) => void;
   701→  isLast: boolean;
   702→  onComplete: () => void;
   703→  onNext?: () => void;
   704→  timeSpentSeconds: number;
   705→}
   706→
   707→function StepRenderer(props: StepRendererProps) {
   708→  const { step, lesson, speak } = props;
   709→
   710→  switch (step.type) {
   711→    case "intro":
   712→      return <IntroStepView step={step} lesson={lesson} />;
   713→    case "concept":
   714→      return <ConceptStepView step={step} />;
   715→    case "example":
   716→      return <ExampleStepView step={step} speak={speak} />;
   717→    case "mouth-diagram":
   718→      return <MouthDiagram step={step} speak={speak} />;
   719→    case "vowel-chart":
   720→      return <VowelChart step={step} speak={speak} />;
   721→    case "compare":
   722→      return <CompareWave step={step} speak={speak} />;
   723→    case "stress-bars":
   724→      return <StressBars step={step} speak={speak} />;
   725→    case "rhythm":
   726→      return <RhythmBeats step={step} speak={speak} />;
   727→    case "linking":
   728→      return <LinkingDiagram step={step} speak={speak} />;
   729→    case "intonation":
   730→      return <IntonationContour step={step} speak={speak} />;
   731→    case "shadow":
   732→      return <ShadowStepView step={step} speak={speak} />;
   733→    case "tap-pronounce":
   734→      return <TapPronounceStepView step={step} speak={speak} />;
   735→    case "tip":
   736→      return <TipStepView step={step} />;
   737→    case "practice":
   738→      return <PracticeStepView {...props} />;
   739→    case "quiz":
   740→      return <QuizStepView {...props} />;
   741→    case "completion":
   742→      return <CompletionStepView step={step} onNext={props.onNext} timeSpentSeconds={props.timeSpentSeconds} practiceScore={props.practiceScore} />;
   743→    default:
   744→      return <div>Unknown step type</div>;
   745→  }
   746→}

   981→function ShadowStepView({ step, speak }: { step: Extract<LessonStep, { type: "shadow" }>; speak: (t: string) => void }) {
   982→  const [count, setCount] = useState(3);
   983→  useEffect(() => {
   984→    if (count <= 0) return;
   985→    const t = setTimeout(() => setCount((c) => c - 1), 1000);
   986→    return () => clearTimeout(t);
   987→  }, [count]);
   988→  return (
   989→    <div className="space-y-4">
