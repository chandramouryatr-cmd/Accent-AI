    86→    86→
    87→    87→function derivePhonemeMastery(
    88→    88→  lessons: Record<string, { completed?: boolean; score: number }>
    89→    89→): PhonemeScore[] {
    90→    90→  const out: PhonemeScore[] = [];
    91→    91→  for (const [ph, { ids, example }] of Object.entries(PHONEME_LESSONS)) {
    92→    92→    const relevant = ids
    93→    93→      .map((id) => lessons[id])
    94→    94→      .filter((l) => l?.completed);
    95→    95→    if (relevant.length === 0) continue;
    96→    96→    const avg = Math.round(
    97→    97→      relevant.reduce((s, l) => s + l.score, 0) / relevant.length
    98→    98→    );
    99→    99→    const sortedById = [...relevant].sort((a, b) => a.score - b.score);
   100→   100→    const bestLessonId =
   101→   101→      ids.find((id) => lessons[id] === sortedById[0]) || ids[0];
   102→   102→    out.push({
   103→   103→      phoneme: ph,
   104→   104→      example,
   105→   105→      score: avg,
   106→   106→      count: relevant.length,
   107→   107→      lessonId: bestLessonId,
   108→   108→    });
   109→   109→  }
   110→   110→  out.sort((a, b) => a.score - b.score);
   111→   111→  return out;
   112→   112→}

   116→   116→function extractJson(text: string): InsightsPlan | null {
   117→   117→  if (!text) return null;
   118→   118→  const trimmed = text.trim();
   119→   119→
   120→   120→  // Attempt 1: direct parse
   121→   121→  try {
   122→   122→    return normalizePlan(JSON.parse(trimmed));
   123→   123→  } catch {
   124→   124→    /* continue */
   125→   125→  }
   126→   126→
   127→   127→  // Attempt 2: unwrap ```json ... ``` or ``` ... ``` fences
   128→   128→  const fenceMatch = trimmed.match(/```(?:json)?\s*([\s\S]*?)```/i);
   129→   129→  if (fenceMatch) {
   130→   130→    try {
   131→   131→      return normalizePlan(JSON.parse(fenceMatch[1].trim()));
   132→   132→    } catch {
   133→   133→      /* continue */
   134→   134→    }
   135→   135→  }
   136→   136→
   137→   137→  // Attempt 3: grab substring between first { and last }
   138→   138→  const firstBrace = trimmed.indexOf("{");
   139→   139→  const lastBrace = trimmed.lastIndexOf("}");
   140→   140→  if (firstBrace !== -1 && lastBrace > firstBrace) {
   141→   141→    const slice = trimmed.slice(firstBrace, lastBrace + 1);
   142→   142→    try {
   143→   143→      return normalizePlan(JSON.parse(slice));
   144→   144→    } catch {
   145→   145→      /* continue */
   146→   146→    }
   147→   147→  }
   148→   148→
   149→   149→  return null;
   150→   150→}

   152→   152→function normalizePlan(raw: unknown): InsightsPlan | null {
   153→   153→  if (!raw || typeof raw !== "object") return null;
   154→   154→  const obj = raw as Record<string, unknown>;
   155→   155→  const focusAreas = Array.isArray(obj.focusAreas)
   156→   156→    ? (obj.focusAreas as unknown[])
   157→   157→        .map((item) => normalizeFocusArea(item))
   158→   158→        .filter((x): x is FocusArea => x !== null)
   159→   159→    : [];
   160→   160→  const recommendedLessons = Array.isArray(obj.recommendedLessons)
   161→   161→    ? (obj.recommendedLessons as unknown[])
   162→   162→        .map((item) => normalizeRecommendedLesson(item))
   163→   163→        .filter((x): x is RecommendedLesson => x !== null)
   164→   164→    : [];
   165→   165→  const tips = Array.isArray(obj.tips)
   166→   166→    ? (obj.tips as unknown[])
   167→   167→        .map((t) => (typeof t === "string" ? t : String(t ?? "")))
   168→   168→        .filter((s) => s.length > 0)
   169→   169→        .slice(0, 4)
   170→   170→    : [];
   171→   171→
   172→   172→  if (focusAreas.length === 0 && recommendedLessons.length === 0 && tips.length === 0) {
   173→   173→    return null;
   174→   174→  }
   175→   175→  return { focusAreas, recommendedLessons, tips };
   176→   176→}

   178→   178→function normalizeFocusArea(item: unknown): FocusArea | null {
   179→   179→  if (!item || typeof item !== "object") return null;
   180→   180→  const obj = item as Record<string, unknown>;
   181→   181→  const phoneme =
   182→   182→    typeof obj.phoneme === "string"
   183→   183→      ? obj.phoneme.replace(/^\/+|\/+$/g, "").trim()
   184→   184→      : "";
   185→   185→  const score =
   186→   186→    typeof obj.score === "number"
   187→   187→      ? Math.max(0, Math.min(100, Math.round(obj.score)))
   188→   188→      : typeof obj.score === "string"
   189→   189→      ? parseInt(obj.score, 10) || 0
   190→   190→      : 0;
   191→   191→  const reason =
   192→   192→    typeof obj.reason === "string" ? obj.reason : typeof obj.reason === "object" && obj.reason ? JSON.stringify(obj.reason) : "";
   193→   193→  if (!phoneme) return null;
   194→   194→  return { phoneme, score, reason };
   195→   195→}
   196→   196→
   197→   197→function normalizeRecommendedLesson(item: unknown): RecommendedLesson | null {
   198→   198→  if (!item || typeof item !== "object") return null;
   199→   199→  const obj = item as Record<string, unknown>;
   200→   200→  const phase =
   201→   201→    typeof obj.phase === "number"
   202→   202→      ? Math.max(1, Math.min(8, Math.round(obj.phase)))
   203→   203→      : typeof obj.phase === "string"
   204→   204→      ? parseInt(obj.phase, 10) || 1
   205→   205→      : 1;
   206→   206→  const lesson = typeof obj.lesson === "string" ? obj.lesson : "";
   207→   207→  const reason =
   208→   208→    typeof obj.reason === "string"
   209→   209→      ? obj.reason
   210→   210→      : typeof obj.reason === "object" && obj.reason
   211→   211→      ? JSON.stringify(obj.reason)
   212→   212→      : "";
   213→   213→  if (!lesson) return null;
   214→   214→  return { phase, lesson, reason };
   215→   215→}
