    28→/**
    29→ * Returns the "primary audio" text for a lesson step — the text that should
    30→ * be spoken when the user presses Space (or wants a quick replay).
    31→ * Returns `null` for step types that have no meaningful auto-play audio.
    32→ */
    33→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    34→  if (!step) return null;
    35→  switch (step.type) {
    36→    case "intro":
    37→      return step.title;
    38→    case "example":
    39→      return step.phrase;
    40→    case "mouth-diagram":
    41→      return step.exampleWord || null;
    42→    case "compare":
    43→      return step.nativePhrase;
    44→    case "stress-bars":
    45→      return step.word;
    46→    case "rhythm":
    47→      return step.phrase;
    48→    case "linking":
    49→      return step.words.join(" ");
    50→    case "shadow":
    51→      return step.phrase;
    52→    case "intonation":
    53→      return step.phrase;
    54→    case "practice":
    55→      return step.phrase;
    56→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    57→    default:
    58→      return null;
    59→  }
    60→}

   163→/** Format seconds as mm:ss */
   164→function formatTimer(seconds: number): string {
   165→  const m = Math.floor(seconds / 60);
   166→  const s = Math.floor(seconds % 60);
   167→  return `${m}:${s.toString().padStart(2, "0")}`;
   168→}

   170→export function LessonModal({ lesson, onClose, onNext }: Props) {
   171→  const [stepIdx, setStepIdx] = useState(0);
   172→  const [direction, setDirection] = useState(1);
   173→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
   174→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
   175→  const [recording, setRecording] = useState(false);
   176→  const [showConfetti, setShowConfetti] = useState(false);
   177→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
   178→  const [showSpaceHint, setShowSpaceHint] = useState(false);
   179→  // Track the previous stepIdx so we can reset per-step interactive state
   180→  // when the step changes — using the official "adjust state during render"
   181→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
   182→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
   183→  const [showNotesPanel, setShowNotesPanel] = useState(false);
   184→  const [ttsSpeed, setTtsSpeed] = useState(1);
   185→
   186→  // ── Lesson Timer ──
   187→  const [timerDisplay, setTimerDisplay] = useState("0:00");
   188→  const timerStartRef = useRef<number>(Date.now());
   189→  const timerAccumulatedRef = useRef<number>(lessonTimerAccumulated.get(lesson.id) ?? 0);
   190→
   191→  const accent = useAppStore((s) => s.accent);
   192→  const completeLesson = useAppStore((s) => s.completeLesson);
   193→  const markReviewed = useAppStore((s) => s.markReviewed);
   194→  const lessons = useAppStore((s) => s.lessons);
   195→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
   196→  const hasLessonNote = useAppStore((s) => (s.lessonNotes[lesson.id] ?? "").trim().length > 0);
   197→
   198→  const step = lesson.steps[stepIdx];
   199→  const isLast = stepIdx === lesson.steps.length - 1;
   200→  const isFirst = stepIdx === 0;
   201→  const totalSteps = lesson.steps.length;
   202→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
   203→
   204→  // Preload voices on mount
   205→  useEffect(() => {
   206→    loadVoices();
   207→  }, []);
   208→
   209→  // ── Timer tick effect ──
   210→  useEffect(() => {
   211→    timerStartRef.current = Date.now();
   212→    const tick = () => {
   213→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   214→      setTimerDisplay(formatTimer(elapsed));
   215→    };
   216→    tick(); // initial tick
   217→    const id = setInterval(tick, 1000);
   218→    return () => {
   219→      clearInterval(id);
   220→      // Save accumulated time for this lesson when modal unmounts (pause)
   221→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   222→      timerAccumulatedRef.current = elapsed;
   223→      lessonTimerAccumulated.set(lesson.id, elapsed);
   224→    };
   225→  }, [lesson.id]);
   226→
   227→  // Reset per-step interactive state when stepIdx changes.
   228→  // Official React pattern: adjust state during render based on a tracked
   229→  // previous value. React re-renders immediately without committing the
   230→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   231→  if (stepIdx !== prevStepIdx) {
   232→    setPrevStepIdx(stepIdx);
   233→    setQuizAnswer(null);
   234→    setPracticeScore(null);
   235→    setRecording(false);
   236→    // Reset the Space-to-play hint based on whether the new step has audio.
   237→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   238→  }
   239→
   240→  // Stop TTS on unmount or step change
   241→  useEffect(() => {
   242→    return () => stopSpeaking();
   243→  }, [stepIdx]);
   244→
   245→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   246→  // The hint itself is shown/hidden via the render-time adjustment above
   247→  // (avoids setState-in-effect cascades).
   248→  useEffect(() => {
   249→    if (!showSpaceHint) return;
   250→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   251→    return () => clearTimeout(t);
   252→  }, [showSpaceHint]);
   253→
   254→  const handleSpeak = useCallback(
   255→    (text: string) => {
   256→      speak(text, { accent, rate: ttsSpeed });
   257→    },
   258→    [accent, ttsSpeed]
   259→  );
   260→
   261→  const goNext = useCallback(() => {
   262→    if (isLast) {
   263→      // already at completion step
   264→      return;
   265→    }
   266→    setDirection(1);
   267→    setStepIdx((i) => i + 1);
   268→  }, [isLast]);
   269→
   270→  const goPrev = useCallback(() => {
   271→    if (isFirst) return;
   272→    setDirection(-1);
   273→    setStepIdx((i) => i - 1);
   274→  }, [isFirst]);
   275→
   276→  // Mark completion when reaching completion step
   277→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   278→  const alreadyCompleted = lessons[lesson.id]?.completed;
   279→
   280→  const handleComplete = useCallback(() => {
   281→    if (alreadyCompleted) return;
   282→    if (completionStep?.type === "completion") {
   283→      // Calculate total time spent
   284→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   285→      completeLesson(
   286→        lesson.id,
   287→        practiceScore ?? 85,
   288→        completionStep.xp,
   289→        completionStep.badge,
   290→        Math.floor(elapsed)
   291→      );
   292→      // Mark as reviewed since they just completed it
   293→      markReviewed(lesson.id);
   294→      addSpeakingTime(8);
   295→      setShowConfetti(true);
   296→      setTimeout(() => setShowConfetti(false), 4000);
   297→      // Clear timer state for this lesson
   298→      lessonTimerAccumulated.delete(lesson.id);
   299→    }
   300→  }, [alreadyCompleted, completionStep, completeLesson, markReviewed, lesson.id, practiceScore, addSpeakingTime]);
   301→
   302→  // Auto-trigger completion when user reaches the completion step.
   303→  // handleComplete calls setState (setShowConfetti + store actions), so we
   304→  // defer it out of the synchronous effect body to satisfy
   305→  // react-hooks/set-state-in-effect (no cascading renders).
   306→  useEffect(() => {
   307→    if (step?.type !== "completion" || alreadyCompleted) return;
   308→    const id = setTimeout(() => handleComplete(), 0);
   309→    return () => clearTimeout(id);
   310→  }, [step, alreadyCompleted, handleComplete]);
   311→
   312→  // ESC to close, Arrows to navigate, Space to play step audio
   313→  useEffect(() => {
   314→    const onKey = (e: KeyboardEvent) => {
   315→      if (e.key === "Escape") {
   316→        // Close the notes panel first if it's open; otherwise close the lesson.
   317→        if (showNotesPanel) {
   318→          setShowNotesPanel(false);
   319→          return;
   320→        }
   321→        onClose();
   322→        return;
   323→      }
   324→      if (e.key === "ArrowRight") {
   325→        goNext();
   326→        return;
   327→      }
   328→      if (e.key === "ArrowLeft") {
   329→        goPrev();
   330→        return;
   331→      }
   332→      // Space → play the current step's primary audio
   333→      if (e.key === " " || e.code === "Space") {
   334→        // When the notes panel is open, the user is taking notes — don't
   335→        // hijack Space for audio playback.
   336→        if (showNotesPanel) return;
   337→        const target = e.target as HTMLElement | null;
   338→        const isTyping =
   339→          target &&
   340→          (target.tagName === "INPUT" ||
   341→            target.tagName === "TEXTAREA" ||
   342→            target.isContentEditable);
   343→        if (isTyping) return;
   344→
   345→        // Don't capture Space when another overlay is open that needs it
   346→        // (AI Coach chat input, or this very shortcuts overlay).
   347→        const coachOpen = document.querySelector(
   348→          '[aria-label="AccentAI Coach chat"]'
   349→        );
   350→        if (coachOpen) return;
   351→        const overlay = document.getElementById("shortcuts-overlay");
   352→        if (overlay && !overlay.classList.contains("hidden")) return;
   353→
   354→        // If an interactive element (button / link) is focused, let the
   355→        // browser's native Space-to-click behaviour win so users who tab
   356→        // through the UI can still activate controls with Space.
   357→        const active = document.activeElement as HTMLElement | null;
   358→        if (
   359→          active &&
   360→          (active.tagName === "BUTTON" ||
   361→            active.tagName === "A" ||
   362→            active.getAttribute("role") === "button")
   363→        ) {
   364→          return;
   365→        }
   366→
   367→        const text = getPrimaryAudioText(step);
   368→        if (text) {
   369→          e.preventDefault();
   370→          handleSpeak(text);
   371→        }
   372→      }
   373→    };
   374→    window.addEventListener("keydown", onKey);
   375→    return () => window.removeEventListener("keydown", onKey);
   376→  }, [goNext, goPrev, onClose, step, handleSpeak, showNotesPanel]);
   377→
   378→  const currentCategory = getStepCategory(step?.type ?? "");
   379→  const currentTint = CATEGORY_TINT[currentCategory];
   380→  const currentGlow = CATEGORY_GLOW[currentCategory];
   381→
   382→  return (
   383→    <motion.div
   384→      initial={{ opacity: 0 }}
   385→      animate={{ opacity: 1 }}
   386→      exit={{ opacity: 0 }}
   387→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col relative overflow-hidden"
   388→    >
   389→      {/* Animated step-type background tint */}
   390→      <motion.div
   391→        key={currentCategory}
   392→        initial={{ opacity: 0 }}
   393→        animate={{ opacity: 1 }}
   394→        exit={{ opacity: 0 }}
   395→        transition={{ duration: 0.6 }}
   396→        className="absolute inset-0 pointer-events-none z-0"
   397→        style={{ background: `radial-gradient(ellipse at 50% 0%, ${currentTint} 0%, transparent 70%)` }}
   398→      />
   399→      {/* Content sits above the tint */}
   400→      <div className="relative z-10 flex flex-col flex-1 min-h-0">
   401→      {showConfetti && <Confetti count={100} />}
   402→
   403→      {/* Header */}
   404→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   405→        <button
   406→          onClick={onClose}
   407→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   408→          aria-label="Close lesson"
   409→        >
   410→          <X className="w-5 h-5" />
   411→        </button>
   412→        <div className="flex-1 px-3 text-center">
   413→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   414→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   415→          </div>
   416→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   417→        </div>
   418→        <div className="flex items-center gap-1.5">
   419→          <motion.button
   420→            onClick={() => setShowNotesPanel(true)}
   421→            whileTap={{ scale: 0.9 }}
   422→            whileHover={{ scale: 1.05 }}
   423→            className="relative w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition text-[var(--t2)] hover:text-[var(--p3)]"
   424→            aria-label="Open lesson notes"
   425→            aria-expanded={showNotesPanel}
   426→          >
   427→            <NotebookPen style={{ width: 18, height: 18 }} />
   428→            <AnimatePresence>
   429→              {hasLessonNote && (
   430→                <motion.span
   431→                  key="note-dot"
   432→                  initial={{ scale: 0 }}
   433→                  animate={{ scale: 1 }}
   434→                  exit={{ scale: 0 }}
   435→                  transition={{ type: "spring", stiffness: 400, damping: 18 }}
   436→                  className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-[var(--p3)] ring-2 ring-[var(--bg2)]"
   437→                  style={{ boxShadow: "0 0 6px rgba(167,139,250,0.8)" }}
   438→                />
   439→              )}
   440→            </AnimatePresence>
   441→          </motion.button>
   442→          {/* Timer display */}
   443→          <div className="flex items-center gap-1 px-2 py-1 rounded-lg bg-[var(--card)]/60 border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]">
   444→            <span className="text-[10px]">⏱</span>
   445→            <span>{timerDisplay}</span>
   446→          </div>
   447→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   448→          <ProgressRing pct={pct} size={36} stroke={3} />
   449→        </div>
   450→      </div>
   451→
   452→      {/* ── Enhanced animated step progress bar ── */}
   453→      <div className="px-3 py-2 bg-[var(--bg2)]/70 border-b border-[var(--border)]">
   454→        <div className="relative max-w-2xl mx-auto" style={{ height: 44 }}>
   455→          {/* Gradient connecting line (background track) */}
   456→          <div className="absolute left-0 right-0 top-1/2 -translate-y-1/2 h-[3px] rounded-full bg-[var(--border)]" />
   457→
   458→          {/* Animated gradient fill line — draws from left to the current step */}
   459→          <motion.div
   460→            className="absolute top-1/2 -translate-y-1/2 h-[3px] rounded-full"
   461→            style={{
   462→              left: 0,
   463→              width: `calc(${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}% + 0px)`,
   464→              background: "linear-gradient(90deg, var(--p), var(--p2))",
   465→              boxShadow: "0 0 8px rgba(99,102,241,0.4)",
   466→            }}
   467→            initial={{ scaleX: 0, transformOrigin: "left" }}
   468→            animate={{ scaleX: 1 }}
   469→            transition={{ duration: 0.5, ease: "easeOut" }}
   470→          />
   471→
   472→          {/* Shimmer sweep on the filled line */}
   473→          <motion.div
   474→            className="absolute top-1/2 -translate-y-1/2 h-[3px] rounded-full"
   475→            style={{
   476→              left: 0,
   477→              width: `calc(${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}% + 0px)`,
   478→              background: "linear-gradient(90deg, transparent, rgba(255,255,255,0.25), transparent)",
   479→              backgroundSize: "200% 100%",

   482→            transition={{ duration: 2.5, repeat: Infinity, repeatDelay: 3, ease: "easeInOut" }}
   483→          />
   484→
   485→          {/* Step dots row */}
   486→          <div className="absolute inset-0 flex items-center justify-between">
   487→            {lesson.steps.map((s, i) => {
   488→              const isCurrent = i === stepIdx;
   489→              const isPast = i < stepIdx;
   490→              const isUpcoming = i > stepIdx;
   491→              const cat = getStepCategory(s.type);
   492→              const glow = CATEGORY_GLOW[cat];
   493→              const icon = STEP_ICON[s.type] || "•";
   494→              const stepTitle = s.title || s.type.replace(/-/g, " ");
   495→
   496→              return (
   497→                <motion.button
   498→                  key={i}
   499→                  onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   500→                  className="relative flex items-center justify-center rounded-full cursor-pointer group"

   729→interface StepRendererProps {
   730→  step: LessonStep;
   731→  speak: (text: string) => void;
   732→  ttsSpeed: number;
   733→  setTtsSpeed: (s: number) => void;
   734→  quizAnswer: number | null;
   735→  setQuizAnswer: (n: number | null) => void;
   736→  practiceScore: number | null;
   737→  setPracticeScore: (n: number | null) => void;
   738→  recording: boolean;
   739→  setRecording: (b: boolean) => void;
   740→  isLast: boolean;
   741→  onComplete: () => void;
   742→  onNext?: () => void;
   743→  timeSpentSeconds: number;
   744→}
   745→
   746→function StepRenderer(props: StepRendererProps) {
   747→  const { step, speak, ttsSpeed, setTtsSpeed } = props;
   748→
   749→  switch (step.type) {
   750→    case "intro":
   751→      return <IntroStepView step={step} speak={speak} ttsSpeed={ttsSpeed} setTtsSpeed={setTtsSpeed} />;
   752→    case "concept":
   753→      return <ConceptStepView step={step} />;
   754→    case "example":
   755→      return <ExampleStepView step={step} speak={speak} />;
   756→    case "mouth-diagram":
   757→      return <MouthDiagram step={step} speak={speak} />;
   758→    case "vowel-chart":
   759→      return <VowelChart step={step} speak={speak} />;
   760→    case "compare":
   761→      return <CompareWave step={step} speak={speak} />;
   762→    case "stress-bars":
   763→      return <StressBars step={step} speak={speak} />;
   764→    case "rhythm":
   765→      return <RhythmBeats step={step} speak={speak} />;
   766→    case "linking":
   767→      return <LinkingDiagram step={step} speak={speak} />;
   768→    case "intonation":
   769→      return <IntonationContour step={step} speak={speak} />;
   770→    case "shadow":
   771→      return <ShadowStepView step={step} speak={speak} />;
   772→    case "tap-pronounce":
   773→      return <TapPronounceStepView step={step} speak={speak} />;
   774→    case "tip":
   775→      return <TipStepView step={step} />;
   776→    case "practice":
   777→      return <PracticeStepView {...props} />;
   778→    case "quiz":
   779→      return <QuizStepView {...props} />;
   780→    case "completion":
   781→      return <CompletionStepView step={step} onNext={props.onNext} timeSpentSeconds={props.timeSpentSeconds} />;
   782→    default:
   783→      return <div>Unknown step type</div>;
   784→  }
   785→}
