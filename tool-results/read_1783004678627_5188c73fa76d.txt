    85→// ─── Phoneme mastery derivation ──────────────────────────────────────────
    86→
    87→function derivePhonemeMastery(
    88→  lessons: Record<string, { completed?: boolean; score: number }>
    89→): PhonemeScore[] {
    90→  const out: PhonemeScore[] = [];
    91→  for (const [ph, { ids, example }] of Object.entries(PHONEME_LESSONS)) {
    92→    const relevant = ids
    93→      .map((id) => lessons[id])
    94→      .filter((l) => l?.completed);
    95→    if (relevant.length === 0) continue;
    96→    const avg = Math.round(
    97→      relevant.reduce((s, l) => s + l.score, 0) / relevant.length
    98→    );
    99→    const sortedById = [...relevant].sort((a, b) => a.score - b.score);
   100→    const bestLessonId =
   101→      ids.find((id) => lessons[id] === sortedById[0]) || ids[0];
   102→    out.push({
   103→      phoneme: ph,
   104→      example,
   105→      score: avg,
   106→      count: relevant.length,
   107→      lessonId: bestLessonId,
   108→    });
   109→  }
   110→  out.sort((a, b) => a.score - b.score);
   111→  return out;
   112→}

   114→// ─── JSON parsing helpers (robust against markdown / extra text) ─────────
   115→
   116→function extractJson(text: string): InsightsPlan | null {
   117→  if (!text) return null;
   118→  const trimmed = text.trim();
   119→
   120→  // Attempt 1: direct parse
   121→  try {
   122→    return normalizePlan(JSON.parse(trimmed));
   123→  } catch {
   124→    /* continue */
   125→  }
   126→
   127→  // Attempt 2: unwrap ```json ... ``` or ``` ... ``` fences
   128→  const fenceMatch = trimmed.match(/```(?:json)?\s*([\s\S]*?)```/i);
   129→  if (fenceMatch) {
   130→    try {
   131→      return normalizePlan(JSON.parse(fenceMatch[1].trim()));
   132→    } catch {
   133→      /* continue */
   134→    }
   135→  }
   136→
   137→  // Attempt 3: grab substring between first { and last }
   138→  const firstBrace = trimmed.indexOf("{");
   139→  const lastBrace = trimmed.lastIndexOf("}");
   140→  if (firstBrace !== -1 && lastBrace > firstBrace) {
   141→    const slice = trimmed.slice(firstBrace, lastBrace + 1);
   142→    try {
   143→      return normalizePlan(JSON.parse(slice));
   144→    } catch {
   145→      /* continue */
   146→    }
   147→  }
   148→
   149→  return null;
   150→}

   152→function normalizePlan(raw: unknown): InsightsPlan | null {
   153→  if (!raw || typeof raw !== "object") return null;
   154→  const obj = raw as Record<string, unknown>;
   155→  const focusAreas = Array.isArray(obj.focusAreas)
   156→    ? (obj.focusAreas as unknown[])
   157→        .map((item) => normalizeFocusArea(item))
   158→        .filter((x): x is FocusArea => x !== null)
   159→    : [];
   160→  const recommendedLessons = Array.isArray(obj.recommendedLessons)
   161→    ? (obj.recommendedLessons as unknown[])
   162→        .map((item) => normalizeRecommendedLesson(item))
   163→        .filter((x): x is RecommendedLesson => x !== null)
   164→    : [];
   165→  const tips = Array.isArray(obj.tips)
   166→    ? (obj.tips as unknown[])
   167→        .map((t) => (typeof t === "string" ? t : String(t ?? "")))
   168→        .filter((s) => s.length > 0)
   169→        .slice(0, 4)
   170→    : [];
   171→
   172→  if (focusAreas.length === 0 && recommendedLessons.length === 0 && tips.length === 0) {
   173→    return null;
   174→  }
   175→  return { focusAreas, recommendedLessons, tips };
   176→}
   177→
   178→function normalizeFocusArea(item: unknown): FocusArea | null {
   179→  if (!item || typeof item !== "object") return null;
   180→  const obj = item as Record<string, unknown>;
   181→  const phoneme =
   182→    typeof obj.phoneme === "string"
   183→      ? obj.phoneme.replace(/^\/+|\/+$/g, "").trim()
   184→      : "";
   185→  const score =
   186→    typeof obj.score === "number"
   187→      ? Math.max(0, Math.min(100, Math.round(obj.score)))
   188→      : typeof obj.score === "string"
   189→      ? parseInt(obj.score, 10) || 0
   190→      : 0;
   191→  const reason =
   192→    typeof obj.reason === "string" ? obj.reason : typeof obj.reason === "object" && obj.reason ? JSON.stringify(obj.reason) : "";
   193→  if (!phoneme) return null;
   194→  return { phoneme, score, reason };
   195→}
   196→
   197→function normalizeRecommendedLesson(item: unknown): RecommendedLesson | null {
   198→  if (!item || typeof item !== "object") return null;
   199→  const obj = item as Record<string, unknown>;
   200→  const phase =
   201→    typeof obj.phase === "number"
   202→      ? Math.max(1, Math.min(8, Math.round(obj.phase)))
   203→      : typeof obj.phase === "string"
   204→      ? parseInt(obj.phase, 10) || 1
   205→      : 1;
   206→  const lesson = typeof obj.lesson === "string" ? obj.lesson : "";
   207→  const reason =
   208→    typeof obj.reason === "string"
   209→      ? obj.reason
   210→      : typeof obj.reason === "object" && obj.reason
   211→      ? JSON.stringify(obj.reason)
   212→      : "";
   213→  if (!lesson) return null;
   214→  return { phase, lesson, reason };
   215→}

   379→export function CoachInsights() {
   380→  const lessons = useAppStore((s) => s.lessons);
   381→  const xp = useAppStore((s) => s.xp);
   382→  const streak = useAppStore((s) => s.streak);
   383→  const accent = useAppStore((s) => s.accent);
   384→  const setActiveLesson = useAppStore((s) => s.setActiveLesson);
   385→
   386→  const [view, setView] = useState<ViewState>("idle");
   387→  const [parsed, setParsed] = useState<InsightsPlan | null>(null);
   388→  const [rawText, setRawText] = useState<string | null>(null);
   389→  const [error, setError] = useState<string | null>(null);
   390→
   391→  const abortRef = useRef<AbortController | null>(null);
   392→
   393→  // Derive phoneme mastery data (weakest first)
   394→  const phonemeMastery = useMemo(() => derivePhonemeMastery(lessons), [lessons]);
   395→
   396→  // Build a signature so cache invalidates if phoneme scores change
   397→  const signature = useMemo(() => {
   398→    return phonemeMastery
   399→      .map((p) => `${p.phoneme}:${p.score}:${p.count ?? 0}`)
   400→      .join("|");
   401→  }, [phonemeMastery]);
   402→
   403→  const completedCount = useMemo(
   404→    () => Object.values(lessons).filter((l) => l?.completed).length,
   405→    [lessons]
   406→  );
   407→
   408→  // Load cached insights for today on mount
   409→  useEffect(() => {
   410→    if (typeof window === "undefined") return;
   411→    try {
   412→      const raw = window.localStorage.getItem(storageKeyForToday());
   413→      if (!raw) return;
   414→      const cached = JSON.parse(raw) as CachedInsight;
   415→      // Only restore if the phoneme signature hasn't materially changed
   416→      // (allow restore even if signature differs by a little — focus on date key)
   417→      if (cached && (cached.parsed || cached.rawText)) {
   418→        setParsed(cached.parsed);
   419→        setRawText(cached.rawText);
   420→        setView("success");
   421→      }
   422→    } catch {
   423→      /* ignore malformed cache */
   424→    }
   425→  }, []);
   426→
   427→  // Cleanup any in-flight request on unmount
   428→  useEffect(() => {
   429→    return () => {
   430→      if (abortRef.current) {
   431→        abortRef.current.abort();
   432→        abortRef.current = null;
   433→      }
   434→    };
   435→  }, []);
   436→
   437→  const handleGetInsights = useCallback(async () => {
   438→    setError(null);
   439→
   440→    // Abort any previous request
   441→    if (abortRef.current) abortRef.current.abort();
   442→    const controller = new AbortController();
   443→    abortRef.current = controller;
   444→
   445→    setView("loading");
   446→    setParsed(null);
   447→    setRawText(null);
   448→
   449→    const timeoutId = setTimeout(() => {
   450→      controller.abort();
   451→    }, FIRST_TOKEN_TIMEOUT_MS);
   452→
   453→    // Build completed lessons list
   454→    const completedLessons = Object.entries(lessons)
   455→      .filter(([, p]) => p?.completed)
   456→      .map(([id]) => id);
   457→
   458→    // Top 5 weakest phonemes for the model
   459→    const topPhonemes = phonemeMastery.slice(0, 5).map((p) => ({
   460→      phoneme: p.phoneme,
   461→      score: p.score,
   462→      example: p.example,
   463→      count: p.count,
   464→    }));
   465→
   466→    try {
   467→      const res = await fetch("/api/ai-coach", {
   468→        method: "POST",
   469→        headers: { "Content-Type": "application/json" },
   470→        body: JSON.stringify({
   471→          mode: "insights",
   472→          messages: [],
   473→          context: {
   474→            accent,
   475→            xp,
   476→            streak,
   477→            completedLessons,
   478→            phonemeMastery: topPhonemes,
   479→          },
   480→        }),
   481→        signal: controller.signal,
   482→      });
   483→
   484→      if (!res.ok) {
   485→        let detail = `Request failed (${res.status})`;
   486→        try {
   487→          const data = await res.json();
   488→          if (data?.error) detail = data.error;
   489→        } catch {
   490→          /* ignore */
   491→        }
   492→        throw new Error(detail);
   493→      }
   494→
   495→      if (!res.body) {
   496→        throw new Error("No response body received.");
   497→      }
   498→
   499→      // Consume SSE stream and accumulate text
   500→      const reader = res.body.getReader();
   501→      const decoder = new TextDecoder();
   502→      let accumulated = "";
   503→      let buffer = "";
   504→      let firstTokenReceived = false;
   505→
   506→      while (true) {
   507→        const { done, value } = await reader.read();
   508→        if (done) break;
   509→
   510→        buffer += decoder.decode(value, { stream: true });
   511→        const lines = buffer.split("\n");
   512→        buffer = lines.pop() || "";
   513→
   514→        for (const line of lines) {
   515→          const trimmedLine = line.trim();
   516→          if (!trimmedLine || !trimmedLine.startsWith("data:")) continue;
   517→
   518→          const dataStr = trimmedLine.slice(5).trim();
   519→          if (dataStr === "[DONE]") {
   520→            break;
   521→          }
   522→
   523→          try {
   524→            const parsedChunk = JSON.parse(dataStr);
   525→            if (parsedChunk.error) {
   526→              throw new Error(parsedChunk.error);
   527→            }
   528→            if (typeof parsedChunk.token === "string") {
   529→              if (!firstTokenReceived) {
   530→                firstTokenReceived = true;
   531→                clearTimeout(timeoutId);
   532→              }
   533→              accumulated += parsedChunk.token;
   534→            }
   535→          } catch (parseErr) {
   536→            // Re-throw our own thrown errors (with a real message)
   537→            if (
   538→              parseErr instanceof Error &&
   539→              !parseErr.message.startsWith("Unexpected")
   540→            ) {
   541→              throw parseErr;
   542→            }
   543→            // Skip malformed JSON lines (expected for partial SSE chunks)
   544→          }
   545→        }
   546→      }
   547→
   548→      clearTimeout(timeoutId);
   549→
   550→      const text = accumulated.trim();
   551→      if (!text) {
   552→        throw new Error("The coach didn't return any insights. Please try again.");
   553→      }
   554→
   555→      const plan = extractJson(text);
   556→      setParsed(plan);
   557→      setRawText(plan ? null : text);
   558→      setView("success");
   559→
   560→      // Cache the result (date-keyed)
   561→      try {
   562→        const payload: CachedInsight = {
   563→          parsed: plan,
   564→          rawText: plan ? null : text,
   565→          generatedAt: Date.now(),
   566→          signature,
   567→        };
   568→        window.localStorage.setItem(
   569→          storageKeyForToday(),
   570→          JSON.stringify(payload)
   571→        );
   572→      } catch {
   573→        /* ignore storage errors */
   574→      }
   575→    } catch (err) {
   576→      clearTimeout(timeoutId);
   577→      if (err instanceof DOMException && err.name === "AbortError") {
   578→        setError("Request timed out — the coach took too long. Please try again.");
   579→      } else {
   580→        const msg = err instanceof Error ? err.message : "Something went wrong.";
   581→        setError(msg);
   582→      }
   583→      setView("error");
   584→    } finally {
   585→      abortRef.current = null;
   586→    }
   587→  }, [lessons, phonemeMastery, accent, xp, streak, signature]);
   588→
   589→  const handleOpenLesson = useCallback(
   590→    (lessonTitle: string) => {
   591→      const lesson = ALL_LESSONS.find((l) => l.title === lessonTitle);
   592→      if (lesson) {
   593→        setActiveLesson(lesson.id);
   594→      }
   595→    },
   596→    [setActiveLesson]
   597→  );
   598→
   599→  // ─── Render ──────────────────────────────────────────────────────────
   600→
   601→  return (
   602→    <motion.section
   603→      initial={{ opacity: 0, y: 12 }}
   604→      animate={{ opacity: 1, y: 0 }}
   605→      transition={{ duration: 0.4, ease: "easeOut" }}
   606→      className="relative"
   607→      aria-label="Coach Insights"
   608→    >
   609→      <div className="flex items-center justify-between mb-2">
   610→        <h2 className="font-d text-base font-bold flex items-center gap-2 text-[var(--t1)]">
   611→          <Sparkles className="w-4 h-4 text-[var(--t1)]" />
   612→          <span>Coach Insights</span>
   613→        </h2>
   614→        {view === "success" && (
   615→          <button
   616→            onClick={handleGetInsights}
   617→            className="text-[10px] font-mono uppercase tracking-wider flex items-center gap-1.5 px-2.5 py-1.5 rounded-full bg-[var(--card)] border border-[var(--border)] text-[var(--t2)] hover:text-[var(--t1)] hover:border-[var(--border2)] transition"
   618→            aria-label="Refresh insights"
   619→          >
   620→            <RefreshCw className="w-3 h-3" />
   621→            Refresh
   622→          </button>
   623→        )}
   624→      </div>
   625→
   626→      {/* Solid card: white bg + thin border, no gradient, no blur, no orbs */}
   627→      <div className="bg-[var(--card)] border border-[var(--border)] rounded-xl">
   628→        <div className="p-5">
   629→          <AnimatePresence mode="wait">
   630→            {/* ─── IDLE STATE ─── */}
   631→            {view === "idle" && (
   632→              <motion.div
   633→                key="idle"
   634→                initial={{ opacity: 0 }}
   635→                animate={{ opacity: 1 }}
   636→                exit={{ opacity: 0, y: -8 }}
   637→                className="text-center py-4"
   638→              >
   639→                <div className="inline-flex items-center justify-center w-16 h-16 rounded-xl mb-3 bg-[var(--card-h)] border border-[var(--border)]">
   640→                  <Sparkles className="w-7 h-7 text-[var(--t1)]" />
   641→                </div>
   642→                <div className="font-d text-base font-bold text-[var(--t1)] mb-1">
   643→                  Get your AI practice plan
   644→                </div>
   645→                <p className="text-xs text-[var(--t2)] leading-relaxed max-w-xs mx-auto mb-4">
   646→                  {completedCount === 0
   647→                    ? "I'll analyze your starting point and recommend the perfect first lessons."
   648→                    : phonemeMastery.length === 0
   649→                    ? "Complete a few lessons so I can spot your weakest sounds."
   650→                    : `Based on ${phonemeMastery.length} sound${
   651→                        phonemeMastery.length !== 1 ? "s" : ""
   652→                      } tracked — I'll build a personalized plan in seconds.`}
   653→                </p>
   654→                <motion.button
   655→                  onClick={handleGetInsights}
   656→                  whileTap={{ scale: 0.98 }}
   657→                  className="inline-flex items-center gap-2 bg-[var(--p)] text-white rounded-xl px-5 py-2.5 text-sm font-semibold hover:opacity-80 transition"
   658→                >
   659→                  <Zap className="w-4 h-4" />
   660→                  Get AI Insights
   661→                </motion.button>
   662→              </motion.div>
   663→            )}
   664→
   665→            {/* ─── LOADING STATE ─── */}
   666→            {view === "loading" && (
   667→              <motion.div
   668→                key="loading"
   669→                initial={{ opacity: 0 }}
   670→                animate={{ opacity: 1 }}
   671→                exit={{ opacity: 0 }}
   672→              >
   673→                <LoadingState />
   674→              </motion.div>
   675→            )}
   676→
   677→            {/* ─── ERROR STATE ─── */}
   678→            {view === "error" && (
   679→              <motion.div
   680→                key="error"
   681→                initial={{ opacity: 0, y: 8 }}
   682→                animate={{ opacity: 1, y: 0 }}
   683→                exit={{ opacity: 0, y: -8 }}
   684→                className="flex flex-col items-center text-center py-6"

   694→                </p>
   695→                <button
   696→                  onClick={handleGetInsights}
   697→                  className="inline-flex items-center gap-2 px-4 py-2 rounded-xl text-xs font-semibold border border-[var(--border2)] text-[var(--t1)] hover:bg-[var(--card-h)] transition"
   698→                >
   699→                  <RefreshCw className="w-3.5 h-3.5" />
   700→                  Try again
   701→                </button>
   702→              </motion.div>
   703→            )}
   704→
   705→            {/* ─── SUCCESS STATE ─── */}
   706→            {view === "success" && (
   707→              <motion.div
   708→                key="success"
   709→                initial={{ opacity: 0 }}
   710→                animate={{ opacity: 1 }}
   711→                exit={{ opacity: 0 }}
   712→                className="space-y-4"
   713→              >
   714→                {/* If we have structured JSON → render 3 sections */}
   715→                {parsed ? (
   716→                  <>
   717→                    {/* Focus Areas */}

   740→                        </div>

   742→                    )}
   743→
   744→                    {/* Recommended Lessons */}
   745→                    {parsed.recommendedLessons.length > 0 && (

   761→                                key={`${rec.lesson}-${i}`}
   762→                                rec={rec}
   763→                                index={i}
   764→                                onOpen={handleOpenLesson}
   765→                              />
   766→                            ))}
   767→                        </div>
   768→                      </section>
   769→                    )}
   770→
   771→                    {/* Practice Tips */}
   772→                    {parsed.tips.length > 0 && (
   773→                      <section>
   774→                        <div className="flex items-center gap-2 mb-2.5">
   775→                          <Lightbulb className="w-3.5 h-3.5 text-[var(--t1)]" />
   776→                          <h3 className="font-d text-sm font-bold text-[var(--t1)]">
   777→                            Practice Tips
   778→                          </h3>
   779→                        </div>
   780→                        <ul className="space-y-2.5">
   781→                          {parsed.tips.map((tip, i) => (
   782→                            <TipItem key={i} tip={tip} index={i} />
   783→                          ))}
   784→                        </ul>
   785→                      </section>
   786→                    )}
   787→
   788→                    {/* Footer note */}
   789→                    <div className="text-[10px] text-[var(--t3)] font-mono text-center pt-2 border-t border-[var(--border)]">
   790→                      Generated by AccentAI Coach ·{" "}
   791→                      <button
   792→                        onClick={handleGetInsights}
   793→                        className="text-[var(--t1)] hover:underline underline-offset-2 transition"
   794→                      >
   795→                        regenerate
   796→                      </button>
   797→                    </div>
   798→                  </>
   799→                ) : (
   800→                  // Fallback: raw text response
   801→                  <section>
   802→                    <div className="flex items-center gap-2 mb-2.5">
   803→                      <Sparkles className="w-3.5 h-3.5 text-[var(--t1)]" />
   804→                      <h3 className="font-d text-sm font-bold text-[var(--t1)]">

   806→                      </h3>
   807→                    </div>

   809→                      {rawText}
   810→                    </div>
   811→                    <div className="text-[10px] text-[var(--t3)] font-mono text-center pt-2 mt-2">
   812→                      Generated by AccentAI Coach
   813→                    </div>
   814→                  </section>
   815→                )}
   816→              </motion.div>
   817→            )}
   818→          </AnimatePresence>
   819→        </div>
   820→      </div>
   821→    </motion.section>
   822→  );
   823→}
