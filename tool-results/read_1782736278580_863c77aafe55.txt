    28→  wave: "🌊",
    29→  mouth: "👄",
    30→  "ipa-chart": "🔤",
    31→  "vowel-chart": "🎯",
    32→  "compare-wave": "📊",

    41→
    42→export function LessonModal({ lesson, onClose, onNext }: Props) {
    43→  const [stepIdx, setStepIdx] = useState(0);
    44→  const [direction, setDirection] = useState(1);
    45→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
    46→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
    47→  const [recording, setRecording] = useState(false);
    48→  const [showConfetti, setShowConfetti] = useState(false);
    49→  // Track the previous stepIdx so we can reset per-step interactive state
    50→  // when the step changes — using the official "adjust state during render"
    51→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
    52→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
    53→  const accent = useAppStore((s) => s.accent);
    54→  const completeLesson = useAppStore((s) => s.completeLesson);
    55→  const lessons = useAppStore((s) => s.lessons);
    56→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    57→
    58→  const step = lesson.steps[stepIdx];
    59→  const isLast = stepIdx === lesson.steps.length - 1;
    60→  const isFirst = stepIdx === 0;
    61→  const totalSteps = lesson.steps.length;
    62→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
    63→
    64→  // Preload voices on mount
    65→  useEffect(() => {
    66→    loadVoices();
    67→  }, []);
    68→
    69→  // Reset per-step interactive state when stepIdx changes.
    70→  // Official React pattern: adjust state during render based on a tracked
    71→  // previous value. React re-renders immediately without committing the
    72→  // stale intermediate state. (Avoids setState-in-effect cascades.)
    73→  if (stepIdx !== prevStepIdx) {
    74→    setPrevStepIdx(stepIdx);
    75→    setQuizAnswer(null);
    76→    setPracticeScore(null);
    77→    setRecording(false);
    78→  }
    79→
    80→  // Stop TTS on unmount or step change
    81→  useEffect(() => {
    82→    return () => stopSpeaking();
    83→  }, [stepIdx]);
    84→
    85→  const handleSpeak = useCallback(
    86→    (text: string) => {
    87→      speak(text, { accent, rate: 0.95 });
    88→    },
    89→    [accent]
    90→  );
    91→
    92→  const goNext = useCallback(() => {
    93→    if (isLast) {
    94→      // already at completion step
    95→      return;
    96→    }
    97→    setDirection(1);
    98→    setStepIdx((i) => i + 1);
    99→  }, [isLast]);
   100→
   101→  const goPrev = useCallback(() => {
   102→    if (isFirst) return;
   103→    setDirection(-1);
   104→    setStepIdx((i) => i - 1);
   105→  }, [isFirst]);
   106→
   107→  // Mark completion when reaching completion step
   108→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   109→  const alreadyCompleted = lessons[lesson.id]?.completed;
   110→
   111→  const handleComplete = useCallback(() => {
   112→    if (alreadyCompleted) return;
   113→    if (completionStep?.type === "completion") {
   114→      completeLesson(
   115→        lesson.id,
   116→        practiceScore ?? 85,
   117→        completionStep.xp,
   118→        completionStep.badge
   119→      );
   120→      addSpeakingTime(8);
   121→      setShowConfetti(true);
   122→      setTimeout(() => setShowConfetti(false), 4000);
   123→    }
   124→  }, [alreadyCompleted, completionStep, completeLesson, lesson.id, practiceScore, addSpeakingTime]);
   125→
   126→  // Auto-trigger completion when user reaches the completion step.
   127→  // handleComplete calls setState (setShowConfetti + store actions), so we
   128→  // defer it out of the synchronous effect body to satisfy
   129→  // react-hooks/set-state-in-effect (no cascading renders).
   130→  useEffect(() => {
   131→    if (step?.type !== "completion" || alreadyCompleted) return;
   132→    const id = setTimeout(() => handleComplete(), 0);
   133→    return () => clearTimeout(id);
   134→  }, [step, alreadyCompleted, handleComplete]);
   135→
   136→  // ESC to close
   137→  useEffect(() => {
   138→    const onKey = (e: KeyboardEvent) => {
   139→      if (e.key === "Escape") onClose();
   140→      else if (e.key === "ArrowRight") goNext();
   141→      else if (e.key === "ArrowLeft") goPrev();
   142→    };
   143→    window.addEventListener("keydown", onKey);
   144→    return () => window.removeEventListener("keydown", onKey);
   145→  }, [goNext, goPrev, onClose]);
   146→
   147→  return (
   148→    <motion.div
   149→      initial={{ opacity: 0 }}
   150→      animate={{ opacity: 1 }}
   151→      exit={{ opacity: 0 }}
   152→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col"
   153→    >
   154→      {showConfetti && <Confetti count={100} />}
   155→
   156→      {/* Header */}
   157→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   158→        <button
   159→          onClick={onClose}
   160→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   161→          aria-label="Close lesson"
   162→        >
   163→          <X className="w-5 h-5" />
   164→        </button>
   165→        <div className="flex-1 px-3 text-center">
   166→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   167→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   168→          </div>
   169→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   170→        </div>
   171→        <div className="flex items-center gap-1.5">
   172→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   173→          <ProgressRing pct={pct} size={36} stroke={3} />
   174→        </div>
   175→      </div>
   176→
   177→      {/* Progress bar */}
   178→      <div className="h-1 bg-[var(--bg2)]">
   179→        <motion.div
   180→          className="h-full bg-[var(--grad-btn)]"
   181→          initial={{ width: 0 }}
   182→          animate={{ width: `${pct}%` }}
   183→          transition={{ duration: 0.4 }}
   184→        />
   185→      </div>
   186→
   187→      {/* Step content */}
   188→      <div className="flex-1 overflow-y-auto">
   189→        <AnimatePresence mode="wait" custom={direction}>
   190→          <motion.div
   191→            key={stepIdx}
   192→            custom={direction}
   193→            initial={{ opacity: 0, x: direction * 30 }}
   194→            animate={{ opacity: 1, x: 0 }}
   195→            exit={{ opacity: 0, x: direction * -30 }}
   196→            transition={{ duration: 0.25 }}
   197→            className="max-w-2xl mx-auto px-5 py-6 pb-32"
   198→          >
   199→            <StepRenderer
   200→              step={step}
   201→              speak={handleSpeak}
   202→              quizAnswer={quizAnswer}
   203→              setQuizAnswer={setQuizAnswer}
   204→              practiceScore={practiceScore}
   205→              setPracticeScore={setPracticeScore}
   206→              recording={recording}
   207→              setRecording={setRecording}
   208→              isLast={isLast}
   209→              onComplete={handleComplete}
   210→              onNext={onNext}
   211→            />
   212→          </motion.div>
   213→        </AnimatePresence>
   214→      </div>
   215→
   216→      {/* Footer nav */}
   217→      <div className="border-t border-[var(--border)] bg-[var(--bg2)]/95 backdrop-blur px-4 py-3 flex items-center gap-3 safe-bottom">
   218→        <button
   219→          onClick={goPrev}
   220→          disabled={isFirst}
   221→          className="px-4 py-2.5 rounded-xl flex items-center gap-1.5 text-sm font-semibold disabled:opacity-30 disabled:cursor-not-allowed hover:bg-[var(--card-h)] transition"
   222→        >
   223→          <ChevronLeft className="w-4 h-4" /> Back
   224→        </button>
   225→        <div className="flex-1" />
   226→        {step?.type === "completion" ? (
   227→          onNext ? (
   228→            <button
   229→              onClick={onNext}
   230→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   231→            >
   232→              Next Lesson <ChevronRight className="w-4 h-4" />
   233→            </button>
   234→          ) : (
   235→            <button
   236→              onClick={onClose}
   237→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   238→            >
   239→              <Check className="w-4 h-4" /> Finish
   240→            </button>
   241→          )
   242→        ) : (
   243→          <button
   244→            onClick={goNext}
   245→            className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   246→          >
   247→            {isLast ? "Complete" : "Continue"} <ChevronRight className="w-4 h-4" />
   248→          </button>
   249→        )}
   250→      </div>
   251→    </motion.div>
   252→  );
   253→}

   255→// ─── Step Renderer ───
   256→
   257→interface StepRendererProps {
   258→  step: LessonStep;
   259→  speak: (text: string) => void;
   260→  quizAnswer: number | null;
   261→  setQuizAnswer: (n: number | null) => void;
   262→  practiceScore: number | null;
   263→  setPracticeScore: (n: number | null) => void;
   264→  recording: boolean;
   265→  setRecording: (b: boolean) => void;
   266→  isLast: boolean;
   267→  onComplete: () => void;
   268→  onNext?: () => void;
   269→}
   270→
   271→function StepRenderer(props: StepRendererProps) {
   272→  const { step, speak } = props;
   273→
   274→  switch (step.type) {
   275→    case "intro":
   276→      return <IntroStepView step={step} speak={speak} />;
   277→    case "concept":
   278→      return <ConceptStepView step={step} />;
   279→    case "example":
   280→      return <ExampleStepView step={step} speak={speak} />;
   281→    case "mouth-diagram":
   282→      return <MouthDiagram step={step} speak={speak} />;
   283→    case "vowel-chart":
   284→      return <VowelChart step={step} speak={speak} />;
   285→    case "compare":
   286→      return <CompareWave step={step} speak={speak} />;
   287→    case "stress-bars":
   288→      return <StressBars step={step} speak={speak} />;
   289→    case "rhythm":
   290→      return <RhythmBeats step={step} speak={speak} />;
   291→    case "linking":
   292→      return <LinkingDiagram step={step} speak={speak} />;
   293→    case "intonation":
   294→      return <IntonationContour step={step} speak={speak} />;
   295→    case "shadow":
   296→      return <ShadowStepView step={step} speak={speak} />;
   297→    case "tap-pronounce":
   298→      return <TapPronounceStepView step={step} speak={speak} />;
   299→    case "tip":
   300→      return <TipStepView step={step} />;
   301→    case "practice":
   302→      return <PracticeStepView {...props} />;
   303→    case "quiz":
   304→      return <QuizStepView {...props} />;
   305→    case "completion":
   306→      return <CompletionStepView step={step} onNext={props.onNext} />;
   307→    default:
   308→      return <div>Unknown step type</div>;
   309→  }
   310→}

   388→  const words = step.phrase.split(/\s+/);
   389→  return (
   390→    <div className="space-y-4">
   391→      {step.title && (
   392→        <div className="flex items-center gap-2">
   393→          <span className="text-xs px-2 py-0.5 rounded-full bg-[rgba(34,211,238,0.15)] text-[#22d3ee] font-mono font-bold uppercase tracking-wider">
   394→            Example
   395→          </span>
   396→        </div>
   397→      )}
   398→      <div className="rounded-2xl p-5 bg-[rgba(99,102,241,0.06)] border border-[rgba(99,102,241,0.25)]">
   399→        <div className="flex flex-wrap gap-1.5 mb-4">
   400→          {words.map((w, i) => {
   401→            const cleaned = w.replace(/[.,!?;:"']/g, "");

   404→              <button
   405→                key={i}

   408→                  isHl
   409→                    ? "bg-[var(--grad-btn)] text-white font-semibold"

   413→                {w}
   414→              </button>
   415→            );
   416→          })}

   420→          onClick={() => speak(step.phrase)}

   434→                key={i}

   437→              >
   438→                <div className="font-d text-sm">{tw.word}</div>

   441→            ))}
   442→          </div>
   443→        </div>
   444→      )}
   445→      {step.tip && (
   446→        <div className="rounded-xl p-3 bg-[rgba(245,158,11,0.08)] border border-[rgba(245,158,11,0.25)] text-xs text-[var(--t2)]">
   447→          <strong className="text-[#f59e0b]">💡 Tip:</strong> {step.tip}
   448→        </div>
   449→      )}
   450→    </div>

   487→function TipStepView({ step }: { step: Extract<LessonStep, { type: "tip" }> }) {
   488→  const variantStyles = {
   489→    info: { bg: "rgba(99,102,241,0.08)", border: "rgba(99,102,241,0.3)", color: "#a78bfa", icon: "💡" },
   490→    success: { bg: "rgba(16,185,129,0.08)", border: "rgba(16,185,129,0.3)", color: "#10b981", icon: "✓" },
   491→    warning: { bg: "rgba(245,158,11,0.08)", border: "rgba(245,158,11,0.3)", color: "#f59e0b", icon: "⚠" },
   492→  };
   493→  const v = variantStyles[step.variant || "info"];
   494→  return (
   495→    <div className="space-y-3">
   496→      <div

   510→function ShadowStepView({ step, speak }: { step: Extract<LessonStep, { type: "shadow" }>; speak: (t: string) => void }) {

   675→function QuizStepView({
   676→  step,
   677→  quizAnswer,
   678→  setQuizAnswer,
   679→}: StepRendererProps) {
   680→  const s = step as Extract<LessonStep, { type: "quiz" }>;
   681→  const answered = quizAnswer !== null;
   682→  const isCorrect = answered && quizAnswer === s.correct;
   683→
   684→  return (
   685→    <div className="space-y-4">
   686→      <div className="flex items-center gap-2">
   687→        <span className="text-xs px-2 py-0.5 rounded-full bg-[rgba(167,139,250,0.15)] text-[#a78bfa] font-mono font-bold uppercase tracking-wider">
   688→          Quiz
   689→        </span>
   690→      </div>
   691→      <h2 className="font-d text-xl font-bold text-[var(--t1)] leading-snug">{s.question}</h2>
   692→      <div className="space-y-2">
   693→        {s.options.map((opt, i) => {
   694→          const isSelected = quizAnswer === i;
   695→          const isCorrectOpt = i === s.correct;
   696→          let style = "bg-[var(--card)] border-[var(--border)] hover:border-[var(--p3)]";
   697→          if (answered) {
   698→            if (isCorrectOpt) {
   699→              style = "bg-[rgba(16,185,129,0.1)] border-[rgba(16,185,129,0.5)]";
   700→            } else if (isSelected) {
   701→              style = "bg-[rgba(239,68,68,0.1)] border-[rgba(239,68,68,0.5)]";
   702→            } else {
   703→              style = "bg-[var(--card)] border-[var(--border)] opacity-50";
   704→            }
   705→          }
   706→          return (
   707→            <motion.button
   708→              key={i}
   709→              whileTap={{ scale: 0.98 }}
   710→              disabled={answered}
   711→              onClick={() => setQuizAnswer(i)}
   712→              className={`w-full p-4 rounded-xl border-2 text-left flex items-center justify-between gap-3 transition ${style}`}
   713→            >
   714→              <span className="text-sm font-medium text-[var(--t1)]">{opt}</span>
   715→              {answered && isCorrectOpt && <Check className="w-5 h-5 text-[#10b981]" />}
   716→              {answered && isSelected && !isCorrectOpt && <X className="w-5 h-5 text-[#ef4444]" />}
   717→            </motion.button>
   718→          );
   719→        })}
   720→      </div>
   721→      {answered && (
   722→        <motion.div
   723→          initial={{ opacity: 0, y: 10 }}
   724→          animate={{ opacity: 1, y: 0 }}
   725→          className={`rounded-xl p-4 border ${
   726→            isCorrect
   727→              ? "bg-[rgba(16,185,129,0.08)] border-[rgba(16,185,129,0.3)]"
   728→              : "bg-[rgba(239,68,68,0.08)] border-[rgba(239,68,68,0.3)]"
   729→          }`}
   730→        >
   731→          <div className="flex items-center gap-2 mb-2">
   732→            <span className={`font-bold ${isCorrect ? "text-[#10b981]" : "text-[#ef4444]"}`}>
   733→              {isCorrect ? "✓ Correct!" : "✗ Not quite"}
   734→            </span>
   735→          </div>
   736→          <p className="text-[var(--t2)] text-sm leading-relaxed">{s.explanation}</p>
   737→        </motion.div>
   738→      )}
   739→    </div>
   740→  );
   741→}
