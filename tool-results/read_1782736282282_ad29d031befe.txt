    42→    42→export function LessonModal({ lesson, onClose, onNext }: Props) {
    43→    43→  const [stepIdx, setStepIdx] = useState(0);
    44→    44→  const [direction, setDirection] = useState(1);
    45→    45→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
    46→    46→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
    47→    47→  const [recording, setRecording] = useState(false);
    48→    48→  const [showConfetti, setShowConfetti] = useState(false);
    49→    49→  // Track the previous stepIdx so we can reset per-step interactive state
    50→    50→  // when the step changes — using the official "adjust state during render"
    51→    51→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
    52→    52→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
    53→    53→  const accent = useAppStore((s) => s.accent);
    54→    54→  const completeLesson = useAppStore((s) => s.completeLesson);
    55→    55→  const lessons = useAppStore((s) => s.lessons);
    56→    56→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    57→    57→
    58→    58→  const step = lesson.steps[stepIdx];
    59→    59→  const isLast = stepIdx === lesson.steps.length - 1;
    60→    60→  const isFirst = stepIdx === 0;
    61→    61→  const totalSteps = lesson.steps.length;
    62→    62→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
    63→    63→
    64→    64→  // Preload voices on mount
    65→    65→  useEffect(() => {
    66→    66→    loadVoices();
    67→    67→  }, []);
    68→    68→
    69→    69→  // Reset per-step interactive state when stepIdx changes.
    70→    70→  // Official React pattern: adjust state during render based on a tracked
    71→    71→  // previous value. React re-renders immediately without committing the
    72→    72→  // stale intermediate state. (Avoids setState-in-effect cascades.)
    73→    73→  if (stepIdx !== prevStepIdx) {
    74→    74→    setPrevStepIdx(stepIdx);
    75→    75→    setQuizAnswer(null);
    76→    76→    setPracticeScore(null);
    77→    77→    setRecording(false);
    78→    78→  }
    79→    79→
    80→    80→  // Stop TTS on unmount or step change
    81→    81→  useEffect(() => {
    82→    82→    return () => stopSpeaking();
    83→    83→  }, [stepIdx]);
    84→    84→
    85→    85→  const handleSpeak = useCallback(
    86→    86→    (text: string) => {
    87→    87→      speak(text, { accent, rate: 0.95 });
    88→    88→    },
    89→    89→    [accent]
    90→    90→  );
    91→    91→
    92→    92→  const goNext = useCallback(() => {
    93→    93→    if (isLast) {
    94→    94→      // already at completion step
    95→    95→      return;
    96→    96→    }
    97→    97→    setDirection(1);
    98→    98→    setStepIdx((i) => i + 1);
    99→    99→  }, [isLast]);
   100→   100→
   101→   101→  const goPrev = useCallback(() => {
   102→   102→    if (isFirst) return;
   103→   103→    setDirection(-1);
   104→   104→    setStepIdx((i) => i - 1);
   105→   105→  }, [isFirst]);
   106→   106→
   107→   107→  // Mark completion when reaching completion step
   108→   108→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   109→   109→  const alreadyCompleted = lessons[lesson.id]?.completed;
   110→   110→
   111→   111→  const handleComplete = useCallback(() => {
   112→   112→    if (alreadyCompleted) return;
   113→   113→    if (completionStep?.type === "completion") {
   114→   114→      completeLesson(
   115→   115→        lesson.id,
   116→   116→        practiceScore ?? 85,
   117→   117→        completionStep.xp,
   118→   118→        completionStep.badge
   119→   119→      );
   120→   120→      addSpeakingTime(8);
   121→   121→      setShowConfetti(true);
   122→   122→      setTimeout(() => setShowConfetti(false), 4000);
   123→   123→    }
   124→   124→  }, [alreadyCompleted, completionStep, completeLesson, lesson.id, practiceScore, addSpeakingTime]);
   125→   125→
   126→   126→  // Auto-trigger completion when user reaches the completion step.
   127→   127→  // handleComplete calls setState (setShowConfetti + store actions), so we
   128→   128→  // defer it out of the synchronous effect body to satisfy
   129→   129→  // react-hooks/set-state-in-effect (no cascading renders).
   130→   130→  useEffect(() => {
   131→   131→    if (step?.type !== "completion" || alreadyCompleted) return;
   132→   132→    const id = setTimeout(() => handleComplete(), 0);
   133→   133→    return () => clearTimeout(id);
   134→   134→  }, [step, alreadyCompleted, handleComplete]);
   135→   135→
   136→   136→  // ESC to close
   137→   137→  useEffect(() => {
   138→   138→    const onKey = (e: KeyboardEvent) => {
   139→   139→      if (e.key === "Escape") onClose();
   140→   140→      else if (e.key === "ArrowRight") goNext();
   141→   141→      else if (e.key === "ArrowLeft") goPrev();
   142→   142→    };
   143→   143→    window.addEventListener("keydown", onKey);
   144→   144→    return () => window.removeEventListener("keydown", onKey);
   145→   145→  }, [goNext, goPrev, onClose]);
   146→   146→
   147→   147→  return (
   148→   148→    <motion.div
   149→   149→      initial={{ opacity: 0 }}
   150→   150→      animate={{ opacity: 1 }}
   151→   151→      exit={{ opacity: 0 }}
   152→   152→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col"
   153→   153→    >
   154→   154→      {showConfetti && <Confetti count={100} />}
   155→   155→
   156→   156→      {/* Header */}
   157→   157→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   158→   158→        <button
   159→   159→          onClick={onClose}
   160→   160→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   161→   161→          aria-label="Close lesson"
   162→   162→        >
   163→   163→          <X className="w-5 h-5" />
   164→   164→        </button>
   165→   165→        <div className="flex-1 px-3 text-center">
   166→   166→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   167→   167→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   168→   168→          </div>
   169→   169→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   170→   170→        </div>
   171→   171→        <div className="flex items-center gap-1.5">
   172→   172→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   173→   173→          <ProgressRing pct={pct} size={36} stroke={3} />
   174→   174→        </div>
   175→   175→      </div>
   176→   176→
   177→   177→      {/* Progress bar */}
   178→   178→      <div className="h-1 bg-[var(--bg2)]">
   179→   179→        <motion.div
   180→   180→          className="h-full bg-[var(--grad-btn)]"
   181→   181→          initial={{ width: 0 }}
   182→   182→          animate={{ width: `${pct}%` }}
   183→   183→          transition={{ duration: 0.4 }}
   184→   184→        />
   185→   185→      </div>
   186→   186→
   187→   187→      {/* Step content */}
   188→   188→      <div className="flex-1 overflow-y-auto">
   189→   189→        <AnimatePresence mode="wait" custom={direction}>
   190→   190→          <motion.div
   191→   191→            key={stepIdx}
   192→   192→            custom={direction}
   193→   193→            initial={{ opacity: 0, x: direction * 30 }}
   194→   194→            animate={{ opacity: 1, x: 0 }}
   195→   195→            exit={{ opacity: 0, x: direction * -30 }}
   196→   196→            transition={{ duration: 0.25 }}
   197→   197→            className="max-w-2xl mx-auto px-5 py-6 pb-32"
   198→   198→          >
   199→   199→            <StepRenderer
   200→   200→              step={step}
   201→   201→              speak={handleSpeak}
   202→   202→              quizAnswer={quizAnswer}
   203→   203→              setQuizAnswer={setQuizAnswer}
   204→   204→              practiceScore={practiceScore}
   205→   205→              setPracticeScore={setPracticeScore}
   206→   206→              recording={recording}
   207→   207→              setRecording={setRecording}
   208→   208→              isLast={isLast}
   209→   209→              onComplete={handleComplete}
   210→   210→              onNext={onNext}
   211→   211→            />
   212→   212→          </motion.div>
   213→   213→        </AnimatePresence>
   214→   214→      </div>
   215→   215→
   216→   216→      {/* Footer nav */}
   217→   217→      <div className="border-t border-[var(--border)] bg-[var(--bg2)]/95 backdrop-blur px-4 py-3 flex items-center gap-3 safe-bottom">
   218→   218→        <button
   219→   219→          onClick={goPrev}
   220→   220→          disabled={isFirst}
   221→   221→          className="px-4 py-2.5 rounded-xl flex items-center gap-1.5 text-sm font-semibold disabled:opacity-30 disabled:cursor-not-allowed hover:bg-[var(--card-h)] transition"
   222→   222→        >
   223→   223→          <ChevronLeft className="w-4 h-4" /> Back
   224→   224→        </button>
   225→   225→        <div className="flex-1" />
   226→   226→        {step?.type === "completion" ? (
   227→   227→          onNext ? (
   228→   228→            <button
   229→   229→              onClick={onNext}
   230→   230→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   231→   231→            >
   232→   232→              Next Lesson <ChevronRight className="w-4 h-4" />
   233→   233→            </button>
   234→   234→          ) : (
   235→   235→            <button
   236→   236→              onClick={onClose}
   237→   237→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   238→   238→            >
   239→   239→              <Check className="w-4 h-4" /> Finish
   240→   240→            </button>
   241→   241→          )
   242→   242→        ) : (
   243→   243→          <button
   244→   244→            onClick={goNext}
   245→   245→            className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   246→   246→          >
   247→   247→            {isLast ? "Complete" : "Continue"} <ChevronRight className="w-4 h-4" />
   248→   248→          </button>
   249→   249→        )}
   250→   250→      </div>
   251→   251→    </motion.div>
   252→   252→  );
   253→   253→}

   255→   255→// ─── Step Renderer ───
   256→   256→
   257→   257→interface StepRendererProps {
   258→   258→  step: LessonStep;
   259→   259→  speak: (text: string) => void;
   260→   260→  quizAnswer: number | null;
   261→   261→  setQuizAnswer: (n: number | null) => void;
   262→   262→  practiceScore: number | null;
   263→   263→  setPracticeScore: (n: number | null) => void;
   264→   264→  recording: boolean;
   265→   265→  setRecording: (b: boolean) => void;
   266→   266→  isLast: boolean;
   267→   267→  onComplete: () => void;
   268→   268→  onNext?: () => void;
   269→   269→}
   270→   270→
   271→   271→function StepRenderer(props: StepRendererProps) {
   272→   272→  const { step, speak } = props;
   273→   273→
   274→   274→  switch (step.type) {
   275→   275→    case "intro":
   276→   276→      return <IntroStepView step={step} speak={speak} />;
   277→   277→    case "concept":
   278→   278→      return <ConceptStepView step={step} />;
   279→   279→    case "example":
   280→   280→      return <ExampleStepView step={step} speak={speak} />;
   281→   281→    case "mouth-diagram":
   282→   282→      return <MouthDiagram step={step} speak={speak} />;
   283→   283→    case "vowel-chart":
   284→   284→      return <VowelChart step={step} speak={speak} />;
   285→   285→    case "compare":
   286→   286→      return <CompareWave step={step} speak={speak} />;
   287→   287→    case "stress-bars":
   288→   288→      return <StressBars step={step} speak={speak} />;
   289→   289→    case "rhythm":
   290→   290→      return <RhythmBeats step={step} speak={speak} />;
   291→   291→    case "linking":
   292→   292→      return <LinkingDiagram step={step} speak={speak} />;
   293→   293→    case "intonation":
   294→   294→      return <IntonationContour step={step} speak={speak} />;
   295→   295→    case "shadow":
   296→   296→      return <ShadowStepView step={step} speak={speak} />;
   297→   297→    case "tap-pronounce":
   298→   298→      return <TapPronounceStepView step={step} speak={speak} />;
   299→   299→    case "tip":
   300→   300→      return <TipStepView step={step} />;
   301→   301→    case "practice":
   302→   302→      return <PracticeStepView {...props} />;
   303→   303→    case "quiz":
   304→   304→      return <QuizStepView {...props} />;
   305→   305→    case "completion":
   306→   306→      return <CompletionStepView step={step} onNext={props.onNext} />;
   307→   307→    default:
   308→   308→      return <div>Unknown step type</div>;
   309→   309→  }
   310→   310→}

   388→   388→  const words = step.phrase.split(/\s+/);
   389→   389→  return (
   390→   390→    <div className="space-y-4">
   391→   391→      {step.title && (
   392→   392→        <div className="flex items-center gap-2">
   393→   393→          <span className="text-xs px-2 py-0.5 rounded-full bg-[rgba(34,211,238,0.15)] text-[#22d3ee] font-mono font-bold uppercase tracking-wider">

   412→   412→              >

   414→   414→              </button>

   417→   417→        </div>

   419→   419→        <button

   426→   426→      {step.tapWords && step.tapWords.length > 0 && (
