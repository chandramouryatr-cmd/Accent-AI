    85→// ─── Phoneme mastery derivation ──────────────────────────────────────────
    86→
    87→function derivePhonemeMastery(
    88→  lessons: Record<string, { completed?: boolean; score: number }>
    89→): PhonemeScore[] {
    90→  const out: PhonemeScore[] = [];
    91→  for (const [ph, { ids, example }] of Object.entries(PHONEME_LESSONS)) {
    92→    const relevant = ids
    93→      .map((id) => lessons[id])
    94→      .filter((l) => l?.completed);
    95→    if (relevant.length === 0) continue;
    96→    const avg = Math.round(
    97→      relevant.reduce((s, l) => s + l.score, 0) / relevant.length
    98→    );
    99→    const sortedById = [...relevant].sort((a, b) => a.score - b.score);
   100→    const bestLessonId =
   101→      ids.find((id) => lessons[id] === sortedById[0]) || ids[0];
   102→    out.push({
   103→      phoneme: ph,
   104→      example,
   105→      score: avg,
   106→      count: relevant.length,
   107→      lessonId: bestLessonId,
   108→    });
   109→  }
   110→  out.sort((a, b) => a.score - b.score);
   111→  return out;
   112→}

   114→// ─── JSON parsing helpers (robust against markdown / extra text) ─────────

   116→function extractJson(text: string): InsightsPlan | null {
   117→  if (!text) return null;
   118→  const trimmed = text.trim();
   119→
   120→  // Attempt 1: direct parse
   121→  try {
   122→    return normalizePlan(JSON.parse(trimmed));
   123→  } catch {
   124→    /* continue */
   125→  }
   126→
   127→  // Attempt 2: unwrap ```json ... ``` or ``` ... ``` fences
   128→  const fenceMatch = trimmed.match(/```(?:json)?\s*([\s\S]*?)```/i);
   129→  if (fenceMatch) {
   130→    try {
   131→      return normalizePlan(JSON.parse(fenceMatch[1].trim()));
   132→    } catch {
   133→      /* continue */
   134→    }
   135→  }
   136→
   137→  // Attempt 3: grab substring between first { and last }
   138→  const firstBrace = trimmed.indexOf("{");
   139→  const lastBrace = trimmed.lastIndexOf("}");
   140→  if (firstBrace !== -1 && lastBrace > firstBrace) {
   141→    const slice = trimmed.slice(firstBrace, lastBrace + 1);
   142→    try {
   143→      return normalizePlan(JSON.parse(slice));
   144→    } catch {
   145→      /* continue */
   146→    }
   147→  }
   148→
   149→  return null;
   150→}

   152→function normalizePlan(raw: unknown): InsightsPlan | null {
   153→  if (!raw || typeof raw !== "object") return null;
   154→  const obj = raw as Record<string, unknown>;
   155→  const focusAreas = Array.isArray(obj.focusAreas)
   156→    ? (obj.focusAreas as unknown[])
   157→        .map((item) => normalizeFocusArea(item))
   158→        .filter((x): x is FocusArea => x !== null)
   159→    : [];
   160→  const recommendedLessons = Array.isArray(obj.recommendedLessons)
   161→    ? (obj.recommendedLessons as unknown[])
   162→        .map((item) => normalizeRecommendedLesson(item))
   163→        .filter((x): x is RecommendedLesson => x !== null)
   164→    : [];
   165→  const tips = Array.isArray(obj.tips)
   166→    ? (obj.tips as unknown[])
   167→        .map((t) => (typeof t === "string" ? t : String(t ?? "")))
   168→        .filter((s) => s.length > 0)
   169→        .slice(0, 4)
   170→    : [];
   171→
   172→  if (focusAreas.length === 0 && recommendedLessons.length === 0 && tips.length === 0) {
   173→    return null;
   174→  }
   175→  return { focusAreas, recommendedLessons, tips };
   176→}
   177→
   178→function normalizeFocusArea(item: unknown): FocusArea | null {
   179→  if (!item || typeof item !== "object") return null;
   180→  const obj = item as Record<string, unknown>;
   181→  const phoneme =
   182→    typeof obj.phoneme === "string"
   183→      ? obj.phoneme.replace(/^\/+|\/+$/g, "").trim()
   184→      : "";
   185→  const score =
   186→    typeof obj.score === "number"
   187→      ? Math.max(0, Math.min(100, Math.round(obj.score)))
   188→      : typeof obj.score === "string"
   189→      ? parseInt(obj.score, 10) || 0
   190→      : 0;
   191→  const reason =
   192→    typeof obj.reason === "string" ? obj.reason : typeof obj.reason === "object" && obj.reason ? JSON.stringify(obj.reason) : "";
   193→  if (!phoneme) return null;
   194→  return { phoneme, score, reason };
   195→}
   196→
   197→function normalizeRecommendedLesson(item: unknown): RecommendedLesson | null {
   198→  if (!item || typeof item !== "object") return null;
   199→  const obj = item as Record<string, unknown>;
   200→  const phase =
   201→    typeof obj.phase === "number"
   202→      ? Math.max(1, Math.min(8, Math.round(obj.phase)))
   203→      : typeof obj.phase === "string"
   204→      ? parseInt(obj.phase, 10) || 1
   205→      : 1;
   206→  const lesson = typeof obj.lesson === "string" ? obj.lesson : "";
   207→  const reason =
   208→    typeof obj.reason === "string"
   209→      ? obj.reason
   210→      : typeof obj.reason === "object" && obj.reason
   211→      ? JSON.stringify(obj.reason)
   212→      : "";
   213→  if (!lesson) return null;
   214→  return { phase, lesson, reason };
   215→}

   457→export function CoachInsights() {
   458→  const lessons = useAppStore((s) => s.lessons);
   459→  const xp = useAppStore((s) => s.xp);
   460→  const streak = useAppStore((s) => s.streak);
   461→  const accent = useAppStore((s) => s.accent);
   462→  const setActiveLesson = useAppStore((s) => s.setActiveLesson);
   463→
   464→  const [view, setView] = useState<ViewState>("idle");
   465→  const [parsed, setParsed] = useState<InsightsPlan | null>(null);
   466→  const [rawText, setRawText] = useState<string | null>(null);
   467→  const [error, setError] = useState<string | null>(null);
   468→
   469→  const abortRef = useRef<AbortController | null>(null);
   470→
   471→  // Derive phoneme mastery data (weakest first)
   472→  const phonemeMastery = useMemo(() => derivePhonemeMastery(lessons), [lessons]);
   473→
   474→  // Build a signature so cache invalidates if phoneme scores change
   475→  const signature = useMemo(() => {
   476→    return phonemeMastery
   477→      .map((p) => `${p.phoneme}:${p.score}:${p.count ?? 0}`)
   478→      .join("|");
   479→  }, [phonemeMastery]);
   480→
   481→  const completedCount = useMemo(
   482→    () => Object.values(lessons).filter((l) => l?.completed).length,
   483→    [lessons]
   484→  );
   485→
   486→  // Load cached insights for today on mount
   487→  useEffect(() => {
   488→    if (typeof window === "undefined") return;
   489→    try {
   490→      const raw = window.localStorage.getItem(storageKeyForToday());
   491→      if (!raw) return;
   492→      const cached = JSON.parse(raw) as CachedInsight;
   493→      // Only restore if the phoneme signature hasn't materially changed
   494→      // (allow restore even if signature differs by a little — focus on date key)
   495→      if (cached && (cached.parsed || cached.rawText)) {
   496→        setParsed(cached.parsed);
   497→        setRawText(cached.rawText);
   498→        setView("success");
   499→      }
   500→    } catch {
   501→      /* ignore malformed cache */
   502→    }
   503→  }, []);
   504→
   505→  // Cleanup any in-flight request on unmount
   506→  useEffect(() => {
   507→    return () => {
   508→      if (abortRef.current) {
   509→        abortRef.current.abort();
   510→        abortRef.current = null;
   511→      }
   512→    };
   513→  }, []);
   514→
   515→  const handleGetInsights = useCallback(async () => {
   516→    setError(null);
   517→
   518→    // Abort any previous request
   519→    if (abortRef.current) abortRef.current.abort();
   520→    const controller = new AbortController();
   521→    abortRef.current = controller;
   522→
   523→    setView("loading");
   524→    setParsed(null);
   525→    setRawText(null);
   526→
   527→    const timeoutId = setTimeout(() => {
   528→      controller.abort();
   529→    }, FIRST_TOKEN_TIMEOUT_MS);
   530→
   531→    // Build completed lessons list
   532→    const completedLessons = Object.entries(lessons)
   533→      .filter(([, p]) => p?.completed)
   534→      .map(([id]) => id);
   535→
   536→    // Top 5 weakest phonemes for the model
   537→    const topPhonemes = phonemeMastery.slice(0, 5).map((p) => ({
   538→      phoneme: p.phoneme,
   539→      score: p.score,
   540→      example: p.example,
   541→      count: p.count,
   542→    }));
   543→
   544→    try {
   545→      const res = await fetch("/api/ai-coach", {
   546→        method: "POST",
   547→        headers: { "Content-Type": "application/json" },
   548→        body: JSON.stringify({
   549→          mode: "insights",
   550→          messages: [],
   551→          context: {
   552→            accent,
   553→            xp,
   554→            streak,
   555→            completedLessons,
   556→            phonemeMastery: topPhonemes,
   557→          },
   558→        }),
   559→        signal: controller.signal,
   560→      });
   561→
   562→      if (!res.ok) {
   563→        let detail = `Request failed (${res.status})`;
   564→        try {
   565→          const data = await res.json();
   566→          if (data?.error) detail = data.error;
   567→        } catch {
   568→          /* ignore */
   569→        }
   570→        throw new Error(detail);
   571→      }
   572→
   573→      if (!res.body) {
   574→        throw new Error("No response body received.");
   575→      }
   576→
   577→      // Consume SSE stream and accumulate text
   578→      const reader = res.body.getReader();
   579→      const decoder = new TextDecoder();
   580→      let accumulated = "";
   581→      let buffer = "";
   582→      let firstTokenReceived = false;
   583→
   584→      while (true) {
   585→        const { done, value } = await reader.read();
   586→        if (done) break;
   587→
   588→        buffer += decoder.decode(value, { stream: true });
   589→        const lines = buffer.split("\n");
   590→        buffer = lines.pop() || "";
   591→
   592→        for (const line of lines) {
   593→          const trimmedLine = line.trim();
   594→          if (!trimmedLine || !trimmedLine.startsWith("data:")) continue;
   595→
   596→          const dataStr = trimmedLine.slice(5).trim();
   597→          if (dataStr === "[DONE]") {
   598→            break;
   599→          }
   600→
   601→          try {
   602→            const parsedChunk = JSON.parse(dataStr);
   603→            if (parsedChunk.error) {
   604→              throw new Error(parsedChunk.error);
   605→            }
   606→            if (typeof parsedChunk.token === "string") {
   607→              if (!firstTokenReceived) {
   608→                firstTokenReceived = true;
   609→                clearTimeout(timeoutId);
   610→              }
   611→              accumulated += parsedChunk.token;
   612→            }
   613→          } catch (parseErr) {
   614→            // Re-throw our own thrown errors (with a real message)
   615→            if (
   616→              parseErr instanceof Error &&
   617→              !parseErr.message.startsWith("Unexpected")
   618→            ) {
   619→              throw parseErr;
   620→            }
   621→            // Skip malformed JSON lines (expected for partial SSE chunks)
   622→          }
   623→        }
   624→      }
   625→
   626→      clearTimeout(timeoutId);
   627→
   628→      const text = accumulated.trim();
   629→      if (!text) {
   630→        throw new Error("The coach didn't return any insights. Please try again.");
   631→      }
   632→
   633→      const plan = extractJson(text);
   634→      setParsed(plan);
   635→      setRawText(plan ? null : text);
   636→      setView("success");
   637→
   638→      // Cache the result (date-keyed)
   639→      try {
   640→        const payload: CachedInsight = {
   641→          parsed: plan,
   642→          rawText: plan ? null : text,
   643→          generatedAt: Date.now(),
   644→          signature,
   645→        };
   646→        window.localStorage.setItem(
   647→          storageKeyForToday(),
   648→          JSON.stringify(payload)
   649→        );
   650→      } catch {
   651→        /* ignore storage errors */
   652→      }
   653→    } catch (err) {
   654→      clearTimeout(timeoutId);
   655→      if (err instanceof DOMException && err.name === "AbortError") {
   656→        setError("Request timed out — the coach took too long. Please try again.");
   657→      } else {
   658→        const msg = err instanceof Error ? err.message : "Something went wrong.";
   659→        setError(msg);
   660→      }
   661→      setView("error");
   662→    } finally {
   663→      abortRef.current = null;
   664→    }
   665→  }, [lessons, phonemeMastery, accent, xp, streak, signature]);
   666→
   667→  const handleOpenLesson = useCallback(
   668→    (lessonTitle: string) => {
   669→      const lesson = ALL_LESSONS.find((l) => l.title === lessonTitle);
   670→      if (lesson) {
   671→        setActiveLesson(lesson.id);
   672→      }
   673→    },
   674→    [setActiveLesson]
   675→  );
   676→

   714→              "linear-gradient(135deg, rgba(99,102,241,0.5), rgba(139,92,246,0.4), rgba(34,211,238,0.35), rgba(99,102,241,0.5))",
