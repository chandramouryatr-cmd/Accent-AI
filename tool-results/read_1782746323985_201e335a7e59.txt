    28→/**
    29→ * Returns the "primary audio" text for a lesson step — the text that should
    30→ * be spoken when the user presses Space (or wants a quick replay).
    31→ * Returns `null` for step types that have no meaningful auto-play audio.
    32→ */
    33→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    34→  if (!step) return null;
    35→  switch (step.type) {
    36→    case "intro":
    37→      return step.title;
    38→    case "example":
    39→      return step.phrase;
    40→    case "mouth-diagram":
    41→      return step.exampleWord || null;
    42→    case "compare":
    43→      return step.nativePhrase;
    44→    case "stress-bars":
    45→      return step.word;
    46→    case "rhythm":
    47→      return step.phrase;
    48→    case "linking":
    49→      return step.words.join(" ");
    50→    case "shadow":
    51→      return step.phrase;
    52→    case "intonation":
    53→      return step.phrase;
    54→    case "practice":
    55→      return step.phrase;
    56→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    57→    default:
    58→      return null;
    59→  }
    60→}

    77→export function LessonModal({ lesson, onClose, onNext }: Props) {
    78→  const [stepIdx, setStepIdx] = useState(0);
    79→  const [direction, setDirection] = useState(1);
    80→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
    81→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
    82→  const [recording, setRecording] = useState(false);
    83→  const [showConfetti, setShowConfetti] = useState(false);
    84→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
    85→  const [showSpaceHint, setShowSpaceHint] = useState(false);
    86→  // Track the previous stepIdx so we can reset per-step interactive state
    87→  // when the step changes — using the official "adjust state during render"
    88→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
    89→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
    90→  const [showNotesPanel, setShowNotesPanel] = useState(false);
    91→  const accent = useAppStore((s) => s.accent);
    92→  const completeLesson = useAppStore((s) => s.completeLesson);
    93→  const lessons = useAppStore((s) => s.lessons);
    94→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    95→  const hasLessonNote = useAppStore((s) => (s.lessonNotes[lesson.id] ?? "").trim().length > 0);
    96→
    97→  const step = lesson.steps[stepIdx];
    98→  const isLast = stepIdx === lesson.steps.length - 1;
    99→  const isFirst = stepIdx === 0;
   100→  const totalSteps = lesson.steps.length;
   101→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
   102→
   103→  // Preload voices on mount
   104→  useEffect(() => {
   105→    loadVoices();
   106→  }, []);
   107→
   108→  // Reset per-step interactive state when stepIdx changes.
   109→  // Official React pattern: adjust state during render based on a tracked
   110→  // previous value. React re-renders immediately without committing the
   111→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   112→  if (stepIdx !== prevStepIdx) {
   113→    setPrevStepIdx(stepIdx);
   114→    setQuizAnswer(null);
   115→    setPracticeScore(null);
   116→    setRecording(false);
   117→    // Reset the Space-to-play hint based on whether the new step has audio.
   118→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   119→  }
   120→
   121→  // Stop TTS on unmount or step change
   122→  useEffect(() => {
   123→    return () => stopSpeaking();
   124→  }, [stepIdx]);
   125→
   126→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   127→  // The hint itself is shown/hidden via the render-time adjustment above
   128→  // (avoids setState-in-effect cascades).
   129→  useEffect(() => {
   130→    if (!showSpaceHint) return;
   131→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   132→    return () => clearTimeout(t);
   133→  }, [showSpaceHint]);
   134→
   135→  const handleSpeak = useCallback(
   136→    (text: string) => {
   137→      speak(text, { accent, rate: 0.95 });
   138→    },
   139→    [accent]
   140→  );
   141→
   142→  const goNext = useCallback(() => {
   143→    if (isLast) {
   144→      // already at completion step
   145→      return;
   146→    }
   147→    setDirection(1);
   148→    setStepIdx((i) => i + 1);
   149→  }, [isLast]);
   150→
   151→  const goPrev = useCallback(() => {
   152→    if (isFirst) return;
   153→    setDirection(-1);
   154→    setStepIdx((i) => i - 1);
   155→  }, [isFirst]);
   156→
   157→  // Mark completion when reaching completion step
   158→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   159→  const alreadyCompleted = lessons[lesson.id]?.completed;
   160→
   161→  const handleComplete = useCallback(() => {
   162→    if (alreadyCompleted) return;
   163→    if (completionStep?.type === "completion") {
   164→      completeLesson(
   165→        lesson.id,
   166→        practiceScore ?? 85,
   167→        completionStep.xp,
   168→        completionStep.badge
   169→      );
   170→      addSpeakingTime(8);
   171→      setShowConfetti(true);
   172→      setTimeout(() => setShowConfetti(false), 4000);
   173→    }
   174→  }, [alreadyCompleted, completionStep, completeLesson, lesson.id, practiceScore, addSpeakingTime]);
   175→
   176→  // Auto-trigger completion when user reaches the completion step.
   177→  // handleComplete calls setState (setShowConfetti + store actions), so we
   178→  // defer it out of the synchronous effect body to satisfy
   179→  // react-hooks/set-state-in-effect (no cascading renders).
   180→  useEffect(() => {
   181→    if (step?.type !== "completion" || alreadyCompleted) return;
   182→    const id = setTimeout(() => handleComplete(), 0);
   183→    return () => clearTimeout(id);
   184→  }, [step, alreadyCompleted, handleComplete]);
   185→
   186→  // ESC to close, Arrows to navigate, Space to play step audio
   187→  useEffect(() => {
   188→    const onKey = (e: KeyboardEvent) => {
   189→      if (e.key === "Escape") {
   190→        // Close the notes panel first if it's open; otherwise close the lesson.
   191→        if (showNotesPanel) {
   192→          setShowNotesPanel(false);
   193→          return;
   194→        }
   195→        onClose();
   196→        return;
   197→      }
   198→      if (e.key === "ArrowRight") {
   199→        goNext();
   200→        return;
   201→      }
   202→      if (e.key === "ArrowLeft") {
   203→        goPrev();
   204→        return;
   205→      }
   206→      // Space → play the current step's primary audio
   207→      if (e.key === " " || e.code === "Space") {
   208→        // When the notes panel is open, the user is taking notes — don't
   209→        // hijack Space for audio playback.
   210→        if (showNotesPanel) return;
   211→        const target = e.target as HTMLElement | null;
   212→        const isTyping =
   213→          target &&
   214→          (target.tagName === "INPUT" ||
   215→            target.tagName === "TEXTAREA" ||
   216→            target.isContentEditable);
   217→        if (isTyping) return;
   218→
   219→        // Don't capture Space when another overlay is open that needs it
   220→        // (AI Coach chat input, or this very shortcuts overlay).
   221→        const coachOpen = document.querySelector(
   222→          '[aria-label="AccentAI Coach chat"]'
   223→        );
   224→        if (coachOpen) return;
   225→        const overlay = document.getElementById("shortcuts-overlay");
   226→        if (overlay && !overlay.classList.contains("hidden")) return;
   227→
   228→        // If an interactive element (button / link) is focused, let the
   229→        // browser's native Space-to-click behaviour win so users who tab
   230→        // through the UI can still activate controls with Space.
   231→        const active = document.activeElement as HTMLElement | null;
   232→        if (
   233→          active &&
   234→          (active.tagName === "BUTTON" ||
   235→            active.tagName === "A" ||
   236→            active.getAttribute("role") === "button")
   237→        ) {
   238→          return;
   239→        }
   240→
   241→        const text = getPrimaryAudioText(step);
   242→        if (text) {
   243→          e.preventDefault();
   244→          handleSpeak(text);
   245→        }
   246→      }
   247→    };
   248→    window.addEventListener("keydown", onKey);
   249→    return () => window.removeEventListener("keydown", onKey);
   250→  }, [goNext, goPrev, onClose, step, handleSpeak, showNotesPanel]);
   251→
   252→  return (
   253→    <motion.div
   254→      initial={{ opacity: 0 }}
   255→      animate={{ opacity: 1 }}
   256→      exit={{ opacity: 0 }}
   257→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col"
   258→    >
   259→      {showConfetti && <Confetti count={100} />}
   260→
   261→      {/* Header */}
   262→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   263→        <button
   264→          onClick={onClose}
   265→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   266→          aria-label="Close lesson"
   267→        >
   268→          <X className="w-5 h-5" />
   269→        </button>
   270→        <div className="flex-1 px-3 text-center">
   271→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   272→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   273→          </div>
   274→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   275→        </div>
   276→        <div className="flex items-center gap-1.5">
   277→          <motion.button
   278→            onClick={() => setShowNotesPanel(true)}
   279→            whileTap={{ scale: 0.9 }}
   280→            whileHover={{ scale: 1.05 }}
   281→            className="relative w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition text-[var(--t2)] hover:text-[var(--p3)]"
   282→            aria-label="Open lesson notes"
   283→            aria-expanded={showNotesPanel}
   284→          >
   285→            <NotebookPen style={{ width: 18, height: 18 }} />
   286→            <AnimatePresence>
   287→              {hasLessonNote && (
   288→                <motion.span
   289→                  key="note-dot"
   290→                  initial={{ scale: 0 }}
   291→                  animate={{ scale: 1 }}
   292→                  exit={{ scale: 0 }}
   293→                  transition={{ type: "spring", stiffness: 400, damping: 18 }}
   294→                  className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-[var(--p3)] ring-2 ring-[var(--bg2)]"
   295→                  style={{ boxShadow: "0 0 6px rgba(167,139,250,0.8)" }}
   296→                />
   297→              )}
   298→            </AnimatePresence>
   299→          </motion.button>
   300→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   301→          <ProgressRing pct={pct} size={36} stroke={3} />
   302→        </div>
   303→      </div>
   304→
   305→      {/* Progress bar with glow */}
   306→      <div className="h-1.5 bg-[var(--bg2)] relative">
   307→        <motion.div
   308→          className="h-full"
   309→          style={{ background: "var(--grad-btn)", boxShadow: "0 0 8px rgba(99,102,241,0.6)" }}
   310→          initial={{ width: 0 }}
   311→          animate={{ width: `${pct}%` }}
   312→          transition={{ duration: 0.4 }}
   313→        />
   314→        {/* Shimmer on progress bar */}
   315→        <motion.div
   316→          className="absolute top-0 h-full w-8 bg-gradient-to-r from-transparent via-white/30 to-transparent"
   317→          animate={{ x: ["-100%", "500%"] }}
   318→          transition={{ duration: 2, repeat: Infinity, repeatDelay: 3, ease: "easeInOut" }}
   319→          style={{ left: `${pct - 5}%` }}
   320→        />
   321→      </div>
   322→
   323→      {/* Step progress dots */}
   324→      <div className="px-4 py-2 bg-[var(--bg2)]/60 border-b border-[var(--border)]">
   325→        <div className="flex items-center gap-1 max-w-2xl mx-auto overflow-x-auto scrollbar-none">
   326→          {lesson.steps.map((s, i) => {
   327→            const isCurrent = i === stepIdx;
   328→            const isPast = i < stepIdx;
   329→            const stepIcon: Record<string, string> = {
   330→              intro: "👋", concept: "📖", example: "💬", "mouth-diagram": "👄",
   331→              "vowel-chart": "🎯", compare: "📊", "stress-bars": "📈", rhythm: "🎵",
   332→              linking: "🔗", shadow: "🪞", intonation: "📐", "tap-pronounce": "👆",
   333→              tip: "💡", practice: "🎙", quiz: "❓", completion: "🏆"
   334→            };
   335→            return (
   336→              <motion.button
   337→                key={i}
   338→                onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   339→                className={`shrink-0 w-7 h-7 rounded-full flex items-center justify-center text-[10px] transition-all relative ${
   340→                  isCurrent
   341→                    ? "bg-[var(--grad-btn)] text-white scale-110 shadow-[0_0_12px_rgba(99,102,241,0.5)]"
   342→                    : isPast
   343→                    ? "bg-[rgba(16,185,129,0.2)] text-[#10b981]"
   344→                    : "bg-[var(--card)] text-[var(--t3)] border border-[var(--border)]"
   345→                }`
   346→                }
   347→                whileHover={{ scale: 1.15 }}
   348→                whileTap={{ scale: 0.9 }}
   349→                title={`${s.type}: ${s.title || s.type}`}
   350→              >
   351→                <span className="relative z-10">{stepIcon[s.type] || "•"}</span>
   352→                {isCurrent && (
   353→                  <motion.span
   354→                    className="absolute inset-0 rounded-full border-2 border-[var(--p)]"
   355→                    animate={{ scale: [1, 1.3, 1], opacity: [0.5, 0, 0.5] }}
   356→                    transition={{ duration: 2, repeat: Infinity }}
   357→                  />
   358→                )}
   359→              </motion.button>
   360→            );
   361→          })}
   362→        </div>
   363→      </div>
   364→
   365→      {/* Step content */}
   366→      <div className="flex-1 overflow-y-auto">
   367→        <AnimatePresence mode="wait" custom={direction}>
   368→          <motion.div
   369→            key={stepIdx}
   370→            custom={direction}
   371→            initial={{ opacity: 0, x: direction * 40, scale: 0.97 }}
   372→            animate={{ opacity: 1, x: 0, scale: 1 }}
   373→            exit={{ opacity: 0, x: direction * -40, scale: 0.97 }}
   374→            transition={{ duration: 0.32, ease: [0.34, 1.2, 0.64, 1] }}
   375→            className="max-w-2xl mx-auto px-5 py-6 pb-32"
   376→          >
   377→            {/* Step-type chip with icon + label, animated entrance */}
   378→            <motion.div
   379→              initial={{ opacity: 0, y: -6 }}
   380→              animate={{ opacity: 1, y: 0 }}
   381→              transition={{ delay: 0.1, duration: 0.3 }}
   382→              className="flex items-center justify-center gap-2 mb-4"
   383→            >
   384→              <div className="flex items-center gap-1.5 px-3 py-1 rounded-full bg-[var(--card)] border border-[var(--border2)] text-[10px] uppercase tracking-wider font-mono text-[var(--t3)]">
   385→                <span className="text-sm leading-none">
   386→                  {(() => {
   387→                    const ico: Record<string, string> = {
   388→                      intro: "👋", concept: "📖", example: "💬", "mouth-diagram": "👄",
   389→                      "vowel-chart": "🎯", compare: "📊", "stress-bars": "📈", rhythm: "🎵",
   390→                      linking: "🔗", shadow: "🪞", intonation: "📐", "tap-pronounce": "👆",
   391→                      tip: "💡", practice: "🎙", quiz: "❓", completion: "🏆"
   392→                    };
   393→                    return ico[step?.type || ""] || "•";
   394→                  })()}
   395→                </span>
   396→                <span>{step?.type?.replace("-", " ")}</span>
   397→                <span className="opacity-50">·</span>
   398→                <span>Step {stepIdx + 1} of {totalSteps}</span>
   399→              </div>
   400→            </motion.div>
   401→            <StepRenderer
   402→              step={step}
   403→              speak={handleSpeak}
   404→              quizAnswer={quizAnswer}
   405→              setQuizAnswer={setQuizAnswer}
   406→              practiceScore={practiceScore}
   407→              setPracticeScore={setPracticeScore}
   408→              recording={recording}
   409→              setRecording={setRecording}
   410→              isLast={isLast}
   411→              onComplete={handleComplete}
   412→              onNext={onNext}
   413→            />
   414→          </motion.div>
   415→        </AnimatePresence>
   416→      </div>
   417→
   418→      {/* "Press Space to play" hint — appears briefly above the footer when a
   419→          step with audio loads. pointer-events-none so it never blocks taps. */}
   420→      <AnimatePresence>
   421→        {showSpaceHint && (
   422→          <motion.div
   423→            initial={{ opacity: 0, y: 14, scale: 0.92 }}
   424→            animate={{ opacity: 1, y: 0, scale: 1 }}
   425→            exit={{ opacity: 0, y: 14, scale: 0.92 }}
   426→            transition={{ type: "spring", stiffness: 320, damping: 24 }}
   427→            className="absolute left-1/2 -translate-x-1/2 bottom-24 z-10 pointer-events-none"
   428→          >
   429→            <div className="flex items-center gap-2 px-3.5 py-2 rounded-full bg-[var(--card)]/95 border border-[var(--border2)] backdrop-blur-md shadow-[0_6px_24px_rgba(0,0,0,0.45)]">
   430→              <span className="text-sm leading-none" aria-hidden="true">⌨</span>
   431→              <kbd className="font-mono text-[10px] font-bold px-1.5 py-0.5 rounded bg-[var(--bg2)] border border-[var(--border2)] text-[var(--p3)] shadow-[0_1px_0_var(--border2)]">
   432→                Space
   433→              </kbd>
   434→              <span className="text-xs text-[var(--t2)] font-medium">Press Space to play</span>
   435→            </div>
   436→          </motion.div>
   437→        )}
   438→      </AnimatePresence>
   439→
   440→      {/* Footer nav */}
   441→      <div className="border-t border-[var(--border)] bg-[var(--bg2)]/95 backdrop-blur px-4 py-3 flex items-center gap-3 safe-bottom">
   442→        <button
   443→          onClick={goPrev}
   444→          disabled={isFirst}
   445→          className="px-4 py-2.5 rounded-xl flex items-center gap-1.5 text-sm font-semibold disabled:opacity-30 disabled:cursor-not-allowed hover:bg-[var(--card-h)] transition"
   446→        >
   447→          <ChevronLeft className="w-4 h-4" /> Back
   448→        </button>

   450→        {step?.type === "completion" ? (
   451→          onNext ? (
   452→            <button
   453→              onClick={onNext}
   454→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"

   463→              <Check className="w-4 h-4" /> Finish

   465→          )

   467→          <button
   468→            onClick={goNext}
   469→            className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   470→          >
   471→            {isLast ? "Complete" : "Continue"} <ChevronRight className="w-4 h-4" />
   472→          </button>
   473→        )}
   474→      </div>
   475→
   476→      {/* Lesson Notes — slide-in side panel overlay */}
   477→      <AnimatePresence>
   478→        {showNotesPanel && (
   479→          <motion.div
   480→            key="notes-backdrop"
   481→            initial={{ opacity: 0 }}
   482→            animate={{ opacity: 1 }}
   483→            exit={{ opacity: 0 }}
   484→            transition={{ duration: 0.2 }}
   485→            className="absolute inset-0 z-[60] bg-black/50 backdrop-blur-sm flex justify-end"
   486→            onClick={() => setShowNotesPanel(false)}
   487→            aria-hidden="false"
   488→          >
   489→            <motion.div
   490→              key="notes-panel"
   491→              initial={{ x: "100%" }}
   492→              animate={{ x: 0 }}
   493→              exit={{ x: "100%" }}
   494→              transition={{ type: "spring", stiffness: 320, damping: 32 }}
   495→              onClick={(e) => e.stopPropagation()}
   496→              className="h-full w-full max-w-md bg-[var(--bg)] border-l border-[var(--border2)] flex flex-col safe-top safe-bottom"

   573→    case "rhythm":
   574→      return <RhythmBeats step={step} speak={speak} />;
   575→    case "linking":
   576→      return <LinkingDiagram step={step} speak={speak} />;
   577→    case "intonation":
   578→      return <IntonationContour step={step} speak={speak} />;
   579→    case "shadow":
   580→      return <ShadowStepView step={step} speak={speak} />;
   581→    case "tap-pronounce":
   582→      return <TapPronounceStepView step={step} speak={speak} />;
