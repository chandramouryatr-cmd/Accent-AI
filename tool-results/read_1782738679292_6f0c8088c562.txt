    27→const VISUAL_EMOJI: Record<string, string> = {
    28→  wave: "🌊",
    29→  mouth: "👄",
    30→  "ipa-chart": "🔤",
    31→  "vowel-chart": "🎯",
    32→  "compare-wave": "📊",
    33→  rhythm: "🎵",
    34→  "phoneme-grid": "🔡",
    35→  "stress-bars": "📈",

    41→
    42→export function LessonModal({ lesson, onClose, onNext }: Props) {
    43→  const [stepIdx, setStepIdx] = useState(0);
    44→  const [direction, setDirection] = useState(1);
    45→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
    46→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
    47→  const [recording, setRecording] = useState(false);
    48→  const [showConfetti, setShowConfetti] = useState(false);
    49→  // Track the previous stepIdx so we can reset per-step interactive state
    50→  // when the step changes — using the official "adjust state during render"
    51→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
    52→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
    53→  const accent = useAppStore((s) => s.accent);
    54→  const completeLesson = useAppStore((s) => s.completeLesson);
    55→  const lessons = useAppStore((s) => s.lessons);
    56→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    57→
    58→  const step = lesson.steps[stepIdx];
    59→  const isLast = stepIdx === lesson.steps.length - 1;
    60→  const isFirst = stepIdx === 0;
    61→  const totalSteps = lesson.steps.length;
    62→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
    63→
    64→  // Preload voices on mount
    65→  useEffect(() => {
    66→    loadVoices();
    67→  }, []);
    68→
    69→  // Reset per-step interactive state when stepIdx changes.
    70→  // Official React pattern: adjust state during render based on a tracked
    71→  // previous value. React re-renders immediately without committing the
    72→  // stale intermediate state. (Avoids setState-in-effect cascades.)
    73→  if (stepIdx !== prevStepIdx) {
    74→    setPrevStepIdx(stepIdx);
    75→    setQuizAnswer(null);
    76→    setPracticeScore(null);
    77→    setRecording(false);
    78→  }
    79→
    80→  // Stop TTS on unmount or step change
    81→  useEffect(() => {
    82→    return () => stopSpeaking();
    83→  }, [stepIdx]);
    84→
    85→  const handleSpeak = useCallback(
    86→    (text: string) => {
    87→      speak(text, { accent, rate: 0.95 });
    88→    },
    89→    [accent]
    90→  );
    91→
    92→  const goNext = useCallback(() => {
    93→    if (isLast) {
    94→      // already at completion step
    95→      return;
    96→    }
    97→    setDirection(1);
    98→    setStepIdx((i) => i + 1);
    99→  }, [isLast]);
   100→
   101→  const goPrev = useCallback(() => {
   102→    if (isFirst) return;
   103→    setDirection(-1);
   104→    setStepIdx((i) => i - 1);
   105→  }, [isFirst]);
   106→
   107→  // Mark completion when reaching completion step
   108→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   109→  const alreadyCompleted = lessons[lesson.id]?.completed;
   110→
   111→  const handleComplete = useCallback(() => {
   112→    if (alreadyCompleted) return;
   113→    if (completionStep?.type === "completion") {
   114→      completeLesson(
   115→        lesson.id,
   116→        practiceScore ?? 85,
   117→        completionStep.xp,
   118→        completionStep.badge
   119→      );
   120→      addSpeakingTime(8);
   121→      setShowConfetti(true);
   122→      setTimeout(() => setShowConfetti(false), 4000);
   123→    }
   124→  }, [alreadyCompleted, completionStep, completeLesson, lesson.id, practiceScore, addSpeakingTime]);
   125→
   126→  // Auto-trigger completion when user reaches the completion step.
   127→  // handleComplete calls setState (setShowConfetti + store actions), so we
   128→  // defer it out of the synchronous effect body to satisfy
   129→  // react-hooks/set-state-in-effect (no cascading renders).
   130→  useEffect(() => {
   131→    if (step?.type !== "completion" || alreadyCompleted) return;
   132→    const id = setTimeout(() => handleComplete(), 0);
   133→    return () => clearTimeout(id);
   134→  }, [step, alreadyCompleted, handleComplete]);
   135→
   136→  // ESC to close
   137→  useEffect(() => {
   138→    const onKey = (e: KeyboardEvent) => {
   139→      if (e.key === "Escape") onClose();
   140→      else if (e.key === "ArrowRight") goNext();
   141→      else if (e.key === "ArrowLeft") goPrev();
   142→    };
   143→    window.addEventListener("keydown", onKey);
   144→    return () => window.removeEventListener("keydown", onKey);
   145→  }, [goNext, goPrev, onClose]);
   146→
   147→  return (
   148→    <motion.div
   149→      initial={{ opacity: 0 }}
   150→      animate={{ opacity: 1 }}
   151→      exit={{ opacity: 0 }}
   152→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col"
   153→    >
   154→      {showConfetti && <Confetti count={100} />}
   155→
   156→      {/* Header */}
   157→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   158→        <button
   159→          onClick={onClose}
   160→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   161→          aria-label="Close lesson"
   162→        >
   163→          <X className="w-5 h-5" />
   164→        </button>
   165→        <div className="flex-1 px-3 text-center">
   166→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   167→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   168→          </div>
   169→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   170→        </div>
   171→        <div className="flex items-center gap-1.5">
   172→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   173→          <ProgressRing pct={pct} size={36} stroke={3} />
   174→        </div>
   175→      </div>
   176→
   177→      {/* Progress bar with glow */}
   178→      <div className="h-1.5 bg-[var(--bg2)] relative">
   179→        <motion.div
   180→          className="h-full"
   181→          style={{ background: "var(--grad-btn)", boxShadow: "0 0 8px rgba(99,102,241,0.6)" }}
   182→          initial={{ width: 0 }}
   183→          animate={{ width: `${pct}%` }}
   184→          transition={{ duration: 0.4 }}
   185→        />
   186→        {/* Shimmer on progress bar */}
   187→        <motion.div
   188→          className="absolute top-0 h-full w-8 bg-gradient-to-r from-transparent via-white/30 to-transparent"
   189→          animate={{ x: ["-100%", "500%"] }}
   190→          transition={{ duration: 2, repeat: Infinity, repeatDelay: 3, ease: "easeInOut" }}
   191→          style={{ left: `${pct - 5}%` }}
   192→        />
   193→      </div>
   194→
   195→      {/* Step progress dots */}
   196→      <div className="px-4 py-2 bg-[var(--bg2)]/60 border-b border-[var(--border)]">
   197→        <div className="flex items-center gap-1 max-w-2xl mx-auto overflow-x-auto scrollbar-none">
   198→          {lesson.steps.map((s, i) => {
   199→            const isCurrent = i === stepIdx;
   200→            const isPast = i < stepIdx;
   201→            const stepIcon: Record<string, string> = {
   202→              intro: "👋", concept: "📖", example: "💬", "mouth-diagram": "👄",
   203→              "vowel-chart": "🎯", compare: "📊", "stress-bars": "📈", rhythm: "🎵",
   204→              linking: "🔗", shadow: "🪞", intonation: "📐", "tap-pronounce": "👆",
   205→              tip: "💡", practice: "🎙", quiz: "❓", completion: "🏆"
   206→            };
   207→            return (
   208→              <motion.button
   209→                key={i}
   210→                onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   211→                className={`shrink-0 w-7 h-7 rounded-full flex items-center justify-center text-[10px] transition-all relative ${
   212→                  isCurrent
   213→                    ? "bg-[var(--grad-btn)] text-white scale-110 shadow-[0_0_12px_rgba(99,102,241,0.5)]"
   214→                    : isPast
   215→                    ? "bg-[rgba(16,185,129,0.2)] text-[#10b981]"
   216→                    : "bg-[var(--card)] text-[var(--t3)] border border-[var(--border)]"
   217→                }`
   218→                }
   219→                whileHover={{ scale: 1.15 }}
   220→                whileTap={{ scale: 0.9 }}
   221→                title={`${s.type}: ${s.title || s.type}`}
   222→              >
   223→                <span className="relative z-10">{stepIcon[s.type] || "•"}</span>
   224→                {isCurrent && (
   225→                  <motion.span
   226→                    className="absolute inset-0 rounded-full border-2 border-[var(--p)]"
   227→                    animate={{ scale: [1, 1.3, 1], opacity: [0.5, 0, 0.5] }}
   228→                    transition={{ duration: 2, repeat: Infinity }}
   229→                  />
   230→                )}
   231→              </motion.button>
   232→            );
   233→          })}
   234→        </div>
   235→      </div>
   236→
   237→      {/* Step content */}
   238→      <div className="flex-1 overflow-y-auto">
   239→        <AnimatePresence mode="wait" custom={direction}>
   240→          <motion.div
   241→            key={stepIdx}
   242→            custom={direction}
   243→            initial={{ opacity: 0, x: direction * 30 }}
   244→            animate={{ opacity: 1, x: 0 }}
   245→            exit={{ opacity: 0, x: direction * -30 }}
   246→            transition={{ duration: 0.25 }}
   247→            className="max-w-2xl mx-auto px-5 py-6 pb-32"
   248→          >
   249→            <StepRenderer
   250→              step={step}
   251→              speak={handleSpeak}
   252→              quizAnswer={quizAnswer}
   253→              setQuizAnswer={setQuizAnswer}
   254→              practiceScore={practiceScore}
   255→              setPracticeScore={setPracticeScore}
   256→              recording={recording}
   257→              setRecording={setRecording}
   258→              isLast={isLast}
   259→              onComplete={handleComplete}
   260→              onNext={onNext}
   261→            />
   262→          </motion.div>
   263→        </AnimatePresence>
   264→      </div>
   265→
   266→      {/* Footer nav */}
   267→      <div className="border-t border-[var(--border)] bg-[var(--bg2)]/95 backdrop-blur px-4 py-3 flex items-center gap-3 safe-bottom">
   268→        <button
   269→          onClick={goPrev}
   270→          disabled={isFirst}
   271→          className="px-4 py-2.5 rounded-xl flex items-center gap-1.5 text-sm font-semibold disabled:opacity-30 disabled:cursor-not-allowed hover:bg-[var(--card-h)] transition"
   272→        >
   273→          <ChevronLeft className="w-4 h-4" /> Back
   274→        </button>
   275→        <div className="flex-1" />
   276→        {step?.type === "completion" ? (
   277→          onNext ? (
   278→            <button
   279→              onClick={onNext}
   280→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   281→            >
   282→              Next Lesson <ChevronRight className="w-4 h-4" />
   283→            </button>
   284→          ) : (
   285→            <button
   286→              onClick={onClose}
   287→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   288→            >
   289→              <Check className="w-4 h-4" /> Finish
   290→            </button>
   291→          )
   292→        ) : (
   293→          <button
   294→            onClick={goNext}
   295→            className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   296→          >
   297→            {isLast ? "Complete" : "Continue"} <ChevronRight className="w-4 h-4" />
   298→          </button>
   299→        )}
   300→      </div>
   301→    </motion.div>
   302→  );
   303→}

   305→// ─── Step Renderer ───
   306→
   307→interface StepRendererProps {
   308→  step: LessonStep;
   309→  speak: (text: string) => void;
   310→  quizAnswer: number | null;
   311→  setQuizAnswer: (n: number | null) => void;
   312→  practiceScore: number | null;
   313→  setPracticeScore: (n: number | null) => void;
   314→  recording: boolean;
   315→  setRecording: (b: boolean) => void;
   316→  isLast: boolean;
   317→  onComplete: () => void;
   318→  onNext?: () => void;
   319→}
   320→
   321→function StepRenderer(props: StepRendererProps) {
   322→  const { step, speak } = props;
   323→
   324→  switch (step.type) {
   325→    case "intro":
   326→      return <IntroStepView step={step} speak={speak} />;
   327→    case "concept":
   328→      return <ConceptStepView step={step} />;
   329→    case "example":
   330→      return <ExampleStepView step={step} speak={speak} />;
   331→    case "mouth-diagram":
   332→      return <MouthDiagram step={step} speak={speak} />;
   333→    case "vowel-chart":
   334→      return <VowelChart step={step} speak={speak} />;
   335→    case "compare":
   336→      return <CompareWave step={step} speak={speak} />;
   337→    case "stress-bars":
   338→      return <StressBars step={step} speak={speak} />;
   339→    case "rhythm":
   340→      return <RhythmBeats step={step} speak={speak} />;
   341→    case "linking":
   342→      return <LinkingDiagram step={step} speak={speak} />;
   343→    case "intonation":
   344→      return <IntonationContour step={step} speak={speak} />;
   345→    case "shadow":
   346→      return <ShadowStepView step={step} speak={speak} />;
   347→    case "tap-pronounce":
   348→      return <TapPronounceStepView step={step} speak={speak} />;
   349→    case "tip":
   350→      return <TipStepView step={step} />;
   351→    case "practice":
   352→      return <PracticeStepView {...props} />;
   353→    case "quiz":
   354→      return <QuizStepView {...props} />;
   355→    case "completion":
   356→      return <CompletionStepView step={step} onNext={props.onNext} />;
   357→    default:
   358→      return <div>Unknown step type</div>;
   359→  }
   360→}

   470→function ExampleStepView({ step, speak }: { step: Extract<LessonStep, { type: "example" }>; speak: (t: string) => void }) {
   471→  const words = step.phrase.split(/\s+/);
   472→  return (
   473→    <div className="space-y-4">
   474→      {step.title && (
   475→        <div className="flex items-center gap-2">
   476→          <span className="text-xs px-2.5 py-1 rounded-full bg-[rgba(34,211,238,0.15)] text-[#22d3ee] font-mono font-bold uppercase tracking-wider border border-[rgba(34,211,238,0.2)]">
   477→            Example
   478→          </span>
   479→        </div>
   480→      )}
   481→      <div className="rounded-2xl p-5 bg-[rgba(99,102,241,0.06)] border border-[rgba(99,102,241,0.25)] relative overflow-hidden">
   482→        {/* Background glow */}
   483→        <div className="absolute -top-10 -right-10 w-32 h-32 rounded-full bg-[rgba(99,102,241,0.06)] blur-[40px] pointer-events-none" />
   484→        
   485→        <div className="flex flex-wrap gap-1.5 mb-4 relative z-10">
   486→          {words.map((w, i) => {
   487→            const cleaned = w.replace(/[.,!?;:"']/g, "");
   488→            const isHl = step.highlightWords?.some((hw) => hw.replace(/[.,!?;:"']/g, "") === cleaned);
   489→            return (
   490→              <motion.button
   491→                key={i}
   492→                initial={{ opacity: 0, y: 5 }}
