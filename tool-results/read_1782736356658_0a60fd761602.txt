    18→// ─── Tip of the Day ───────────────────────────────────────────────────────
    19→function dayOfYearTipIndex(): number {
    20→  const now = new Date();
    21→  const start = new Date(now.getFullYear(), 0, 0);
    22→  const dayOfYear = Math.floor(
    23→    (now.getTime() - start.getTime()) / 86400000
    24→  );
    25→  return dayOfYear % TIPS.length;
    26→}
