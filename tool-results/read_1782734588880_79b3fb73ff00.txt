    17→// ─── Tip of the Day ───────────────────────────────────────────────────────
    18→function dayOfYearTipIndex(): number {
    19→  const now = new Date();
    20→  const start = new Date(now.getFullYear(), 0, 0);
    21→  const dayOfYear = Math.floor(
    22→    (now.getTime() - start.getTime()) / 86400000
    23→  );
    24→  return dayOfYear % TIPS.length;
    25→}

   621→                    Complete lessons to see your phoneme mastery levels
   622→                  </p>
   623→                )}
   624→              </>
   625→            );
   626→          })()}
   627→        </div>
   628→      </div>
   629→
   630→      {/* Quick actions */}
   631→      <div>
   632→        <h2 className="font-d text-base font-bold mb-2">Quick Actions</h2>
   633→        <div className="grid grid-cols-3 gap-2">
   634→          {[
   635→            { icon: "📚", lbl: "Continue", tab: "journey" as const },

   645→            >
   646→              <div className="text-2xl mb-1">{a.icon}</div>
   647→              <div className="text-xs font-medium text-[var(--t2)]">{a.lbl}</div>

   649→          ))}
   650→        </div>
   651→      </div>
