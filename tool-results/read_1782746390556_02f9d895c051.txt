    28→    28→/**
    29→    29→ * Returns the "primary audio" text for a lesson step — the text that should
    30→    30→ * be spoken when the user presses Space (or wants a quick replay).
    31→    31→ * Returns `null` for step types that have no meaningful auto-play audio.
    32→    32→ */
    33→    33→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    34→    34→  if (!step) return null;
    35→    35→  switch (step.type) {
    36→    36→    case "intro":
    37→    37→      return step.title;
    38→    38→    case "example":
    39→    39→      return step.phrase;
    40→    40→    case "mouth-diagram":
    41→    41→      return step.exampleWord || null;
    42→    42→    case "compare":
    43→    43→      return step.nativePhrase;
    44→    44→    case "stress-bars":
    45→    45→      return step.word;
    46→    46→    case "rhythm":
    47→    47→      return step.phrase;
    48→    48→    case "linking":
    49→    49→      return step.words.join(" ");
    50→    50→    case "shadow":
    51→    51→      return step.phrase;
    52→    52→    case "intonation":
    53→    53→      return step.phrase;
    54→    54→    case "practice":
    55→    55→      return step.phrase;
    56→    56→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    57→    57→    default:
    58→    58→      return null;
    59→    59→  }
    60→    60→}

   261→   261→      {/* Header */}

   263→   263→        <button
   264→   264→          onClick={onClose}
   265→   265→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   266→   266→          aria-label="Close lesson"
   267→   267→        >
   268→   268→          <X className="w-5 h-5" />
   269→   269→        </button>
   270→   270→        <div className="flex-1 px-3 text-center">

   541→   541→interface StepRendererProps {
   542→   542→  step: LessonStep;
   543→   543→  speak: (text: string) => void;
   544→   544→  quizAnswer: number | null;
   545→   545→  setQuizAnswer: (n: number | null) => void;

   555→   555→function StepRenderer(props: StepRendererProps) {
   556→   556→  const { step, speak } = props;
   557→   557→
   558→   558→  switch (step.type) {
   559→   559→    case "intro":
   560→   560→      return <IntroStepView step={step} speak={speak} />;
   561→   561→    case "concept":
   562→   562→      return <ConceptStepView step={step} />;
   563→   563→    case "example":
   564→   564→      return <ExampleStepView step={step} speak={speak} />;
   565→   565→    case "mouth-diagram":
   566→   566→      return <MouthDiagram step={step} speak={speak} />;
   567→   567→    case "vowel-chart":
   568→   568→      return <VowelChart step={step} speak={speak} />;
   569→   569→    case "compare":
   570→   570→      return <CompareWave step={step} speak={speak} />;
   571→   571→    case "stress-bars":
   572→   572→      return <StressBars step={step} speak={speak} />;
   573→   573→    case "rhythm":
   574→   574→      return <RhythmBeats step={step} speak={speak} />;
   575→   575→    case "linking":
   576→   576→      return <LinkingDiagram step={step} speak={speak} />;
   577→   577→    case "intonation":
   578→   578→      return <IntonationContour step={step} speak={speak} />;
   579→   579→    case "shadow":
   580→   580→      return <ShadowStepView step={step} speak={speak} />;
   581→   581→    case "tap-pronounce":
   582→   582→      return <TapPronounceStepView step={step} speak={speak} />;
   583→   583→    case "tip":
   584→   584→      return <TipStepView step={step} />;
   585→   585→    case "practice":
   586→   586→      return <PracticeStepView {...props} />;
   587→   587→    case "quiz":
   588→   588→      return <QuizStepView {...props} />;
   589→   589→    case "completion":
   590→   590→      return <CompletionStepView step={step} onNext={props.onNext} />;
   591→   591→    default:
   592→   592→      return <div>Unknown step type</div>;
   593→   593→  }
   594→   594→}
