    15→    15→// ─── Animated Counter Hook ─────────────────────────────────────────────────
    16→    16→function useAnimatedCounter(target: number, duration = 1200): number {
    17→    17→  const [display, setDisplay] = useState(0);
    18→    18→  const prevRef = useRef(0);
    19→    19→  useEffect(() => {
    20→    20→    if (typeof target !== "number" || isNaN(target)) return;
    21→    21→    const start = prevRef.current;
    22→    22→    const diff = target - start;
    23→    23→    if (diff === 0) return;
    24→    24→    const startTime = performance.now();
    25→    25→    let raf: number;
    26→    26→    const tick = (now: number) => {
    27→    27→      const elapsed = now - startTime;
    28→    28→      const progress = Math.min(1, elapsed / duration);
    29→    29→      // ease-out cubic
    30→    30→      const eased = 1 - Math.pow(1 - progress, 3);
    31→    31→      const current = Math.round(start + diff * eased);
    32→    32→      setDisplay(current);
    33→    33→      if (progress < 1) {
    34→    34→        raf = requestAnimationFrame(tick);
    35→    35→      } else {
    36→    36→        prevRef.current = target;
    37→    37→      }
    38→    38→    };
    39→    39→    raf = requestAnimationFrame(tick);
    40→    40→    return () => cancelAnimationFrame(raf);
    41→    41→  }, [target, duration]);
    42→    42→  return display;
    43→    43→}

    45→    45→function AnimatedStatValue({ value }: { value: number | string }) {
    46→    46→  // If value is a string like "85%" or "3m", animate the numeric part
    47→    47→  const str = String(value);
    48→    48→  const match = str.match(/^(\d+(?:\.\d+)?)(.*)$/);
    49→    49→  const num = match ? parseFloat(match[1]) : 0;
    50→    50→  const suffix = match ? match[2] : "";
    51→    51→  const animated = useAnimatedCounter(num);
    52→    52→  if (!match) return <>{str}</>;
    53→    53→  return <>{animated}{suffix}</>;
    54→    54→}
