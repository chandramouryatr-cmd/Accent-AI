    30→/**
    31→ * Returns the "primary audio" text for a lesson step — the text that should
    32→ * be spoken when the user presses Space (or wants a quick replay).
    33→ * Returns `null` for step types that have no meaningful auto-play audio.
    34→ */
    35→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    36→  if (!step) return null;
    37→  switch (step.type) {
    38→    case "intro":
    39→      return step.title;
    40→    case "example":
    41→      return step.phrase;
    42→    case "mouth-diagram":
    43→      return step.exampleWord || null;
    44→    case "compare":
    45→      return step.nativePhrase;
    46→    case "stress-bars":
    47→      return step.word;
    48→    case "rhythm":
    49→      return step.phrase;
    50→    case "linking":
    51→      return step.words.join(" ");
    52→    case "shadow":
    53→      return step.phrase;
    54→    case "intonation":
    55→      return step.phrase;
    56→    case "practice":
    57→      return step.phrase;
    58→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    59→    default:
    60→      return null;
    61→  }
    62→}

   165→/** Format seconds as mm:ss */
   166→function formatTimer(seconds: number): string {
   167→  const m = Math.floor(seconds / 60);
   168→  const s = Math.floor(seconds % 60);
   169→  return `${m}:${s.toString().padStart(2, "0")}`;
   170→}

   172→export function LessonModal({ lesson, onClose, onNext }: Props) {
   173→  const [stepIdx, setStepIdx] = useState(0);
   174→  const [direction, setDirection] = useState(1);
   175→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
   176→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
   177→  const [recording, setRecording] = useState(false);
   178→  const [showConfetti, setShowConfetti] = useState(false);
   179→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
   180→  const [showSpaceHint, setShowSpaceHint] = useState(false);
   181→  // Track the previous stepIdx so we can reset per-step interactive state
   182→  // when the step changes — using the official "adjust state during render"
   183→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
   184→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
   185→  const [showNotesPanel, setShowNotesPanel] = useState(false);
   186→  const [ttsSpeed, setTtsSpeed] = useState(1);
   187→
   188→  // ── Lesson Timer ──
   189→  const [timerDisplay, setTimerDisplay] = useState("0:00");
   190→  const timerStartRef = useRef<number>(Date.now());
   191→  const timerAccumulatedRef = useRef<number>(lessonTimerAccumulated.get(lesson.id) ?? 0);
   192→
   193→  const accent = useAppStore((s) => s.accent);
   194→  const completeLesson = useAppStore((s) => s.completeLesson);
   195→  const markReviewed = useAppStore((s) => s.markReviewed);
   196→  const lessons = useAppStore((s) => s.lessons);
   197→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
   198→  const hasLessonNote = useAppStore((s) => (s.lessonNotes[lesson.id] ?? "").trim().length > 0);
   199→
   200→  const step = lesson.steps[stepIdx];
   201→  const isLast = stepIdx === lesson.steps.length - 1;
   202→  const isFirst = stepIdx === 0;
   203→  const totalSteps = lesson.steps.length;
   204→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
   205→
   206→  // Preload voices on mount
   207→  useEffect(() => {
   208→    loadVoices();
   209→  }, []);
   210→
   211→  // ── Timer tick effect ──
   212→  useEffect(() => {
   213→    timerStartRef.current = Date.now();
   214→    const tick = () => {
   215→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   216→      setTimerDisplay(formatTimer(elapsed));
   217→    };
   218→    tick(); // initial tick
   219→    const id = setInterval(tick, 1000);
   220→    return () => {
   221→      clearInterval(id);
   222→      // Save accumulated time for this lesson when modal unmounts (pause)
   223→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   224→      timerAccumulatedRef.current = elapsed;
   225→      lessonTimerAccumulated.set(lesson.id, elapsed);
   226→    };
   227→  }, [lesson.id]);
   228→
   229→  // Reset per-step interactive state when stepIdx changes.
   230→  // Official React pattern: adjust state during render based on a tracked
   231→  // previous value. React re-renders immediately without committing the
   232→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   233→  if (stepIdx !== prevStepIdx) {
   234→    setPrevStepIdx(stepIdx);
   235→    setQuizAnswer(null);
   236→    setPracticeScore(null);
   237→    setRecording(false);
   238→    // Reset the Space-to-play hint based on whether the new step has audio.
   239→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   240→  }
   241→
   242→  // Stop TTS on unmount or step change
   243→  useEffect(() => {
   244→    return () => stopSpeaking();
   245→  }, [stepIdx]);
   246→
   247→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   248→  // The hint itself is shown/hidden via the render-time adjustment above
   249→  // (avoids setState-in-effect cascades).
   250→  useEffect(() => {
   251→    if (!showSpaceHint) return;
   252→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   253→    return () => clearTimeout(t);
   254→  }, [showSpaceHint]);
   255→
   256→  const handleSpeak = useCallback(
   257→    (text: string) => {
   258→      speak(text, { accent, rate: ttsSpeed });
   259→    },
   260→    [accent, ttsSpeed]
   261→  );
   262→
   263→  const goNext = useCallback(() => {
   264→    if (isLast) {
   265→      // already at completion step
   266→      return;
   267→    }
   268→    setDirection(1);
   269→    setStepIdx((i) => i + 1);
   270→  }, [isLast]);
   271→
   272→  const goPrev = useCallback(() => {
   273→    if (isFirst) return;
   274→    setDirection(-1);
   275→    setStepIdx((i) => i - 1);
   276→  }, [isFirst]);
   277→
   278→  // Mark completion when reaching completion step
   279→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   280→  const alreadyCompleted = lessons[lesson.id]?.completed;
   281→
   282→  const handleComplete = useCallback(() => {
   283→    if (alreadyCompleted) return;
   284→    if (completionStep?.type === "completion") {
   285→      // Calculate total time spent
   286→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   287→      completeLesson(
   288→        lesson.id,
   289→        practiceScore ?? 85,
   290→        completionStep.xp,
   291→        completionStep.badge,
   292→        Math.floor(elapsed)
   293→      );
   294→      // Mark as reviewed since they just completed it
   295→      markReviewed(lesson.id);
   296→      addSpeakingTime(8);
   297→      setShowConfetti(true);
   298→      setTimeout(() => setShowConfetti(false), 4000);
   299→      // Clear timer state for this lesson
   300→      lessonTimerAccumulated.delete(lesson.id);
   301→    }
   302→  }, [alreadyCompleted, completionStep, completeLesson, markReviewed, lesson.id, practiceScore, addSpeakingTime]);
   303→
   304→  // Auto-trigger completion when user reaches the completion step.
   305→  // handleComplete calls setState (setShowConfetti + store actions), so we
   306→  // defer it out of the synchronous effect body to satisfy
   307→  // react-hooks/set-state-in-effect (no cascading renders).
   308→  useEffect(() => {
   309→    if (step?.type !== "completion" || alreadyCompleted) return;
   310→    const id = setTimeout(() => handleComplete(), 0);
   311→    return () => clearTimeout(id);
   312→  }, [step, alreadyCompleted, handleComplete]);
   313→
   314→  // ESC to close, Arrows to navigate, Space to play step audio
   315→  useEffect(() => {
   316→    const onKey = (e: KeyboardEvent) => {
   317→      if (e.key === "Escape") {
   318→        // Close the notes panel first if it's open; otherwise close the lesson.
   319→        if (showNotesPanel) {
   320→          setShowNotesPanel(false);
   321→          return;
   322→        }
   323→        onClose();
   324→        return;
   325→      }
   326→      if (e.key === "ArrowRight") {
   327→        goNext();
   328→        return;
   329→      }
   330→      if (e.key === "ArrowLeft") {
   331→        goPrev();
   332→        return;
   333→      }
   334→      // Space → play the current step's primary audio
   335→      if (e.key === " " || e.code === "Space") {
   336→        // When the notes panel is open, the user is taking notes — don't
   337→        // hijack Space for audio playback.
   338→        if (showNotesPanel) return;
   339→        const target = e.target as HTMLElement | null;
   340→        const isTyping =
   341→          target &&
   342→          (target.tagName === "INPUT" ||
   343→            target.tagName === "TEXTAREA" ||
   344→            target.isContentEditable);
   345→        if (isTyping) return;
   346→
   347→        // Don't capture Space when another overlay is open that needs it
   348→        // (AI Coach chat input, or this very shortcuts overlay).
   349→        const coachOpen = document.querySelector(
   350→          '[aria-label="AccentAI Coach chat"]'
   351→        );
   352→        if (coachOpen) return;
   353→        const overlay = document.getElementById("shortcuts-overlay");
   354→        if (overlay && !overlay.classList.contains("hidden")) return;
   355→
   356→        // If an interactive element (button / link) is focused, let the
   357→        // browser's native Space-to-click behaviour win so users who tab
   358→        // through the UI can still activate controls with Space.
   359→        const active = document.activeElement as HTMLElement | null;
   360→        if (
   361→          active &&
   362→          (active.tagName === "BUTTON" ||
   363→            active.tagName === "A" ||
   364→            active.getAttribute("role") === "button")
   365→        ) {
   366→          return;
   367→        }
   368→
   369→        const text = getPrimaryAudioText(step);
   370→        if (text) {
   371→          e.preventDefault();
   372→          handleSpeak(text);
   373→        }
   374→      }
   375→    };
   376→    window.addEventListener("keydown", onKey);
   377→    return () => window.removeEventListener("keydown", onKey);
   378→  }, [goNext, goPrev, onClose, step, handleSpeak, showNotesPanel]);
   379→
   380→  const currentCategory = getStepCategory(step?.type ?? "");
   381→  const currentTint = CATEGORY_TINT[currentCategory];
   382→  const currentGlow = CATEGORY_GLOW[currentCategory];
   383→
   384→  return (
   385→    <motion.div
   386→      initial={{ opacity: 0 }}
   387→      animate={{ opacity: 1 }}
   388→      exit={{ opacity: 0 }}
   389→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col relative overflow-hidden"
   390→    >
   391→      {/* Animated step-type background tint */}
   392→      <motion.div
   393→        key={currentCategory}
   394→        initial={{ opacity: 0 }}
   395→        animate={{ opacity: 1 }}
   396→        exit={{ opacity: 0 }}
   397→        transition={{ duration: 0.6 }}
   398→        className="absolute inset-0 pointer-events-none z-0"
   399→        style={{ background: `radial-gradient(ellipse at 50% 0%, ${currentTint} 0%, transparent 70%)` }}
   400→      />
   401→      {/* Content sits above the tint */}
   402→      <div className="relative z-10 flex flex-col flex-1 min-h-0">
   403→      {showConfetti && <Confetti count={100} />}
   404→
   405→      {/* Header */}
   406→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   407→        <button
   408→          onClick={onClose}
   409→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   410→          aria-label="Close lesson"
   411→        >
   412→          <X className="w-5 h-5" />
   413→        </button>
   414→        <div className="flex-1 px-3 text-center">
   415→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   416→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   417→          </div>
   418→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   419→        </div>
   420→        <div className="flex items-center gap-1.5">
   421→          <motion.button
   422→            onClick={() => setShowNotesPanel(true)}
   423→            whileTap={{ scale: 0.9 }}
   424→            whileHover={{ scale: 1.05 }}
   425→            className="relative w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition text-[var(--t2)] hover:text-[var(--p3)]"
   426→            aria-label="Open lesson notes"
   427→            aria-expanded={showNotesPanel}
   428→          >
   429→            <NotebookPen style={{ width: 18, height: 18 }} />
   430→            <AnimatePresence>
   431→              {hasLessonNote && (
   432→                <motion.span
   433→                  key="note-dot"
   434→                  initial={{ scale: 0 }}
   435→                  animate={{ scale: 1 }}
   436→                  exit={{ scale: 0 }}
   437→                  transition={{ type: "spring", stiffness: 400, damping: 18 }}
   438→                  className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-[var(--p3)] ring-2 ring-[var(--bg2)]"
   439→                  style={{ boxShadow: "0 0 6px rgba(167,139,250,0.8)" }}
   440→                />
   441→              )}
   442→            </AnimatePresence>
   443→          </motion.button>
   444→          {/* Timer display */}
   445→          <div className="flex items-center gap-1 px-2 py-1 rounded-lg bg-[var(--card)]/60 border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]">
   446→            <span className="text-[10px]">⏱</span>
   447→            <span>{timerDisplay}</span>
   448→          </div>
   449→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   450→          <ProgressRing pct={pct} size={36} stroke={3} />
   451→        </div>
   452→      </div>
   453→
   454→      {/* ── Enhanced animated step progress bar ── */}
   455→      <div className="px-3 py-2 bg-[var(--bg2)]/70 border-b border-[var(--border)]">
   456→        <div className="relative max-w-2xl mx-auto" style={{ height: 44 }}>
   457→          {/* Gradient connecting line (background track) */}
   458→          <div className="absolute left-0 right-0 top-1/2 -translate-y-1/2 h-[3px] rounded-full bg-[var(--border)]" />
   459→
   460→          {/* Animated gradient fill line — draws from left to the current step */}
   461→          <motion.div
   462→            className="absolute top-1/2 -translate-y-1/2 h-[3px] rounded-full"
   463→            style={{
   464→              left: 0,
   465→              width: `calc(${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}% + 0px)`,
   466→              background: "linear-gradient(90deg, var(--p), var(--p2))",
   467→              boxShadow: "0 0 8px rgba(99,102,241,0.4)",
   468→            }}
   469→            initial={{ scaleX: 0, transformOrigin: "left" }}
   470→            animate={{ scaleX: 1 }}
   471→            transition={{ duration: 0.5, ease: "easeOut" }}
   472→          />
   473→
   474→          {/* Shimmer sweep on the filled line */}
   475→          <motion.div
   476→            className="absolute top-1/2 -translate-y-1/2 h-[3px] rounded-full"
   477→            style={{
   478→              left: 0,
   479→              width: `calc(${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}% + 0px)`,

   485→          />
   486→
   487→          {/* Step dots row */}
   488→          <div className="absolute inset-0 flex items-center justify-between">
   489→            {lesson.steps.map((s, i) => {
   490→              const isCurrent = i === stepIdx;
   491→              const isPast = i < stepIdx;
   492→              const isUpcoming = i > stepIdx;
   493→              const cat = getStepCategory(s.type);
   494→              const glow = CATEGORY_GLOW[cat];
   495→              const icon = STEP_ICON[s.type] || "•";
   496→              const stepTitle = s.title || s.type.replace(/-/g, " ");
   497→
   498→              return (
   499→                <motion.button
   500→                  key={i}
   501→                  onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   502→                  className="relative flex items-center justify-center rounded-full cursor-pointer group"

   732→interface StepRendererProps {
   733→  step: LessonStep;
   734→  lesson: Lesson;
   735→  speak: (text: string) => void;
   736→  ttsSpeed: number;
   737→  setTtsSpeed: (s: number) => void;
   738→  quizAnswer: number | null;
   739→  setQuizAnswer: (n: number | null) => void;
   740→  practiceScore: number | null;
   741→  setPracticeScore: (n: number | null) => void;
   742→  recording: boolean;
   743→  setRecording: (b: boolean) => void;
   744→  isLast: boolean;
   745→  onComplete: () => void;
   746→  onNext?: () => void;
   747→  timeSpentSeconds: number;
   748→}
   749→
   750→function StepRenderer(props: StepRendererProps) {
   751→  const { step, lesson, speak, ttsSpeed, setTtsSpeed } = props;
   752→
   753→  switch (step.type) {
   754→    case "intro":
   755→      return <IntroStepView step={step} lesson={lesson} speak={speak} ttsSpeed={ttsSpeed} setTtsSpeed={setTtsSpeed} />;
   756→    case "concept":
   757→      return <ConceptStepView step={step} />;
   758→    case "example":
   759→      return <ExampleStepView step={step} speak={speak} />;
   760→    case "mouth-diagram":
   761→      return <MouthDiagram step={step} speak={speak} />;
   762→    case "vowel-chart":
   763→      return <VowelChart step={step} speak={speak} />;
   764→    case "compare":
   765→      return <CompareWave step={step} speak={speak} />;
   766→    case "stress-bars":
   767→      return <StressBars step={step} speak={speak} />;
   768→    case "rhythm":
   769→      return <RhythmBeats step={step} speak={speak} />;
   770→    case "linking":
   771→      return <LinkingDiagram step={step} speak={speak} />;
   772→    case "intonation":
   773→      return <IntonationContour step={step} speak={speak} />;
   774→    case "shadow":
   775→      return <ShadowStepView step={step} speak={speak} />;
   776→    case "tap-pronounce":
   777→      return <TapPronounceStepView step={step} speak={speak} />;
   778→    case "tip":
   779→      return <TipStepView step={step} />;
   780→    case "practice":
   781→      return <PracticeStepView {...props} />;
   782→    case "quiz":
   783→      return <QuizStepView {...props} />;
   784→    case "completion":
   785→      return <CompletionStepView step={step} onNext={props.onNext} timeSpentSeconds={props.timeSpentSeconds} practiceScore={props.practiceScore} />;
   786→    default:
   787→      return <div>Unknown step type</div>;
   788→  }
   789→}
