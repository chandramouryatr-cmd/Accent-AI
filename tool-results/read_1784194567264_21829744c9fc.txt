    35→/**
    36→ * Returns the "primary audio" text for a lesson step — the text that should
    37→ * be spoken when the user presses Space (or wants a quick replay).
    38→ * Returns `null` for step types that have no meaningful auto-play audio.
    39→ */
    40→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    41→  if (!step) return null;
    42→  switch (step.type) {
    43→    case "intro":
    44→      return step.title;
    45→    case "example":
    46→      return step.phrase;
    47→    case "mouth-diagram":
    48→      return step.exampleWord || null;
    49→    case "compare":
    50→      return step.nativePhrase;
    51→    case "stress-bars":
    52→      return step.word;
    53→    case "rhythm":
    54→      return step.phrase;
    55→    case "linking":
    56→      return step.words.join(" ");
    57→    case "shadow":
    58→      return step.phrase;
    59→    case "intonation":
    60→      return step.phrase;
    61→    case "practice":
    62→      return step.phrase;
    63→    case "karaoke":
    64→      return step.phrase;
    65→    case "contrast":
    66→      return step.phrase;
    67→    case "role-play":
    68→      return step.turns.find((t) => t.speaker === "coach")?.text ?? null;
    69→    case "phone-call":
    70→      return step.line;
    71→    case "media-clip":
    72→      return step.phrase;
    73→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion, ipa-tutorial
    74→    default:
    75→      return null;
    76→  }
    77→}

   149→/** Format seconds as mm:ss */
   150→function formatTimer(seconds: number): string {
   151→  const m = Math.floor(seconds / 60);
   152→  const s = Math.floor(seconds % 60);
   153→  return `${m}:${s.toString().padStart(2, "0")}`;
   154→}

   156→export function LessonModal({ lesson, onClose, onNext }: Props) {
   157→  const [stepIdx, setStepIdx] = useState(0);
   158→  const [direction, setDirection] = useState(1);
   159→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
   160→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
   161→  const [recording, setRecording] = useState(false);
   162→  const [showConfetti, setShowConfetti] = useState(false);
   163→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
   164→  const [showSpaceHint, setShowSpaceHint] = useState(false);
   165→  // Track the previous stepIdx so we can reset per-step interactive state
   166→  // when the step changes — using the official "adjust state during render"
   167→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
   168→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
   169→  const [showNotesPanel, setShowNotesPanel] = useState(false);
   170→  const ttsSpeed = 0.95;
   171→  // Speaking state — drives the header indicator (Bug 3 fix)
   172→  const [speaking, setSpeaking] = useState(false);
   173→
   174→  // ── Lesson Timer ──
   175→  const [timerDisplay, setTimerDisplay] = useState("0:00");
   176→  const timerStartRef = useRef<number>(Date.now());
   177→  const timerAccumulatedRef = useRef<number>(lessonTimerAccumulated.get(lesson.id) ?? 0);
   178→
   179→  // Scroll container ref — reset to top on step change (Bug 1 fix)
   180→  const scrollContainerRef = useRef<HTMLDivElement>(null);
   181→  // Root ref — for attaching one-time TTS unlock listener (Bug 3 fix)
   182→  const rootRef = useRef<HTMLDivElement>(null);
   183→
   184→  const accent = useAppStore((s) => s.accent);
   185→  const completeLesson = useAppStore((s) => s.completeLesson);
   186→  const markReviewed = useAppStore((s) => s.markReviewed);
   187→  const lessons = useAppStore((s) => s.lessons);
   188→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
   189→  const hasLessonNote = useAppStore((s) => (s.lessonNotes[lesson.id] ?? "").trim().length > 0);
   190→
   191→  const step = lesson.steps[stepIdx];
   192→  const isLast = stepIdx === lesson.steps.length - 1;
   193→  const isFirst = stepIdx === 0;
   194→  const totalSteps = lesson.steps.length;
   195→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
   196→
   197→  // Preload voices on mount
   198→  useEffect(() => {
   199→    loadVoices();
   200→  }, []);
   201→
   202→  // Reset scroll position to top whenever the step changes (Bug 1 fix).
   203→  // Without this, scrolling down on step N opens step N+1 already scrolled
   204→  // to the bottom.
   205→  useEffect(() => {
   206→    if (scrollContainerRef.current) {
   207→      scrollContainerRef.current.scrollTop = 0;
   208→    }
   209→  }, [stepIdx]);
   210→
   211→  // Unlock TTS on first user interaction with the modal (Bug 3 fix).
   212→  // Mobile Safari / Chrome Android block speechSynthesis.speak() until a
   213→  // user gesture has occurred. We attach a one-time pointerdown listener
   214→  // to the modal root that calls unlockTTS() and then removes itself.
   215→  useEffect(() => {
   216→    const root = rootRef.current;
   217→    if (!root) return;
   218→    const onFirstInteract = () => {
   219→      unlockTTS();
   220→      root.removeEventListener("pointerdown", onFirstInteract);
   221→    };
   222→    root.addEventListener("pointerdown", onFirstInteract);
   223→    return () => root.removeEventListener("pointerdown", onFirstInteract);
   224→  }, []);
   225→
   226→  // Subscribe to TTS speaking state so the header indicator stays in sync
   227→  // with actual audio playback (Bug 3 fix).
   228→  useEffect(() => {
   229→    const unsub = onSpeakingChange(() => setSpeaking(isSpeaking()));
   230→    setSpeaking(isSpeaking());
   231→    return unsub;
   232→  }, []);
   233→
   234→  // ── Timer tick effect ──
   235→  useEffect(() => {
   236→    timerStartRef.current = Date.now();
   237→    const tick = () => {
   238→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   239→      setTimerDisplay(formatTimer(elapsed));
   240→    };
   241→    tick(); // initial tick
   242→    const id = setInterval(tick, 1000);
   243→    return () => {
   244→      clearInterval(id);
   245→      // Save accumulated time for this lesson when modal unmounts (pause)
   246→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   247→      timerAccumulatedRef.current = elapsed;
   248→      lessonTimerAccumulated.set(lesson.id, elapsed);
   249→    };
   250→  }, [lesson.id]);
   251→
   252→  // Reset per-step interactive state when stepIdx changes.
   253→  // Official React pattern: adjust state during render based on a tracked
   254→  // previous value. React re-renders immediately without committing the
   255→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   256→  if (stepIdx !== prevStepIdx) {
   257→    setPrevStepIdx(stepIdx);
   258→    setQuizAnswer(null);
   259→    setPracticeScore(null);
   260→    setRecording(false);
   261→    // Reset the Space-to-play hint based on whether the new step has audio.
   262→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   263→  }
   264→
   265→  // Stop TTS on unmount or step change
   266→  useEffect(() => {
   267→    return () => stopSpeaking();
   268→  }, [stepIdx]);
   269→
   270→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   271→  // The hint itself is shown/hidden via the render-time adjustment above
   272→  // (avoids setState-in-effect cascades).
   273→  useEffect(() => {
   274→    if (!showSpaceHint) return;
   275→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   276→    return () => clearTimeout(t);
   277→  }, [showSpaceHint]);
   278→
   279→  const handleSpeak = useCallback(
   280→    (text: string) => {
   281→      speakTTS(text, { accent, rate: ttsSpeed });
   282→    },
   283→    [accent, ttsSpeed]
   284→  );
   285→
   286→  const goNext = useCallback(() => {
   287→    if (isLast) {
   288→      // already at completion step
   289→      return;
   290→    }
   291→    setDirection(1);
   292→    setStepIdx((i) => i + 1);
   293→  }, [isLast]);
   294→
   295→  const goPrev = useCallback(() => {
   296→    if (isFirst) return;
   297→    setDirection(-1);
   298→    setStepIdx((i) => i - 1);
   299→  }, [isFirst]);
   300→
   301→  // Mark completion when reaching completion step
   302→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   303→  const alreadyCompleted = lessons[lesson.id]?.completed;
   304→
   305→  const handleComplete = useCallback(() => {
   306→    if (alreadyCompleted) return;
   307→    if (completionStep?.type === "completion") {
   308→      // Calculate total time spent
   309→      const elapsed = timerAccumulatedRef.current + (Date.now() - timerStartRef.current) / 1000;
   310→      completeLesson(
   311→        lesson.id,
   312→        practiceScore ?? 85,
   313→        completionStep.xp,
   314→        completionStep.badge,
   315→        Math.floor(elapsed)
   316→      );
   317→      // Mark as reviewed since they just completed it
   318→      markReviewed(lesson.id);
   319→      addSpeakingTime(8);
   320→      setShowConfetti(true);
   321→      setTimeout(() => setShowConfetti(false), 4000);
   322→      // Clear timer state for this lesson
   323→      lessonTimerAccumulated.delete(lesson.id);
   324→    }
   325→  }, [alreadyCompleted, completionStep, completeLesson, markReviewed, lesson.id, practiceScore, addSpeakingTime]);
   326→
   327→  // Auto-trigger completion when user reaches the completion step.
   328→  // handleComplete calls setState (setShowConfetti + store actions), so we
   329→  // defer it out of the synchronous effect body to satisfy
   330→  // react-hooks/set-state-in-effect (no cascading renders).
   331→  useEffect(() => {
   332→    if (step?.type !== "completion" || alreadyCompleted) return;
   333→    const id = setTimeout(() => handleComplete(), 0);
   334→    return () => clearTimeout(id);
   335→  }, [step, alreadyCompleted, handleComplete]);
   336→
   337→  // ESC to close, Arrows to navigate, Space to play step audio
   338→  useEffect(() => {
   339→    const onKey = (e: KeyboardEvent) => {
   340→      if (e.key === "Escape") {
   341→        // Close the notes panel first if it's open; otherwise close the lesson.
   342→        if (showNotesPanel) {
   343→          setShowNotesPanel(false);
   344→          return;
   345→        }
   346→        onClose();
   347→        return;
   348→      }
   349→      if (e.key === "ArrowRight") {
   350→        goNext();
   351→        return;
   352→      }
   353→      if (e.key === "ArrowLeft") {
   354→        goPrev();
   355→        return;
   356→      }
   357→      // Space → play the current step's primary audio
   358→      if (e.key === " " || e.code === "Space") {
   359→        // When the notes panel is open, the user is taking notes — don't
   360→        // hijack Space for audio playback.
   361→        if (showNotesPanel) return;
   362→        const target = e.target as HTMLElement | null;
   363→        const isTyping =
   364→          target &&
   365→          (target.tagName === "INPUT" ||
   366→            target.tagName === "TEXTAREA" ||
   367→            target.isContentEditable);
   368→        if (isTyping) return;
   369→
   370→        // Don't capture Space when another overlay is open that needs it
   371→        // (AI Coach chat input, or this very shortcuts overlay).
   372→        const coachOpen = document.querySelector(
   373→          '[aria-label="AccentAI Coach chat"]'
   374→        );
   375→        if (coachOpen) return;
   376→        const overlay = document.getElementById("shortcuts-overlay");
   377→        if (overlay && !overlay.classList.contains("hidden")) return;
   378→
   379→        // If an interactive element (button / link) is focused, let the
   380→        // browser's native Space-to-click behaviour win so users who tab
   381→        // through the UI can still activate controls with Space.
   382→        const active = document.activeElement as HTMLElement | null;
   383→        if (
   384→          active &&
   385→          (active.tagName === "BUTTON" ||
   386→            active.tagName === "A" ||
   387→            active.getAttribute("role") === "button")
   388→        ) {
   389→          return;
   390→        }
   391→
   392→        const text = getPrimaryAudioText(step);
   393→        if (text) {
   394→          e.preventDefault();
   395→          handleSpeak(text);
   396→        }
   397→      }
   398→    };
   399→    window.addEventListener("keydown", onKey);
   400→    return () => window.removeEventListener("keydown", onKey);
   401→  }, [goNext, goPrev, onClose, step, handleSpeak, showNotesPanel]);
   402→
   403→  return (
   404→    <motion.div
   405→      ref={rootRef}
   406→      initial={{ opacity: 0 }}
   407→      animate={{ opacity: 1 }}
   408→      exit={{ opacity: 0 }}
   409→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col overflow-hidden"
   410→    >
   411→      {/* Content */}
   412→      <div className="relative z-10 flex flex-col flex-1 min-h-0">
   413→      {showConfetti && <Confetti count={100} />}
   414→
   415→      {/* Header */}
   416→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg)] safe-top">
   417→        <button
   418→          onClick={onClose}
   419→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   420→          aria-label="Close lesson"
   421→        >
   422→          <X className="w-5 h-5" />
   423→        </button>
   424→        <div className="flex-1 px-3 text-center">
   425→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   426→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   427→          </div>
   428→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   429→        </div>
   430→        <div className="flex items-center gap-1.5">
   431→          <motion.button
   432→            onClick={() => setShowNotesPanel(true)}
   433→            whileTap={{ scale: 0.9 }}
   434→            whileHover={{ scale: 1.05 }}
   435→            className="relative w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition text-[var(--t2)] hover:text-[var(--p3)]"
   436→            aria-label="Open lesson notes"
   437→            aria-expanded={showNotesPanel}
   438→          >
   439→            <NotebookPen style={{ width: 18, height: 18 }} />
   440→            <AnimatePresence>
   441→              {hasLessonNote && (
   442→                <motion.span
   443→                  key="note-dot"
   444→                  initial={{ scale: 0 }}
   445→                  animate={{ scale: 1 }}
   446→                  exit={{ scale: 0 }}
   447→                  transition={{ type: "spring", stiffness: 400, damping: 18 }}
   448→                  className="absolute top-1.5 right-1.5 w-2 h-2 rounded-full bg-[var(--p3)] ring-2 ring-[var(--bg2)]"
   449→                  style={{ boxShadow: "0 0 6px rgba(167,139,250,0.8)" }}
   450→                />
   451→              )}
   452→            </AnimatePresence>
   453→          </motion.button>
   454→          {/* Speaking indicator (Bug 3 fix) — pulsing dot + icon while TTS is active */}
   455→          {speaking && (
   456→            <span
   457→              className="flex items-center gap-1 px-2 py-1 rounded-lg border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]"
   458→              aria-label="Audio playing"
   459→            >
   460→              <motion.span
   461→                animate={{ scale: [1, 1.5, 1], opacity: [1, 0.4, 1] }}
   462→                transition={{ duration: 0.9, repeat: Infinity, ease: "easeInOut" }}
   463→                className="w-1.5 h-1.5 rounded-full bg-[var(--p)]"
   464→              />
   465→              <Volume2 className="w-3 h-3" />
   466→            </span>
   467→          )}
   468→          {/* Timer display */}
   469→          <div className="flex items-center gap-1 px-2 py-1 rounded-lg border border-[var(--border2)] text-[11px] font-mono text-[var(--t2)]">
   470→            <span className="text-[10px]">⏱</span>
   471→            <span>{timerDisplay}</span>
   472→          </div>
   473→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   474→          <ProgressRing pct={pct} size={36} stroke={3} />
   475→        </div>
   476→      </div>
   477→
   478→      {/* ── Minimal step progress bar ── */}
   479→      <div className="px-3 py-2 bg-[var(--bg)] border-b border-[var(--border)]">
   480→        <div className="relative max-w-2xl mx-auto" style={{ height: 32 }}>
   481→          {/* Thin gray track */}
   482→          <div className="absolute left-0 right-0 top-1/2 -translate-y-1/2 h-[2px] rounded-full bg-[var(--border)]" />
   483→
   484→          {/* Solid fill up to the current step */}
   485→          <div
   486→            className="absolute top-1/2 -translate-y-1/2 h-[2px] rounded-full bg-[var(--p)] transition-all duration-300 ease-out"
   487→            style={{
   488→              left: 0,
   489→              width: `${(stepIdx / Math.max(totalSteps - 1, 1)) * 100}%`,
   490→            }}
   491→          />
   492→
   493→          {/* Step dots — current = solid p, past = solid gray, upcoming = outline */}
   494→          <div className="absolute inset-0 flex items-center justify-between">
   495→            {lesson.steps.map((s, i) => {
   496→              const isCurrent = i === stepIdx;
   497→              const isPast = i < stepIdx;
   498→              const isUpcoming = i > stepIdx;
   499→              const icon = STEP_ICON[s.type] || "•";
   500→              const stepTitle = s.title || s.type.replace(/-/g, " ");
   501→
   502→              return (
   503→                <button
   504→                  key={i}
   505→                  onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   506→                  className="relative flex items-center justify-center rounded-full cursor-pointer group transition-transform hover:scale-110 active:scale-95"
   507→                  style={{
   508→                    width: isCurrent ? 26 : 20,
   509→                    height: isCurrent ? 26 : 20,
   510→                    fontSize: isCurrent ? 11 : 9,
   511→                    background: isCurrent
   512→                      ? "var(--p)"
   513→                      : isPast
   514→                      ? "var(--t2)"
   515→                      : "var(--bg)",
   516→                    border: isUpcoming ? "1px solid var(--border2)" : "none",

   709→interface StepRendererProps {
   710→  step: LessonStep;
   711→  lesson: Lesson;

   713→  quizAnswer: number | null;
   714→  setQuizAnswer: (n: number | null) => void;

   725→function StepRenderer(props: StepRendererProps) {
   726→  const { step, lesson, speak } = props;
   727→
   728→  switch (step.type) {
   729→    case "intro":

   735→    case "mouth-diagram":

   737→    case "vowel-chart":
   738→      return <VowelChart step={step} speak={speak} />;
   739→    case "compare":
   740→      return <CompareWave step={step} speak={speak} />;
   741→    case "stress-bars":
   742→      return <StressBars step={step} speak={speak} />;
   743→    case "rhythm":
   744→      return <RhythmBeats step={step} speak={speak} />;
   745→    case "linking":
   746→      return <LinkingDiagram step={step} speak={speak} />;
   747→    case "intonation":
   748→      return <IntonationContour step={step} speak={speak} />;
   749→    case "shadow":
   750→      return <ShadowStepView step={step} speak={speak} />;
   751→    case "tap-pronounce":
   752→      return <TapPronounceStepView step={step} speak={speak} />;
   753→    case "ipa-tutorial":
   754→      return <IpaTutorial step={step} speak={speak} />;
   755→    case "karaoke":
   756→      return <KaraokePhrase step={step} speak={speak} />;
   757→    case "role-play":
   758→      return <RolePlay step={step} speak={speak} />;
   759→    case "contrast":
   760→      return <ContrastToggle step={step} speak={speak} />;
   761→    case "phone-call":
   762→      return <PhoneCall step={step} speak={speak} />;
   763→    case "media-clip":
   764→      return <MediaClip step={step} speak={speak} />;
   765→    case "tip":
   766→      return <TipStepView step={step} />;
   767→    case "practice":
   768→      return <PracticeStepView {...props} />;
   769→    case "quiz":
   770→      return <QuizStepView {...props} />;
   771→    case "completion":
   772→      return <CompletionStepView step={step} onNext={props.onNext} timeSpentSeconds={props.timeSpentSeconds} practiceScore={props.practiceScore} />;
   773→    default:
   774→      return <div>Unknown step type</div>;
   775→  }
