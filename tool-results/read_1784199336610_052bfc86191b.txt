     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import { AnimatePresence, motion } from "framer-motion";
     5→import {
     6→  BookOpen,
     7→  Volume2,
     8→  X,
     9→  Search,
    10→  Loader2,
    11→  AlertCircle,
    12→  Compass,
    13→  History,
    14→  ExternalLink,
    15→} from "lucide-react";
    16→import { speak as speakTTS, unlockTTS } from "@/lib/tts";
    17→
    18→/* ────────────────────────────────────────────────────────────────────
    19→ * DictionaryLookup
    20→ *
    21→ * A two-stage widget:
    22→ *   1. A floating "📖 Dictionary" pill that appears whenever the user
    23→ *      selects a word anywhere in the app (mouseup / keyboard selection
    24→ *      / mobile long-press). Positioned just above the selection rect.
    25→ *   2. A centered modal popup that opens when the pill is clicked. Shows
    26→ *      the word, IPA, audio button, definitions, examples, synonyms, and
    27→ *      a Cancel button. Falls back to in-app TTS when the dictionary
    28→ *      service has no native audio.
    29→ *
    30→ * Wired globally via AppShell so it works on every view.
    31→ * ──────────────────────────────────────────────────────────────────── */
    32→
    33→interface SelectionInfo {
    34→  text: string;
    35→  rect: { top: number; left: number; width: number; height: number };
    36→}
    37→
    38→interface Meaning {
    39→  partOfSpeech: string;
    40→  definitions: {
    41→    definition: string;
    42→    example?: string;
    43→    synonyms?: string[];
    44→    antonyms?: string[];
    45→  }[];
    46→  synonyms?: string[];
    47→  antonyms?: string[];
    48→}
    49→
    50→interface DictResult {
    51→  word: string;
    52→  phonetic?: string;
    53→  ipa?: string;
    54→  audioUrl?: string;
    55→  meanings: Meaning[];
    56→  origin?: string;
    57→  sourceUrl?: string;
    58→  fromCache?: boolean;
    59→}
    60→
    61→const RECENT_KEY = "accentai-dict-recent";
    62→const MAX_RECENT = 8;
    63→
    64→function loadRecent(): string[] {
    65→  if (typeof window === "undefined") return [];
    66→  try {
    67→    const raw = localStorage.getItem(RECENT_KEY);
    68→    if (!raw) return [];
    69→    const arr = JSON.parse(raw);
    70→    return Array.isArray(arr) ? arr.filter((x) => typeof x === "string").slice(0, MAX_RECENT) : [];
    71→  } catch {
    72→    return [];
    73→  }
    74→}
    75→
    76→function saveRecent(word: string) {
    77→  if (typeof window === "undefined") return;
    78→  try {
    79→    const prev = loadRecent();
    80→    const lower = word.toLowerCase();
    81→    const next = [word, ...prev.filter((w) => w.toLowerCase() !== lower)].slice(0, MAX_RECENT);
    82→    localStorage.setItem(RECENT_KEY, JSON.stringify(next));
    83→  } catch {
    84→    /* ignore quota errors */
    85→  }
    86→}
    87→
    88→/** A single-word selection is 1–64 chars, letters + optional internal
    89→ *  apostrophes / hyphens, no whitespace. */
    90→function isSingleWord(text: string): boolean {
    91→  const t = text.trim();
    92→  if (t.length < 2 || t.length > 64) return false;
    93→  return /^[A-Za-z][A-Za-z''\-]*$/.test(t);
    94→}
    95→
    96→export function DictionaryLookup() {
    97→  /* ── Floating-pill selection state ── */
    98→  const [selection, setSelection] = useState<SelectionInfo | null>(null);
    99→  const [pillVisible, setPillVisible] = useState(false);
   100→
   101→  /* ── Modal state ── */
   102→  const [open, setOpen] = useState(false);
   103→  const [query, setQuery] = useState("");
   104→  const [result, setResult] = useState<DictResult | null>(null);
   105→  const [status, setStatus] = useState<"idle" | "loading" | "error" | "notfound" | "ok">("idle");
   106→  const [errorMsg, setErrorMsg] = useState("");
   107→  const [suggestion, setSuggestion] = useState<string | undefined>();
   108→  const [recent, setRecent] = useState<string[]>([]);
   109→  const [audioLoading, setAudioLoading] = useState(false);
   110→  const [playedOnce, setPlayedOnce] = useState(false);
   111→
   112→  const abortRef = useRef<AbortController | null>(null);
   113→  const modalCardRef = useRef<HTMLDivElement>(null);
   114→  const cancelBtnRef = useRef<HTMLButtonElement>(null);
   115→  const inputRef = useRef<HTMLInputElement>(null);
   116→
   117→  /* ── Listen for text selection across the document ── */
   118→  useEffect(() => {
   119→    let rafId = 0;
   120→    const check = () => {
   121→      rafId = 0;
   122→      // Don't show the pill while the modal is open
   123→      if (open) {
   124→        setSelection(null);
   125→        setPillVisible(false);
   126→        return;
   127→      }
   128→      const sel = window.getSelection();
   129→      const text = sel?.toString().trim() ?? "";
   130→      if (!sel || !text || !isSingleWord(text)) {
   131→        setSelection(null);
   132→        setPillVisible(false);
   133→        return;
   134→      }
   135→      // Ignore selections inside form fields / editable regions / our own popup
   136→      const anchor = sel.anchorNode;
   137→      if (!anchor) {
   138→        setSelection(null);
   139→        setPillVisible(false);
   140→        return;
   141→      }
   142→      const el = anchor.nodeType === Node.TEXT_NODE ? anchor.parentElement : (anchor as Element | null);
   143→      if (el?.closest('input, textarea, [contenteditable="true"], [data-dictionary-popup="true"]')) {
   144→        setSelection(null);
   145→        setPillVisible(false);
   146→        return;
   147→      }
   148→      try {
   149→        if (sel.rangeCount === 0) return;
   150→        const range = sel.getRangeAt(0);
   151→        const rect = range.getBoundingClientRect();
   152→        if (rect.width === 0 && rect.height === 0) return;
   153→        setSelection({ text, rect: { top: rect.top, left: rect.left, width: rect.width, height: rect.height } });
   154→        setPillVisible(true);
   155→      } catch {
   156→        /* rangeCount can throw in some edge cases */
   157→      }
   158→    };
   159→
   160→    const schedule = () => {
   161→      if (rafId) return;
   162→      rafId = requestAnimationFrame(check);
   163→    };
   164→
   165→    // mouseup + keyup = end of user-initiated selection (desktop)
   166→    document.addEventListener("mouseup", schedule);
   167→    document.addEventListener("keyup", schedule);
   168→    // selectionchange = mobile long-press + selection-handle dragging
   169→    document.addEventListener("selectionchange", schedule);
   170→
   171→    // Any scroll invalidates the cached rect — hide the pill.
   172→    const onScroll = () => {
   173→      setSelection(null);
   174→      setPillVisible(false);
   175→    };
   176→    window.addEventListener("scroll", onScroll, true);
   177→    window.addEventListener("resize", onScroll);
   178→
   179→    return () => {
   180→      if (rafId) cancelAnimationFrame(rafId);
   181→      document.removeEventListener("mouseup", schedule);
   182→      document.removeEventListener("keyup", schedule);
   183→      document.removeEventListener("selectionchange", schedule);
   184→      window.removeEventListener("scroll", onScroll, true);
   185→      window.removeEventListener("resize", onScroll);
   186→    };
   187→  }, [open]);
   188→
   189→  /* ── Load recent lookups on mount ── */
   190→  useEffect(() => {
   191→    setRecent(loadRecent());
   192→  }, []);
   193→
   194→  /* ── Core lookup function ── */
   195→  const lookup = useCallback(async (word: string) => {
   196→    const w = word.trim();
   197→    if (!w) return;
   198→    // Abort any in-flight request
   199→    abortRef.current?.abort();
   200→    const ctrl = new AbortController();
   201→    abortRef.current = ctrl;
   202→
   203→    setQuery(w);
   204→    setResult(null);
   205→    setStatus("loading");
   206→    setErrorMsg("");
   207→    setSuggestion(undefined);
   208→    setPlayedOnce(false);
   209→    setOpen(true);
   210→    setSelection(null);
   211→    setPillVisible(false);
   212→    // Clear the native selection so it doesn't visually persist behind the modal
   213→    try {
   214→      window.getSelection()?.removeAllRanges();
   215→    } catch {
   216→      /* ignore */
   217→    }
   218→    // Unlock TTS audio on this user gesture (mobile Safari requirement)
   219→    unlockTTS();
   220→
   221→    try {
   222→      const res = await fetch(`/api/dictionary?word=${encodeURIComponent(w)}`, {
   223→        signal: ctrl.signal,
   224→      });
   225→      if (res.status === 404) {
   226→        const body = await res.json().catch(() => ({}));
   227→        setStatus("notfound");
   228→        setSuggestion(body.suggestion);
   229→        return;
   230→      }
   231→      if (!res.ok) {
   232→        const body = await res.json().catch(() => ({}));
   233→        setStatus("error");
   234→        setErrorMsg(body.error || `Lookup failed (HTTP ${res.status}).`);
   235→        return;
   236→      }
   237→      const data = (await res.json()) as DictResult;
   238→      setResult(data);
   239→      setStatus("ok");
   240→      saveRecent(data.word || w);
   241→      setRecent(loadRecent());
   242→    } catch (err) {
   243→      if ((err as Error).name === "AbortError") return;
   244→      setStatus("error");
   245→      setErrorMsg("Couldn't reach the dictionary service. Check your connection and try again.");
   246→    }
   247→  }, []);
   248→
   249→  /* ── Close modal ── */
   250→  const close = useCallback(() => {
   251→    abortRef.current?.abort();
   252→    setOpen(false);
   253→    setResult(null);
   254→    setStatus("idle");
   255→    setQuery("");
   256→    setErrorMsg("");
   257→    setSuggestion(undefined);
   258→    setPlayedOnce(false);
   259→  }, []);
   260→
   261→  /* ── Open modal empty (no pre-filled word) — used by the header
   262→   *  dictionary button. Opens straight to a searchable state showing
   263→   *  recent lookups so the user can type or tap a recent word. ── */
   264→  const openEmpty = useCallback(() => {
   265→    abortRef.current?.abort();
   266→    setResult(null);
   267→    setStatus("idle");
   268→    setQuery("");
   269→    setErrorMsg("");
   270→    setSuggestion(undefined);
   271→    setPlayedOnce(false);
   272→    setSelection(null);
   273→    setPillVisible(false);
   274→    setOpen(true);
   275→    unlockTTS();
   276→    // Focus the search input shortly after the modal mounts
   277→    setTimeout(() => inputRef.current?.focus(), 120);
   278→  }, []);
   279→
   280→  /* ── Listen for global "open-dictionary" events (from the header
   281→   *  dictionary button). Supports an optional detail.word to look up
   282→   *  immediately, or no detail to open empty. ── */
   283→  useEffect(() => {
   284→    const onOpen = (e: Event) => {
   285→      const detail = (e as CustomEvent).detail as { word?: string } | undefined;
   286→      if (detail?.word) {
   287→        lookup(detail.word);
   288→      } else {
   289→        openEmpty();
   290→      }
   291→    };
   292→    window.addEventListener("accentai:open-dictionary", onOpen as EventListener);
   293→    return () => window.removeEventListener("accentai:open-dictionary", onOpen as EventListener);
   294→  }, [lookup, openEmpty]);
   295→
   296→  /* ── Body scroll lock + Esc-to-close while modal is open ── */
   297→  useEffect(() => {
   298→    if (!open) return;
   299→    const prev = document.body.style.overflow;
   300→    document.body.style.overflow = "hidden";
   301→    const onKey = (e: KeyboardEvent) => {
   302→      if (e.key === "Escape") {
   303→        e.preventDefault();
   304→        close();
   305→      }
   306→    };
   307→    window.addEventListener("keydown", onKey);
   308→    // Focus the cancel button for keyboard users (after the entrance anim)
   309→    const t = setTimeout(() => cancelBtnRef.current?.focus(), 120);
   310→    return () => {
   311→      document.body.style.overflow = prev;
   312→      window.removeEventListener("keydown", onKey);
   313→      clearTimeout(t);
   314→    };
   315→  }, [open, close]);
   316→
   317→  /* ── Click on the floating pill ── */
   318→  const onPillClick = useCallback(() => {
   319→    if (selection?.text) lookup(selection.text);
   320→  }, [selection, lookup]);
   321→
   322→  /* ── Play pronunciation ── */
   323→  const playAudio = useCallback(() => {
   324→    if (!result) return;
   325→    setAudioLoading(true);
   326→    setPlayedOnce(true);
   327→    try {
   328→      if (result.audioUrl) {
   329→        const audio = new Audio(result.audioUrl);
   330→        audio.crossOrigin = "anonymous";
   331→        audio.onended = () => setAudioLoading(false);
   332→        audio.onerror = () => {
   333→          // Fall back to TTS if the audio URL fails
   334→          speakTTS(result.word, { rate: 0.9 });
   335→          setTimeout(() => setAudioLoading(false), 600);
   336→        };
   337→        audio.play().catch(() => {
   338→          // Autoplay rejection / network error → TTS fallback
   339→          speakTTS(result.word, { rate: 0.9 });
   340→          setTimeout(() => setAudioLoading(false), 600);
   341→        });
   342→      } else {
   343→        // No native audio — use the in-app TTS with the learner's accent
   344→        speakTTS(result.word, { rate: 0.9 });
   345→        setTimeout(() => setAudioLoading(false), 600);
   346→      }
   347→    } catch {
   348→      setAudioLoading(false);
   349→    }
   350→  }, [result]);
   351→
   352→  /* ── Submit handler for the in-modal search input ── */
   353→  const onSubmitSearch = (e: React.FormEvent) => {
   354→    e.preventDefault();
   355→    const v = query.trim();
   356→    if (v) lookup(v);
   357→  };
   358→
   359→  /* ── Pill position: centered above the selection, clamped to viewport ── */
   360→  const pillStyle = selection
   361→    ? (() => {
   362→        const PILL_W = 132; // estimated; pill is roughly this wide
   363→        const PILL_H = 36;
   364→        const gap = 8;
   365→        let left = selection.rect.left + selection.rect.width / 2 - PILL_W / 2;
   366→        let top = selection.rect.top - PILL_H - gap;
   367→        // Clamp horizontally
   368→        left = Math.max(8, Math.min(left, window.innerWidth - PILL_W - 8));
   369→        // If too close to the top, flip below
   370→        if (top < 8) {
   371→          top = selection.rect.top + selection.rect.height + gap;
   372→        }
   373→        return { left, top };
   374→      })()
   375→    : { left: 0, top: 0 };
   376→
   377→  return (
   378→    <>
   379→      {/* ── Floating "Dictionary" pill ── */}
   380→      <AnimatePresence>
   381→        {pillVisible && selection && (
   382→          <motion.button
   383→            key="dict-pill"
   384→            initial={{ opacity: 0, y: 6, scale: 0.92 }}
   385→            animate={{ opacity: 1, y: 0, scale: 1 }}
   386→            exit={{ opacity: 0, y: 4, scale: 0.95 }}
   387→            transition={{ duration: 0.16, ease: [0.25, 0.46, 0.45, 0.94] }}
   388→            onClick={onPillClick}
   389→            onMouseDown={(e) => e.preventDefault()} // don't steal focus / collapse selection
   390→            style={{ position: "fixed", left: pillStyle.left, top: pillStyle.top, zIndex: 80 }}
   391→            className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-[var(--p)] text-white text-xs font-semibold shadow-lg hover:opacity-90 active:scale-95 transition"
   392→            aria-label={`Look up "${selection.text}" in dictionary`}
   393→          >
   394→            <BookOpen className="w-3.5 h-3.5" />
   395→            Dictionary
   396→          </motion.button>
   397→        )}
   398→      </AnimatePresence>
   399→
   400→      {/* ── Modal popup ── */}
   401→      <AnimatePresence>
   402→        {open && (
   403→          <motion.div
   404→            key="dict-backdrop"
   405→            initial={{ opacity: 0 }}
   406→            animate={{ opacity: 1 }}
   407→            exit={{ opacity: 0 }}
   408→            transition={{ duration: 0.18 }}
   409→            className="fixed inset-0 z-[90] bg-black/50 backdrop-blur-sm flex items-end sm:items-center justify-center p-0 sm:p-4"
   410→            onClick={close}
   411→            data-dictionary-popup="true"
   412→          >
   413→            <motion.div
   414→              key="dict-card"
   415→              ref={modalCardRef}
   416→              initial={{ opacity: 0, y: 30, scale: 0.97 }}
   417→              animate={{ opacity: 1, y: 0, scale: 1 }}
   418→              exit={{ opacity: 0, y: 20, scale: 0.98 }}
   419→              transition={{ duration: 0.22, ease: [0.25, 0.46, 0.45, 0.94] }}
   420→              onClick={(e) => e.stopPropagation()}
   421→              role="dialog"
   422→              aria-modal="true"
   423→              aria-label={`Dictionary lookup for ${query}`}
   424→              className="w-full sm:max-w-md bg-[var(--bg)] border border-[var(--border)] rounded-t-3xl sm:rounded-3xl shadow-2xl flex flex-col max-h-[92vh] sm:max-h-[85vh] overflow-hidden"
   425→            >
   426→              {/* ── Header: search bar ── */}
   427→              <div className="flex items-center gap-2 px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]">
   428→                <div className="w-8 h-8 rounded-lg flex items-center justify-center bg-[var(--p)] text-white shrink-0">
   429→                  <BookOpen className="w-4 h-4" />
   430→                </div>
   431→                <form onSubmit={onSubmitSearch} className="flex-1 flex items-center gap-2">
   432→                  <div className="relative flex-1">
   433→                    <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-[var(--t3)] pointer-events-none" />
   434→                    <input
   435→                      ref={inputRef}
   436→                      type="text"
   437→                      value={query}
   438→                      onChange={(e) => setQuery(e.target.value)}
   439→                      placeholder="Look up a word…"
   440→                      className="w-full pl-8 pr-3 py-2 rounded-lg bg-[var(--bg)] border border-[var(--border)] text-sm text-[var(--t1)] placeholder:text-[var(--t3)] focus:outline-none focus:border-[var(--p)] focus:ring-1 focus:ring-[var(--p)] transition"
   441→                      autoComplete="off"
   442→                      spellCheck={false}
   443→                    />
   444→                  </div>
   445→                  <button
   446→                    type="submit"
   447→                    className="px-3 py-2 rounded-lg bg-[var(--p)] text-white text-xs font-semibold hover:opacity-90 active:scale-95 transition"
   448→                  >
   449→                    Go
   450→                  </button>
   451→                </form>
   452→                <button
   453→                  onClick={close}
   454→                  className="w-8 h-8 rounded-lg flex items-center justify-center text-[var(--t2)] hover:text-[var(--t1)] hover:bg-[var(--card-h)] transition shrink-0"
   455→                  aria-label="Close dictionary"
   456→                >
   457→                  <X className="w-4 h-4" />
   458→                </button>
   459→              </div>
   460→
   461→              {/* ── Body: result / loading / error / idle ── */}
   462→              <div className="flex-1 overflow-y-auto px-4 py-4">
   463→                {status === "idle" && (
   464→                  <div className="space-y-4">
   465→                    {/* Friendly prompt */}
   466→                    <div className="flex flex-col items-center justify-center py-6 px-4 text-center">
   467→                      <div className="w-12 h-12 rounded-full bg-[var(--card-h)] flex items-center justify-center mb-3">
   468→                        <BookOpen className="w-6 h-6 text-[var(--t2)]" />
   469→                      </div>
   470→                      <p className="text-sm font-semibold text-[var(--t1)] mb-1">
   471→                        Look up any word
   472→                      </p>
   473→                      <p className="text-xs text-[var(--t2)] max-w-xs leading-relaxed">
   474→                        Type a word above and press <span className="font-mono font-semibold text-[var(--t1)]">Go</span>, or select any word in a lesson to see its definition, IPA, and pronunciation.
   475→                      </p>
   476→                    </div>
   477→
   478→                    {/* Recent lookups */}
   479→                    {recent.length > 0 && (
   480→                      <div>
   481→                        <div className="flex items-center gap-1.5 text-[10px] uppercase tracking-wider font-mono text-[var(--t3)] mb-2">
   482→                          <History className="w-3 h-3" /> Recent lookups
   483→                        </div>
   484→                        <div className="flex flex-wrap gap-1.5">
   485→                          {recent.map((w) => (
   486→                            <button
   487→                              key={w}
   488→                              onClick={() => lookup(w)}
   489→                              className="px-2.5 py-1 rounded-full bg-[var(--card-h)] border border-[var(--border)] text-xs text-[var(--t2)] hover:text-[var(--t1)] hover:border-[var(--p)] active:scale-95 transition"
   490→                            >
   491→                              {w}
   492→                            </button>
   493→                          ))}
   494→                        </div>
   495→                      </div>
   496→                    )}
   497→                  </div>
   498→                )}
   499→
   500→                {status === "loading" && <LoadingSkeleton />}
   501→
   502→                {status === "error" && (
   503→                  <div className="flex flex-col items-center justify-center py-10 px-4 text-center">
   504→                    <div className="w-12 h-12 rounded-full bg-[rgba(220,38,38,0.1)] flex items-center justify-center mb-3">
   505→                      <AlertCircle className="w-6 h-6 text-[var(--rd)]" />
   506→                    </div>
   507→                    <p className="text-sm font-semibold text-[var(--t1)] mb-1">Something went wrong</p>
   508→                    <p className="text-xs text-[var(--t2)] mb-4 max-w-xs">{errorMsg}</p>
   509→                    <button
   510→                      onClick={() => lookup(query)}
   511→                      className="px-4 py-2 rounded-lg bg-[var(--p)] text-white text-xs font-semibold hover:opacity-90 active:scale-95 transition"
   512→                    >
   513→                      Try again
   514→                    </button>
   515→                  </div>
   516→                )}
   517→
   518→                {status === "notfound" && (
   519→                  <div className="flex flex-col items-center justify-center py-10 px-4 text-center">
   520→                    <div className="w-12 h-12 rounded-full bg-[rgba(217,119,6,0.1)] flex items-center justify-center mb-3">
   521→                      <Compass className="w-6 h-6 text-[var(--yl)]" />
   522→                    </div>
   523→                    <p className="text-sm font-semibold text-[var(--t1)] mb-1">
   524→                      No entry for <span className="font-mono">“{query}”</span>
   525→                    </p>
   526→                    <p className="text-xs text-[var(--t2)] mb-4 max-w-xs">
   527→                      This word isn&apos;t in the dictionary. Check the spelling or try a variant.
   528→                    </p>
   529→                    {suggestion && (
   530→                      <button
   531→                        onClick={() => lookup(suggestion)}
   532→                        className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-[rgba(217,119,6,0.1)] border border-[rgba(217,119,6,0.3)] text-xs font-semibold text-[var(--yl)] hover:bg-[rgba(217,119,6,0.15)] transition"
   533→                      >

   602→function PartOfSpeechChip({ pos }: { pos: string }) {
   603→  // Subtle color coding by part of speech — all muted to fit the minimal theme
   604→  const palette: Record<string, { bg: string; fg: string; border: string }> = {
   605→    noun: { bg: "rgba(8,145,178,0.08)", fg: "#0891b2", border: "rgba(8,145,178,0.2)" },
   606→    verb: { bg: "rgba(5,150,105,0.08)", fg: "#059669", border: "rgba(5,150,105,0.2)" },
   607→    adjective: { bg: "rgba(217,119,6,0.08)", fg: "#d97706", border: "rgba(217,119,6,0.2)" },
   608→    adverb: { bg: "rgba(139,92,246,0.08)", fg: "#7c3aed", border: "rgba(139,92,246,0.2)" },
   609→  };
   610→  const c = palette[pos.toLowerCase()] || { bg: "var(--overlay-2)", fg: "var(--t2)", border: "var(--border)" };
   611→  return (
   612→    <span
   613→      className="inline-flex items-center px-2 py-0.5 rounded-full text-[10px] font-mono font-bold uppercase tracking-wider border"
   614→      style={{ background: c.bg, color: c.fg, borderColor: c.border }}
   615→    >
   616→      {pos}
   617→    </span>
   618→  );
   619→}

   621→function ResultView({
   622→  result,
