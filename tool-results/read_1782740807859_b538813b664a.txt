    27→/**
    28→ * Returns the "primary audio" text for a lesson step — the text that should
    29→ * be spoken when the user presses Space (or wants a quick replay).
    30→ * Returns `null` for step types that have no meaningful auto-play audio.
    31→ */
    32→function getPrimaryAudioText(step: LessonStep | undefined): string | null {
    33→  if (!step) return null;
    34→  switch (step.type) {
    35→    case "intro":
    36→      return step.title;
    37→    case "example":
    38→      return step.phrase;
    39→    case "mouth-diagram":
    40→      return step.exampleWord || null;
    41→    case "compare":
    42→      return step.nativePhrase;
    43→    case "stress-bars":
    44→      return step.word;
    45→    case "rhythm":
    46→      return step.phrase;
    47→    case "linking":
    48→      return step.words.join(" ");
    49→    case "shadow":
    50→      return step.phrase;
    51→    case "intonation":
    52→      return step.phrase;
    53→    case "practice":
    54→      return step.phrase;
    55→    // No primary audio: concept, vowel-chart, tap-pronounce, tip, quiz, completion
    56→    default:
    57→      return null;
    58→  }
    59→}

    62→  wave: "🌊",
    63→  mouth: "👄",
    64→  "ipa-chart": "🔤",
    65→  "vowel-chart": "🎯",
    66→  "compare-wave": "📊",
    67→  rhythm: "🎵",
    68→  "phoneme-grid": "🔡",
    69→  "stress-bars": "📈",
    70→  linking: "🔗",
    71→  shadow: "🪞",
    72→  intonation: "📐",
    73→  "emoji-burst": "✨",
    74→};
    75→
    76→export function LessonModal({ lesson, onClose, onNext }: Props) {
    77→  const [stepIdx, setStepIdx] = useState(0);
    78→  const [direction, setDirection] = useState(1);
    79→  const [quizAnswer, setQuizAnswer] = useState<number | null>(null);
    80→  const [practiceScore, setPracticeScore] = useState<number | null>(null);
    81→  const [recording, setRecording] = useState(false);
    82→  const [showConfetti, setShowConfetti] = useState(false);
    83→  // Briefly shows a "⌨ Press Space to play" hint when a step with audio loads.
    84→  const [showSpaceHint, setShowSpaceHint] = useState(false);
    85→  // Track the previous stepIdx so we can reset per-step interactive state
    86→  // when the step changes — using the official "adjust state during render"
    87→  // pattern instead of setState-in-effect (react-hooks/set-state-in-effect).
    88→  const [prevStepIdx, setPrevStepIdx] = useState(stepIdx);
    89→  const accent = useAppStore((s) => s.accent);
    90→  const completeLesson = useAppStore((s) => s.completeLesson);
    91→  const lessons = useAppStore((s) => s.lessons);
    92→  const addSpeakingTime = useAppStore((s) => s.addSpeakingTime);
    93→
    94→  const step = lesson.steps[stepIdx];
    95→  const isLast = stepIdx === lesson.steps.length - 1;
    96→  const isFirst = stepIdx === 0;
    97→  const totalSteps = lesson.steps.length;
    98→  const pct = Math.round(((stepIdx + 1) / totalSteps) * 100);
    99→
   100→  // Preload voices on mount
   101→  useEffect(() => {
   102→    loadVoices();
   103→  }, []);
   104→
   105→  // Reset per-step interactive state when stepIdx changes.
   106→  // Official React pattern: adjust state during render based on a tracked
   107→  // previous value. React re-renders immediately without committing the
   108→  // stale intermediate state. (Avoids setState-in-effect cascades.)
   109→  if (stepIdx !== prevStepIdx) {
   110→    setPrevStepIdx(stepIdx);
   111→    setQuizAnswer(null);
   112→    setPracticeScore(null);
   113→    setRecording(false);
   114→    // Reset the Space-to-play hint based on whether the new step has audio.
   115→    setShowSpaceHint(!!getPrimaryAudioText(lesson.steps[stepIdx]));
   116→  }
   117→
   118→  // Stop TTS on unmount or step change
   119→  useEffect(() => {
   120→    return () => stopSpeaking();
   121→  }, [stepIdx]);
   122→
   123→  // Auto-hide the "Press Space to play" hint after 3s when it's showing.
   124→  // The hint itself is shown/hidden via the render-time adjustment above
   125→  // (avoids setState-in-effect cascades).
   126→  useEffect(() => {
   127→    if (!showSpaceHint) return;
   128→    const t = setTimeout(() => setShowSpaceHint(false), 3000);
   129→    return () => clearTimeout(t);
   130→  }, [showSpaceHint]);
   131→
   132→  const handleSpeak = useCallback(
   133→    (text: string) => {
   134→      speak(text, { accent, rate: 0.95 });
   135→    },
   136→    [accent]
   137→  );
   138→
   139→  const goNext = useCallback(() => {
   140→    if (isLast) {
   141→      // already at completion step
   142→      return;
   143→    }
   144→    setDirection(1);
   145→    setStepIdx((i) => i + 1);
   146→  }, [isLast]);
   147→
   148→  const goPrev = useCallback(() => {
   149→    if (isFirst) return;
   150→    setDirection(-1);
   151→    setStepIdx((i) => i - 1);
   152→  }, [isFirst]);
   153→
   154→  // Mark completion when reaching completion step
   155→  const completionStep = lesson.steps.find((s) => s.type === "completion");
   156→  const alreadyCompleted = lessons[lesson.id]?.completed;
   157→
   158→  const handleComplete = useCallback(() => {
   159→    if (alreadyCompleted) return;
   160→    if (completionStep?.type === "completion") {
   161→      completeLesson(
   162→        lesson.id,
   163→        practiceScore ?? 85,
   164→        completionStep.xp,
   165→        completionStep.badge
   166→      );
   167→      addSpeakingTime(8);
   168→      setShowConfetti(true);
   169→      setTimeout(() => setShowConfetti(false), 4000);
   170→    }
   171→  }, [alreadyCompleted, completionStep, completeLesson, lesson.id, practiceScore, addSpeakingTime]);
   172→
   173→  // Auto-trigger completion when user reaches the completion step.
   174→  // handleComplete calls setState (setShowConfetti + store actions), so we
   175→  // defer it out of the synchronous effect body to satisfy
   176→  // react-hooks/set-state-in-effect (no cascading renders).
   177→  useEffect(() => {
   178→    if (step?.type !== "completion" || alreadyCompleted) return;
   179→    const id = setTimeout(() => handleComplete(), 0);
   180→    return () => clearTimeout(id);
   181→  }, [step, alreadyCompleted, handleComplete]);
   182→
   183→  // ESC to close, Arrows to navigate, Space to play step audio
   184→  useEffect(() => {
   185→    const onKey = (e: KeyboardEvent) => {
   186→      if (e.key === "Escape") {
   187→        onClose();
   188→        return;
   189→      }
   190→      if (e.key === "ArrowRight") {
   191→        goNext();
   192→        return;
   193→      }
   194→      if (e.key === "ArrowLeft") {
   195→        goPrev();
   196→        return;
   197→      }
   198→      // Space → play the current step's primary audio
   199→      if (e.key === " " || e.code === "Space") {
   200→        const target = e.target as HTMLElement | null;
   201→        const isTyping =
   202→          target &&
   203→          (target.tagName === "INPUT" ||
   204→            target.tagName === "TEXTAREA" ||
   205→            target.isContentEditable);
   206→        if (isTyping) return;
   207→
   208→        // Don't capture Space when another overlay is open that needs it
   209→        // (AI Coach chat input, or this very shortcuts overlay).
   210→        const coachOpen = document.querySelector(
   211→          '[aria-label="AccentAI Coach chat"]'
   212→        );
   213→        if (coachOpen) return;
   214→        const overlay = document.getElementById("shortcuts-overlay");
   215→        if (overlay && !overlay.classList.contains("hidden")) return;
   216→
   217→        // If an interactive element (button / link) is focused, let the
   218→        // browser's native Space-to-click behaviour win so users who tab
   219→        // through the UI can still activate controls with Space.
   220→        const active = document.activeElement as HTMLElement | null;
   221→        if (
   222→          active &&
   223→          (active.tagName === "BUTTON" ||
   224→            active.tagName === "A" ||
   225→            active.getAttribute("role") === "button")
   226→        ) {
   227→          return;
   228→        }
   229→
   230→        const text = getPrimaryAudioText(step);
   231→        if (text) {
   232→          e.preventDefault();
   233→          handleSpeak(text);
   234→        }
   235→      }
   236→    };
   237→    window.addEventListener("keydown", onKey);
   238→    return () => window.removeEventListener("keydown", onKey);
   239→  }, [goNext, goPrev, onClose, step, handleSpeak]);
   240→
   241→  return (
   242→    <motion.div
   243→      initial={{ opacity: 0 }}
   244→      animate={{ opacity: 1 }}
   245→      exit={{ opacity: 0 }}
   246→      className="fixed inset-0 z-[200] bg-[var(--bg)] flex flex-col"
   247→    >
   248→      {showConfetti && <Confetti count={100} />}
   249→
   250→      {/* Header */}
   251→      <div className="flex items-center justify-between px-4 py-3 border-b border-[var(--border)] bg-[var(--bg2)]/80 backdrop-blur safe-top">
   252→        <button
   253→          onClick={onClose}
   254→          className="w-9 h-9 rounded-xl flex items-center justify-center hover:bg-[var(--card-h)] transition"
   255→          aria-label="Close lesson"
   256→        >
   257→          <X className="w-5 h-5" />
   258→        </button>
   259→        <div className="flex-1 px-3 text-center">
   260→          <div className="text-[10px] text-[var(--t3)] uppercase tracking-wider font-mono">
   261→            Phase {lesson.phaseId + 1} · Lesson {lesson.lessonIndex + 1}
   262→          </div>
   263→          <div className="text-sm font-d font-semibold truncate">{lesson.title}</div>
   264→        </div>
   265→        <div className="flex items-center gap-1.5">
   266→          <span className="text-xs font-mono text-[var(--t2)]">{stepIdx + 1}/{totalSteps}</span>
   267→          <ProgressRing pct={pct} size={36} stroke={3} />
   268→        </div>
   269→      </div>
   270→
   271→      {/* Progress bar with glow */}
   272→      <div className="h-1.5 bg-[var(--bg2)] relative">
   273→        <motion.div
   274→          className="h-full"
   275→          style={{ background: "var(--grad-btn)", boxShadow: "0 0 8px rgba(99,102,241,0.6)" }}
   276→          initial={{ width: 0 }}
   277→          animate={{ width: `${pct}%` }}
   278→          transition={{ duration: 0.4 }}
   279→        />
   280→        {/* Shimmer on progress bar */}
   281→        <motion.div
   282→          className="absolute top-0 h-full w-8 bg-gradient-to-r from-transparent via-white/30 to-transparent"
   283→          animate={{ x: ["-100%", "500%"] }}
   284→          transition={{ duration: 2, repeat: Infinity, repeatDelay: 3, ease: "easeInOut" }}
   285→          style={{ left: `${pct - 5}%` }}
   286→        />
   287→      </div>
   288→
   289→      {/* Step progress dots */}
   290→      <div className="px-4 py-2 bg-[var(--bg2)]/60 border-b border-[var(--border)]">
   291→        <div className="flex items-center gap-1 max-w-2xl mx-auto overflow-x-auto scrollbar-none">
   292→          {lesson.steps.map((s, i) => {
   293→            const isCurrent = i === stepIdx;
   294→            const isPast = i < stepIdx;
   295→            const stepIcon: Record<string, string> = {
   296→              intro: "👋", concept: "📖", example: "💬", "mouth-diagram": "👄",
   297→              "vowel-chart": "🎯", compare: "📊", "stress-bars": "📈", rhythm: "🎵",
   298→              linking: "🔗", shadow: "🪞", intonation: "📐", "tap-pronounce": "👆",
   299→              tip: "💡", practice: "🎙", quiz: "❓", completion: "🏆"
   300→            };
   301→            return (
   302→              <motion.button
   303→                key={i}
   304→                onClick={() => { setDirection(i > stepIdx ? 1 : -1); setStepIdx(i); }}
   305→                className={`shrink-0 w-7 h-7 rounded-full flex items-center justify-center text-[10px] transition-all relative ${
   306→                  isCurrent
   307→                    ? "bg-[var(--grad-btn)] text-white scale-110 shadow-[0_0_12px_rgba(99,102,241,0.5)]"
   308→                    : isPast
   309→                    ? "bg-[rgba(16,185,129,0.2)] text-[#10b981]"
   310→                    : "bg-[var(--card)] text-[var(--t3)] border border-[var(--border)]"
   311→                }`
   312→                }
   313→                whileHover={{ scale: 1.15 }}
   314→                whileTap={{ scale: 0.9 }}
   315→                title={`${s.type}: ${s.title || s.type}`}
   316→              >
   317→                <span className="relative z-10">{stepIcon[s.type] || "•"}</span>
   318→                {isCurrent && (
   319→                  <motion.span
   320→                    className="absolute inset-0 rounded-full border-2 border-[var(--p)]"
   321→                    animate={{ scale: [1, 1.3, 1], opacity: [0.5, 0, 0.5] }}
   322→                    transition={{ duration: 2, repeat: Infinity }}
   323→                  />
   324→                )}
   325→              </motion.button>
   326→            );
   327→          })}
   328→        </div>
   329→      </div>
   330→
   331→      {/* Step content */}
   332→      <div className="flex-1 overflow-y-auto">
   333→        <AnimatePresence mode="wait" custom={direction}>
   334→          <motion.div
   335→            key={stepIdx}
   336→            custom={direction}
   337→            initial={{ opacity: 0, x: direction * 30 }}
   338→            animate={{ opacity: 1, x: 0 }}
   339→            exit={{ opacity: 0, x: direction * -30 }}
   340→            transition={{ duration: 0.25 }}
   341→            className="max-w-2xl mx-auto px-5 py-6 pb-32"
   342→          >
   343→            <StepRenderer
   344→              step={step}
   345→              speak={handleSpeak}
   346→              quizAnswer={quizAnswer}
   347→              setQuizAnswer={setQuizAnswer}
   348→              practiceScore={practiceScore}
   349→              setPracticeScore={setPracticeScore}
   350→              recording={recording}
   351→              setRecording={setRecording}
   352→              isLast={isLast}
   353→              onComplete={handleComplete}
   354→              onNext={onNext}
   355→            />
   356→          </motion.div>
   357→        </AnimatePresence>
   358→      </div>
   359→
   360→      {/* "Press Space to play" hint — appears briefly above the footer when a
   361→          step with audio loads. pointer-events-none so it never blocks taps. */}
   362→      <AnimatePresence>
   363→        {showSpaceHint && (
   364→          <motion.div
   365→            initial={{ opacity: 0, y: 14, scale: 0.92 }}
   366→            animate={{ opacity: 1, y: 0, scale: 1 }}
   367→            exit={{ opacity: 0, y: 14, scale: 0.92 }}
   368→            transition={{ type: "spring", stiffness: 320, damping: 24 }}
   369→            className="absolute left-1/2 -translate-x-1/2 bottom-24 z-10 pointer-events-none"
   370→          >
   371→            <div className="flex items-center gap-2 px-3.5 py-2 rounded-full bg-[var(--card)]/95 border border-[var(--border2)] backdrop-blur-md shadow-[0_6px_24px_rgba(0,0,0,0.45)]">
   372→              <span className="text-sm leading-none" aria-hidden="true">⌨</span>
   373→              <kbd className="font-mono text-[10px] font-bold px-1.5 py-0.5 rounded bg-[var(--bg2)] border border-[var(--border2)] text-[var(--p3)] shadow-[0_1px_0_var(--border2)]">
   374→                Space
   375→              </kbd>
   376→              <span className="text-xs text-[var(--t2)] font-medium">Press Space to play</span>
   377→            </div>
   378→          </motion.div>
   379→        )}
   380→      </AnimatePresence>
   381→
   382→      {/* Footer nav */}
   383→      <div className="border-t border-[var(--border)] bg-[var(--bg2)]/95 backdrop-blur px-4 py-3 flex items-center gap-3 safe-bottom">
   384→        <button
   385→          onClick={goPrev}
   386→          disabled={isFirst}
   387→          className="px-4 py-2.5 rounded-xl flex items-center gap-1.5 text-sm font-semibold disabled:opacity-30 disabled:cursor-not-allowed hover:bg-[var(--card-h)] transition"
   388→        >
   389→          <ChevronLeft className="w-4 h-4" /> Back
   390→        </button>
   391→        <div className="flex-1" />
   392→        {step?.type === "completion" ? (
   393→          onNext ? (
   394→            <button
   395→              onClick={onNext}
   396→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   397→            >
   398→              Next Lesson <ChevronRight className="w-4 h-4" />
   399→            </button>
   400→          ) : (
   401→            <button
   402→              onClick={onClose}
   403→              className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   404→            >
   405→              <Check className="w-4 h-4" /> Finish
   406→            </button>
   407→          )
   408→        ) : (
   409→          <button
   410→            onClick={goNext}
   411→            className="px-5 py-2.5 rounded-xl bg-[var(--grad-btn)] text-white text-sm font-semibold flex items-center gap-1.5 hover:opacity-90 transition"
   412→          >
   413→            {isLast ? "Complete" : "Continue"} <ChevronRight className="w-4 h-4" />
   414→          </button>
   415→        )}
   416→      </div>
   417→    </motion.div>
   418→  );
   419→}

   421→// ─── Step Renderer ───
   422→
   423→interface StepRendererProps {
   424→  step: LessonStep;
   425→  speak: (text: string) => void;
   426→  quizAnswer: number | null;
   427→  setQuizAnswer: (n: number | null) => void;
   428→  practiceScore: number | null;
   429→  setPracticeScore: (n: number | null) => void;
   430→  recording: boolean;
   431→  setRecording: (b: boolean) => void;
   432→  isLast: boolean;
   433→  onComplete: () => void;
   434→  onNext?: () => void;
   435→}
   436→
   437→function StepRenderer(props: StepRendererProps) {
   438→  const { step, speak } = props;
   439→
   440→  switch (step.type) {
   441→    case "intro":
   442→      return <IntroStepView step={step} speak={speak} />;
   443→    case "concept":
   444→      return <ConceptStepView step={step} />;
   445→    case "example":
   446→      return <ExampleStepView step={step} speak={speak} />;
   447→    case "mouth-diagram":
   448→      return <MouthDiagram step={step} speak={speak} />;
   449→    case "vowel-chart":
   450→      return <VowelChart step={step} speak={speak} />;
   451→    case "compare":
   452→      return <CompareWave step={step} speak={speak} />;
   453→    case "stress-bars":
   454→      return <StressBars step={step} speak={speak} />;
   455→    case "rhythm":
   456→      return <RhythmBeats step={step} speak={speak} />;
   457→    case "linking":
   458→      return <LinkingDiagram step={step} speak={speak} />;
   459→    case "intonation":
   460→      return <IntonationContour step={step} speak={speak} />;
   461→    case "shadow":
   462→      return <ShadowStepView step={step} speak={speak} />;
   463→    case "tap-pronounce":
   464→      return <TapPronounceStepView step={step} speak={speak} />;
   465→    case "tip":
   466→      return <TipStepView step={step} />;
   467→    case "practice":
   468→      return <PracticeStepView {...props} />;
   469→    case "quiz":
   470→      return <QuizStepView {...props} />;
   471→    case "completion":
   472→      return <CompletionStepView step={step} onNext={props.onNext} />;
   473→    default:
   474→      return <div>Unknown step type</div>;
   475→  }
   476→}

   478→// ─── Individual step views ───
   479→
   480→function IntroStepView({ step, speak }: { step: Extract<LessonStep, { type: "intro" }>; speak: (t: string) => void }) {
   481→  return (
   482→    <div className="space-y-5 text-center relative">
   483→      {/* Subtle background glow */}
   484→      <div className="absolute inset-0 pointer-events-none">
   485→        <div className="absolute top-1/4 left-1/2 -translate-x-1/2 w-64 h-64 rounded-full bg-[rgba(99,102,241,0.06)] blur-[60px]" />
   486→      </div>
   487→
   488→      <motion.div
   489→        initial={{ scale: 0.7, opacity: 0, rotate: -10 }}
   490→        animate={{ scale: 1, opacity: 1, rotate: 0 }}
   491→        transition={{ type: "spring", stiffness: 200, damping: 14 }}
   492→        className="text-7xl mb-2 relative z-10 animate-gentle-float"
   493→      >
   494→        {step.emoji || VISUAL_EMOJI[step.visual] || "✨"}
   495→      </motion.div>
   496→      <motion.div
   497→        initial={{ opacity: 0, y: 15 }}
   498→        animate={{ opacity: 1, y: 0 }}
   499→        transition={{ delay: 0.2 }}
   500→        className="relative z-10"
   501→      >
   502→        <div className="text-[10px] uppercase tracking-[0.2em] text-[var(--t3)] font-mono mb-2">
   503→          Lesson Introduction
   504→        </div>
   505→        <h1 className="font-d text-3xl font-bold mb-2">
   506→          <span className="grad-text">{step.title}</span>
   507→        </h1>
   508→        <p className="text-[var(--t2)] text-base mb-4">{step.subtitle}</p>
   509→        <p className="text-[var(--t2)] text-sm leading-relaxed max-w-md mx-auto">
   510→          {step.description}
   511→        </p>
   512→      </motion.div>
   513→      <motion.div
   514→        initial={{ opacity: 0, y: 10 }}

   953→function CompletionStepView({
   954→  step,
   955→  onNext,
   956→}: {
   957→  step: Extract<LessonStep, { type: "completion" }>;
   958→  onNext?: () => void;
   959→}) {
   960→  const [displayXp, setDisplayXp] = useState(0);
   961→  const targetXp = step.xp;
   962→
   963→  // Animated XP counter
   964→  useEffect(() => {
   965→    if (targetXp <= 0) return;
   966→    const duration = 1200; // ms
   967→    const steps = 30;
   968→    const interval = duration / steps;
   969→    let current = 0;
   970→    const increment = targetXp / steps;
   971→    const timer = setInterval(() => {
   972→      current += increment;
   973→      if (current >= targetXp) {
   974→        setDisplayXp(targetXp);
   975→        clearInterval(timer);
   976→      } else {
   977→        setDisplayXp(Math.round(current));
   978→      }
   979→    }, interval);
   980→    return () => clearInterval(timer);
   981→  }, [targetXp]);
   982→
   983→  return (
   984→    <div className="text-center space-y-6 py-8 relative">
   985→      {/* Background celebration particles */}
   986→      {[...Array(12)].map((_, i) => (
   987→        <motion.div
   988→          key={i}
